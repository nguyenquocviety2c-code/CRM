   379→   379→      const startMs = dayBase + min * 60 * 1000;
   380→   380→      const endMs = startMs + durationMs;
   381→   381→      const overlaps = intervals.some((iv) => startMs < iv.endMs && iv.startMs < endMs);
   382→   382→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   383→   383→      const mStr = String(min % 60).padStart(2, "0");

   386→   386→        hm[hStr].add(mStr);
   387→   387→      } else {
   388→   388→        hourHasFeasible[hStr] = true;
   389→   389→      }
   390→   390→    }
   391→   391→    for (let h = 0; h < 24; h++) {
   392→   392→      const hStr = String(h).padStart(2, "0");
   393→   393→      if (!hourHasFeasible[hStr]) hh.add(hStr);
   394→   394→    }
   395→   395→    return { hiddenHours: hh, hiddenMinutes: hm };
