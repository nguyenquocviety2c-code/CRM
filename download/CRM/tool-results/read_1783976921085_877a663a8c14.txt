    66→
    67→  // --- Form state ---------------------------------------------------------

    70→  const [bookingDate, setBookingDate] = useState(""); // "DD/MM/YYYY"

    72→  const [serviceId, setServiceId] = useState("");

    74→  const [bookingTime, setBookingTime] = useState(""); // "HH:MM"
    75→  const [submitting, setSubmitting] = useState(false);
    76→  const [success, setSuccess] = useState(false);
    77→  const [error, setError] = useState("");
    78→  // Existing-booking confirmation: when the customer (by phone) already has
    79→  // non-cancelled bookings, we show a confirmation prompt listing them before
    80→  // actually creating a new booking. `pendingExisting` holds the list while
    81→  // the user decides; `skipExistingCheck` is set true after they confirm so
    82→  // the re-submit skips the check.
    83→  const [pendingExisting, setPendingExisting] = useState<
    84→    Array<{ date: string; time: string; status: string; branchName: string; services: Array<{ name: string; staffName: string }> }>
    85→  >([]);
    86→  const [skipExistingCheck, setSkipExistingCheck] = useState(false);
    87→  // Customer autocomplete: when the user types a name OR phone prefix (≥2
    88→  // chars), fetch matching customers and show a dropdown. Selecting a result
    89→  // fills BOTH name + phone. `activeField` tracks which input the user is
    90→  // currently typing in so the dropdown anchors under it.
    91→  const [suggestions, setSuggestions] = useState<
    92→    Array<{ id: string; name: string; phone: string | null; code: string | null }>
    93→  >([]);
    94→  const [showSuggestions, setShowSuggestions] = useState(false);
    95→  const [activeField, setActiveField] = useState<"name" | "phone" | null>(null);
    96→
    97→  // --- Fetch branches (customer-facing page has no admin BranchSelector) ---
    98→  // The branch list is needed to render the dropdown below. We also populate
    99→  // the store so downstream queries (categories/staff/bookings) can use it.
   100→  useQuery({
   101→    queryKey: ["dat-lich-branches"],
   102→    queryFn: async () => {
   103→      const res = await fetch("/api/supabase/branches?active=true");
   104→      const json = await res.json();
   105→      if (json.ok && Array.isArray(json.data)) {
   106→        setBranches(json.data);
   107→        // Auto-select the first active branch if nothing is selected yet.
   108→        if (!selectedBranchId && json.data.length > 0) {
   109→          setSelectedBranchId(json.data[0].id);
   110→        }
   111→      }
   112→      return json.data || [];
   113→    },
   114→  });
   115→
   116→  // --- Reference data: service categories ---------------------------------
   117→  const { data: categoriesData } = useQuery<ServiceCategory[]>({
   118→    queryKey: ["dat-lich-categories", selectedBranchId],
   119→    queryFn: async () => {
   120→      const res = await fetch("/api/supabase/service-categories?active=true");
   121→      const json = await res.json();
   122→      const all = (json.data || []) as ServiceCategory[];
   123→      // Keep only categories available at the selected branch.
   124→      if (!selectedBranchId || selectedBranchId === "all") return all;
   125→      return all.filter((c) => (c.branches || []).includes(selectedBranchId));
   126→    },
   127→    enabled: !!selectedBranchId,
   128→  });
   129→
   130→  // --- Reference data: services -------------------------------------------
   131→  const { data: servicesData } = useQuery<Service[]>({
   132→    queryKey: ["dat-lich-services"],
   133→    queryFn: async () => {
   134→      const res = await fetch("/api/supabase/services?limit=200");
   135→      const json = await res.json();
   136→      return (json.data || []) as Service[];
   137→    },
   138→  });
   139→
   140→  // --- Reference data: staff (hairdressers only) --------------------------
   141→  const { data: staffData } = useQuery<Staff[]>({
   142→    queryKey: ["dat-lich-staff", selectedBranchId],
   143→    queryFn: async () => {
   144→      if (!selectedBranchId) return [];
   145→      const res = await fetch(
   146→        `/api/supabase/staff?branch_id=${selectedBranchId}&active=true&limit=200`
   147→      );
   148→      const json = await res.json();
   149→      if (!json.ok) return [];
   150→      return (json.data as Array<Record<string, unknown>>)
   151→        .filter((s) => {
   152→          const groupName = (s.group as { name?: string } | null)?.name;
   153→          return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   154→        })
   155→        .map((s) => ({
   156→          id: s.id as string,
   157→          name: s.name as string,
   158→          groupName: (s.group as { name?: string } | null)?.name,
   159→        }));
   160→    },
   161→    enabled: !!selectedBranchId,
   162→  });
   163→
   164→  // --- Phone lookup: determine if the customer is new or returning -------
   165→  // A "returning" (old) customer is one whose exact phone number already
   166→  // exists in the customers table. This drives the service-category filter
   167→  // (new customers see the "Dành cho khách hàng mới" cut; old customers see
   168→  // the regular "Dịch Vụ Cắt").
   169→  const trimmedPhone = customerPhone.trim();
   170→  const { data: phoneLookup } = useQuery<
   171→    { id: string; phone?: string | null }[] | null
   172→  >({
   173→    queryKey: ["dat-lich-phone-lookup", trimmedPhone],
   174→    queryFn: async () => {
   175→      if (!trimmedPhone) return null;
   176→      const res = await fetch(
   177→        `/api/supabase/customers?search=${encodeURIComponent(trimmedPhone)}&limit=20`
   178→      );
   179→      const json = await res.json();
   180→      if (!json.ok) return null;
   181→      return (json.data || []) as { id: string; phone?: string | null }[];
   182→    },
   183→    enabled: trimmedPhone.length >= 9,
   184→  });
   185→  const isOldCustomer = useMemo(() => {
   186→    if (!trimmedPhone || !phoneLookup) return false;
   187→    return phoneLookup.some((c) => c.phone === trimmedPhone);
   188→  }, [trimmedPhone, phoneLookup]);
   189→
   190→  // --- Customer autocomplete ----------------------------------------------
   191→  // Debounced search: when the user types ≥2 chars in EITHER the name or phone
   192→  // field, fetch matching customers (by name prefix OR phone prefix) and show
   193→  // a dropdown. Selecting a suggestion fills both name + phone.
   194→  useEffect(() => {
   195→    // Determine the search term based on which field the user is typing in.
   196→    const term =
   197→      activeField === "phone"
   198→        ? customerPhone.trim()
   199→        : activeField === "name"
   200→          ? customerName.trim()
   201→          : "";
   202→    let cancelled = false;
   203→    const t = setTimeout(async () => {
   204→      if (term.length < 2) {
   205→        if (!cancelled) {
   206→          setSuggestions([]);
   207→          setShowSuggestions(false);
   208→        }
   209→        return;
   210→      }
   211→      try {
   212→        const res = await fetch(
   213→          `/api/supabase/customers?search=${encodeURIComponent(term)}&limit=10`
   214→        );
   215→        const json = await res.json();
   216→        if (cancelled) return;
   217→        if (json.ok && Array.isArray(json.data)) {
   218→          // Filter to prefix matches on name OR phone (the API's `search`
   219→          // does an ilike contains; we refine to prefix for relevance).
   220→          const lower = term.toLowerCase();
   221→          const matches = (json.data as Array<{ id: string; name: string; phone?: string | null; code?: string | null }>)
   222→            .filter((c) => {
   223→              const nameMatch = (c.name || "").toLowerCase().startsWith(lower);
   224→              const phoneMatch = (c.phone || "").startsWith(term);
   225→              return nameMatch || phoneMatch;
   226→            })
   227→            .map((c) => ({ id: c.id, name: c.name, phone: c.phone ?? null, code: c.code ?? null }));
   228→          if (!cancelled) {
   229→            setSuggestions(matches);
   230→            setShowSuggestions(matches.length > 0);
   231→          }
   232→        }
   233→      } catch {
   234→        if (!cancelled) {
   235→          setSuggestions([]);
   236→          setShowSuggestions(false);
   237→        }
   238→      }
   239→    }, 250); // 250ms debounce
   240→    return () => {
   241→      cancelled = true;
   242→      clearTimeout(t);
   243→    };
   244→  }, [customerName, customerPhone, activeField]);
   245→
   246→  const selectSuggestion = (s: { name: string; phone: string | null }) => {
   247→    setCustomerName(s.name);
   248→    setCustomerPhone(s.phone || "");
   249→    setShowSuggestions(false);
   250→    setActiveField(null);
   251→  };
   252→  const customerType: "new" | "old" = isOldCustomer ? "old" : "new";
   253→
   254→  // Categories filtered by branch AND customer type (mirrors booking-dialog):
   255→  // - New customer (phone not in DB): hide "Dịch Vụ Cắt", keep new-customer cut.
   256→  // - Old customer (phone in DB): hide "Dành cho khách hàng mới - DV Cắt".
   257→  const categories = useMemo(() => {
   258→    const all = categoriesData || [];
   259→    const byBranch =
   260→      !selectedBranchId || selectedBranchId === "all"
   261→        ? all
   262→        : all.filter((c) => (c.branches || []).includes(selectedBranchId));
   263→    return byBranch.filter((c) => {
   264→      const name = c.name.toLowerCase();
   265→      const isNewCustomerCut = name.includes("dành cho khách hàng mới");
   266→      const isRegularCut =
   267→        name.includes("dịch vụ cắt") || name === "dịch vụ cắt";
   268→      if (customerType === "new") {
   269→        if (isRegularCut && !isNewCustomerCut) return false;
   270→      } else {
   271→        if (isNewCustomerCut) return false;
   272→      }
   273→      return true;
   274→    });
   275→  }, [categoriesData, selectedBranchId, customerType]);
   276→
   277→  const allServices = servicesData || [];
   278→  const allStaff = staffData || [];
   279→
   280→  // If the selected category is no longer visible after the customer-type
   281→  // filter changes (e.g. phone typed → old customer → "Dành cho KH mới" hidden),
   282→  // reset the service selection chain to avoid a stale hidden selection.
   283→  useEffect(() => {
   284→    if (categoryId && !categories.some((c) => c.id === categoryId)) {
   285→      setCategoryId("");
   286→      setServiceId("");
   287→      setStaffId("");
   288→      setBookingTime("");
   289→    }
   290→  }, [categories, categoryId]);
   291→
   292→  // Services filtered by the selected category.
   293→  const servicesInCategory = useMemo(
   294→    () => allServices.filter((s) => !categoryId || s.category_id === categoryId),
   295→    [allServices, categoryId]
   296→  );
   297→
   298→  // Staff filtered by the selected service (name-based narrowing, mirrors
   299→  // booking-dialog: "Creative Director" in name → only that group, etc.).
   300→  const staffForService = useMemo(() => {
   301→    if (!serviceId) return allStaff;
   302→    const svc = allServices.find((s) => s.id === serviceId);
   303→    if (!svc) return allStaff;
   304→    const n = (svc.name || "").toLowerCase();
   305→    if (n.includes("creative director")) {
   306→      return allStaff.filter((s) => s.groupName === "Creative Director");
   307→    }
   308→    if (n.includes("artist")) {
   309→      return allStaff.filter((s) => s.groupName === "Artist");
   310→    }
   311→    return allStaff;
   312→  }, [allStaff, allServices, serviceId]);
   313→
   314→  // The selected service (for its duration — needed for feasibility).
   315→  const selectedService = useMemo(
   316→    () => allServices.find((s) => s.id === serviceId),
   317→    [allServices, serviceId]
   318→  );
   319→
   320→  // --- Feasible-time computation (mirrors booking-dialog) -----------------
   321→  // Fetch the day's existing bookings (for the branch) and compute which
   322→  // HH:MM start times would overlap the selected staff's existing bookings.
   323→  const isoDay = useMemo(() => {
   324→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   325→    return m ? `${m[3]}-${m[2]}-${m[1]}` : null;
   326→  }, [bookingDate]);
   327→
   328→  const { data: dayBookings } = useQuery<ExistingBooking[]>({
   329→    queryKey: ["dat-lich-day-bookings", isoDay, selectedBranchId],
   330→    queryFn: async () => {
   331→      if (!isoDay) return [];
   332→      const params = new URLSearchParams();
   333→      params.set("page", "1");
   334→      params.set("limit", "200");
   335→      // Timezone-safe: isoDay is a Vietnam calendar day; convert to its UTC
   336→      // range so Supabase filters the correct window (no 7-hour shift).
   337→      const dl = localDayToUtcRange(isoDay);
   338→      params.set("date_from", dl.from);
   339→      params.set("date_to", dl.to);
   340→      if (selectedBranchId) params.set("branch_id", selectedBranchId);
   341→      const res = await fetch(`/api/supabase/bookings?${params.toString()}`);
   342→      const json = await res.json();
   343→      if (!json.ok) return [];
   344→      return (json.data || []) as ExistingBooking[];
   345→    },
   346→    enabled: !!isoDay && !!staffId,
   347→  });
   348→
   349→  // Compute hiddenHours + hiddenMinutes for the TimePicker.
   350→  const { hiddenHours, hiddenMinutes } = useMemo(() => {
   351→    const empty = { hiddenHours: new Set<string>(), hiddenMinutes: {} as Record<string, Set<string>> };
   352→    if (!staffId || !isoDay || !selectedService) return empty;
   353→    const intervals: { startMs: number; endMs: number }[] = [];
   354→    for (const b of dayBookings || []) {
   355→      if (b.status === "cancelled" || b.status === "no_show") continue;
   356→      const exStart = new Date(b.date_time).getTime();
   357→      if (isNaN(exStart)) continue;
   358→      for (const s of b.services || []) {
   359→        if (s.staff_id !== staffId) continue;
   360→        const dur =
   361→          (Number(s.duration) || Number(s.service?.duration) || 60) * 60 * 1000;
   362→        intervals.push({ startMs: exStart, endMs: exStart + dur });
   363→      }
   364→    }
   365→    if (intervals.length === 0) return empty;
   366→    const durationMs = (selectedService.duration || 60) * 60 * 1000;
   367→    if (durationMs <= 0) return empty;
   368→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   369→    if (!m) return empty;
   370→    // Use the +07:00 offset so dayBase is the correct UTC instant for VN
   371→    // midnight (same fix as the booking dialog + cashier service selector).
   372→    const dayBase = new Date(`${m[3]}-${m[2]}-${m[1]}T00:00:00+07:00`).getTime();
   373→    if (isNaN(dayBase)) return empty;
   374→    const hh = new Set<string>();
   375→    const hm: Record<string, Set<string>> = {};
   376→    const hourHasFeasible: Record<string, boolean> = {};
   377→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   378→    for (let min = 0; min < 60 * 24; min++) {
   379→      const startMs = dayBase + min * 60 * 1000;
   380→      const endMs = startMs + durationMs;
   381→      const overlaps = intervals.some((iv) => startMs < iv.endMs && iv.startMs < endMs);
   382→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   383→      const mStr = String(min % 60).padStart(2, "0");
   384→      if (overlaps) {
   385→        if (!hm[hStr]) hm[hStr] = new Set<string>();
   386→        hm[hStr].add(mStr);
   387→      } else {
   388→        hourHasFeasible[hStr] = true;
   389→      }
   390→    }
   391→    for (let h = 0; h < 24; h++) {
   392→      const hStr = String(h).padStart(2, "0");
   393→      if (!hourHasFeasible[hStr]) hh.add(hStr);
   394→    }
   395→    return { hiddenHours: hh, hiddenMinutes: hm };
   396→  }, [staffId, isoDay, selectedService, dayBookings, bookingDate]);
   397→
   398→  // Reset downstream selections when an upstream one changes.
   399→  const onCategoryChange = (id: string) => {
   400→    setCategoryId(id);
   401→    setServiceId("");
   402→    setStaffId("");
   403→    setBookingTime("");
   404→  };
   405→  const onServiceChange = (id: string) => {
   406→    setServiceId(id);
   407→    setStaffId("");
   408→    setBookingTime("");
   409→  };
   410→  // Changing the branch invalidates the whole service chain — staff and
   411→  // service availability are branch-specific. Reset everything downstream.
   412→  const onBranchChange = (id: string) => {
   413→    setSelectedBranchId(id);
   414→    setCategoryId("");
   415→    setServiceId("");
   416→    setStaffId("");
   417→    setBookingTime("");
   418→  };
   419→
   420→  // Part 1 (customer info) must be fully filled before Part 2 (service) opens.
   421→  const part1Complete = !!(
   422→    customerName.trim() &&
   423→    customerPhone.trim() &&
   424→    bookingDate
   425→  );
   426→  // Khung giờ requires Nhóm DV + Dịch vụ + Kỹ thuật viên all selected first.
   427→  const timePickerReady = part1Complete && !!serviceId && !!staffId;
   428→
   429→  const canSubmit =
   430→    part1Complete &&
   431→    serviceId &&
   432→    staffId &&
   433→    bookingTime &&
   434→    !!selectedBranchId &&
   435→    !submitting;
   436→
   437→  // --- Submit: lookup/create customer, then create booking -----------------
   438→  const handleSubmit = async () => {
   439→    if (!canSubmit) return;
   440→    setError("");
   441→    setSubmitting(true);
   442→    try {
   443→      // 0. One-time offer check: if the selected category is "Dành cho khách
   444→      //    hàng mới - DV Cắt", the customer (by phone) may only book it ONCE.
   445→      //    If they already have a non-cancelled booking with this category,
   446→      //    block the submit with a message pointing to the existing booking's
   447→      //    date and CSKH for changes.
   448→      const selectedCat = categories.find((c) => c.id === categoryId);
   449→      const isNewCustomerCut =
   450→        !!selectedCat &&
   451→        selectedCat.name.toLowerCase().includes("dành cho khách hàng mới");
   452→      if (isNewCustomerCut && customerPhone.trim()) {
   453→        try {
   454→          const checkRes = await fetch(
   455→            `/api/supabase/bookings/check-new-customer-cut?phone=${encodeURIComponent(customerPhone.trim())}`
   456→          );
   457→          const checkJson = await checkRes.json();
   458→          if (checkJson.ok && checkJson.data?.exists) {
   459→            const existingDate = checkJson.data.existingDate || "";
   460→            // Staff (logged in) see the short internal notice; customers
   461→            // additionally see the CSKH contact line.
   462→            setError(
   463→              `Không thể đặt lịch vì đang có 1 lịch cắt "Dành cho khách hàng mới - DV Cắt" vào ngày ${existingDate}.` +
   464→                (isLoggedIn
   465→                  ? ""
   466→                  : " Để đặt ngày khác vui lòng liên hệ bộ phận CSKH để thay đổi lịch.")
   467→            );
   468→            setSubmitting(false);
   469→            return;
   470→          }
   471→        } catch {
   472→          /* best-effort — don't block on network errors */
   473→        }
   474→      }
   475→
   476→      // 0b. Existing-booking confirmation: if the customer (by phone) already
   477→      //     has any non-cancelled booking, show a confirmation prompt listing
   478→      //     them (date/time/services/branch) and ask "are you sure?". Only
   479→      //     proceed when the user confirms. Skipped on the re-submit after
   480→      //     confirmation (skipExistingCheck) or when there's no phone.
   481→      if (!skipExistingCheck && customerPhone.trim()) {
   482→        try {
   483→          const existRes = await fetch(
   484→            `/api/supabase/bookings/by-phone?phone=${encodeURIComponent(customerPhone.trim())}`
   485→          );
   486→          const existJson = await existRes.json();
   487→          if (existJson.ok && Array.isArray(existJson.data) && existJson.data.length > 0) {
   488→            setPendingExisting(existJson.data);
   489→            setSubmitting(false);
   490→            return; // stop — the confirmation UI takes over
   491→          }
   492→        } catch {
   493→          /* best-effort — don't block on network errors */
   494→        }
   495→      }
   496→
   497→      // 1. Look up the customer by phone (exact match). If found, reuse id.
   498→      let customerId: string | null = null;
   499→      try {
   500→        const searchRes = await fetch(
   501→          `/api/supabase/customers?search=${encodeURIComponent(customerPhone.trim())}&limit=20`
   502→        );
   503→        const searchJson = await searchRes.json();
