   386→   386→
   387→   387→    // Build per-row busy intervals + duration. Each row contributes its own

   389→   389→    const rowSpecs: Array<{
   390→   390→      intervals: { startMs: number; endMs: number }[];
   391→   391→      durationMs: number;
   392→   392→    }> = [];
   393→   393→    for (const row of activeRows) {
   394→   394→      const svc = allServices.find((s) => s.id === row.serviceId);
   395→   395→      const durationMs = (svc?.duration || 60) * 60 * 1000;
   396→   396→      if (durationMs <= 0) continue;
   397→   397→      const intervals: { startMs: number; endMs: number }[] = [];

   399→   399→        if (b.status === "cancelled" || b.status === "no_show") continue;
   400→   400→        const exStart = new Date(b.date_time).getTime();
   401→   401→        if (isNaN(exStart)) continue;
   402→   402→        for (const s of b.services || []) {
   403→   403→          if (s.staff_id !== row.staffId) continue;
   404→   404→          const dur =
   405→   405→            (Number(s.duration) || Number(s.service?.duration) || 60) * 60 * 1000;
   406→   406→          intervals.push({ startMs: exStart, endMs: exStart + dur });
   407→   407→        }
   408→   408→      }
   409→   409→      rowSpecs.push({ intervals, durationMs });
   410→   410→    }
   411→   411→    if (rowSpecs.length === 0) return empty;
   412→   412→
   413→   413→    const hh = new Set<string>();
   414→   414→    const hm: Record<string, Set<string>> = {};
   415→   415→    const hourHasFeasible: Record<string, boolean> = {};
   416→   416→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   417→   417→    for (let min = 0; min < 60 * 24; min++) {
   418→   418→      const startMs = dayBase + min * 60 * 1000;
   419→   419→      // A minute conflicts if ANY row's staff is busy during [start, start+dur].
