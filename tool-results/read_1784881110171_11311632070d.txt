    69→
    70→interface StaffColumn {
    71→  staff: Staff | null;
    72→  segments: ServiceSegment[];
    73→}
    74→
    75→/** A service-segment of a booking, assigned to one staff's timeline. A
    76→ *  segment = one or more CONSECUTIVE services performed by the SAME staff,
    77→ *  merged into a single slot. Services run back-to-back (cursor advances by
    78→ *  each service's duration), so consecutive in sort_order = adjacent in time.
    79→ *  A run of same-staff services becomes one merged slot whose height ∝ the
    80→ *  sum of their durations; different-staff services stay in separate slots
    81→ *  (each in its own staff's column). Matches the server-side conflict check in
    82→ *  /api/supabase/bookings/route.ts. */
    83→interface ServiceSegment {
    84→  booking: Booking;
    85→  /** One or more consecutive same-staff services merged into this slot. */
    86→  services: BookingServiceRow[];
    87→  staffId: string;
    88→  staffName: string;
    89→  /** 0-based index of this segment (group) within the booking's segment list. */
    90→  segmentIndex: number;
    91→  /** Total number of segments (groups) in the booking (for the "1/2" badge). */
    92→  totalSegments: number;
    93→  /** Minutes from the timeline's START_HOUR (08:00) to this segment's start.
    94→   *  May be negative when the booking starts before 08:00 — callers clamp
    95→   *  into the visible window. */
    96→  startMin: number;
    97→  /** This segment's total duration = Σ of its services' durations (minutes). */
    98→  duration: number;
    99→}
   100→
   101→/** Pixels per hour on the vertical timeline. Slot height is proportional to
   102→ *  duration. 1.5× the previous value (60 → 90) so each hour band is taller and
   103→ *  multi-service segment blocks have more room to show their content. */
   104→const PX_PER_HOUR = 90;
   105→/** Timeline starts at 08:00 and ends at 21:00 (13 hours). */
   106→const START_HOUR = 8;
   107→const END_HOUR = 21;
   108→/** Width of each staff column. The timeline scrolls horizontally when there
   109→ *  are more staff than fit on screen. */
   110→const STAFF_COL_WIDTH = 180;
   111→/** Width of the fixed "Giờ" column on the left. */
   112→const TIME_COL_WIDTH = 64;
   113→/** Minimum column width when dragging (prevents columns from vanishing). */
   114→const MIN_COL_WIDTH = 50;
   115→/** Width of the drag handle on each header cell's right edge. */
   116→const RESIZER_WIDTH = 8;

   118→// Hairdresser groups — same filter as the booking dialog. Only these groups

   120→const HAIRDRESSER_GROUPS = ["Artist", "Creative Director", "Master", "Junior"];

   122→export function BookingStaffView({
   123→  bookings,
   124→  currentDate,
   125→  dateRange,
   126→  onBookingClick,
   127→  onStatusChange,
   128→  onEdit,
   129→  onAssignStaff,
   130→  onDelete,
   131→  onShowInvoice,
   132→  onSlotClick,
   133→  branchId,
   134→}: BookingStaffViewProps) {
   135→  const { hasPermission } = useAuthStore();
   136→  const canViewCustomerPhone = hasPermission("view_customer_phone");
   137→  const canBookPastDate = hasPermission("book_past_date");
   138→  // Whether the viewed date is in the past (slot creation blocked when
   139→  // !canBookPastDate).
   140→  const isPastDate = (() => {
   141→    if (!currentDate) return false;
   142→    const today = new Date();
   143→    today.setHours(0, 0, 0, 0);
   144→    const d = new Date(currentDate);
   145→    d.setHours(0, 0, 0, 0);
   146→    return d < today;
   147→  })();
   148→  const slotLocked = isPastDate && !canBookPastDate;
   149→
   150→  // Hour-band resize (single-day timeline only). The user drags a horizontal
   151→  // handle on the "Giờ" axis to grow/shrink every hour band. Persisted.
   152→  const MIN_PX_PER_HOUR = 40;
   153→  const MAX_PX_PER_HOUR = 200;
   154→  const [pxPerHour, setPxPerHour] = useState<number>(() => {
   155→    try {
   156→      const raw = localStorage.getItem("crm-staff-timegrid-pxph");
   157→      const n = raw ? parseInt(raw, 10) : NaN;
   158→      if (!isNaN(n) && n >= MIN_PX_PER_HOUR && n <= MAX_PX_PER_HOUR) return n;
   159→    } catch { /* ignore */ }
   160→    return PX_PER_HOUR;
   161→  });
   162→  const setPxPerHourPersisted = (n: number) => {
   163→    const clamped = Math.max(MIN_PX_PER_HOUR, Math.min(MAX_PX_PER_HOUR, n));
   164→    setPxPerHour(clamped);
   165→    try { localStorage.setItem("crm-staff-timegrid-pxph", String(clamped)); } catch { /* ignore */ }
   166→  };
   167→
   168→  // Tracks which (column, hour) slot the mouse is hovering over in the
   169→  // single-day staff-column layout. Used to highlight ONLY that slot (not
   170→  // the entire column) with an orange overlay on hover.
   171→  const [hoveredSlot, setHoveredSlot] = useState<{ col: number; hour: number } | null>(null);
   172→
   173→  // Number of days in the selected range. When 1 day → use the legacy
   174→  // per-staff column layout (one column per staff). When 2+ days → use the
   175→  // DayRangeGrid (one column per day). Capped at 5 visible days; 6+ scroll.
   176→  const rangeDayCount = useMemo(() => {
   177→    if (!dateRange) return 0;
   178→    const from = new Date(dateRange.from);
   179→    from.setHours(0, 0, 0, 0);
   180→    const to = new Date(dateRange.to);
   181→    to.setHours(0, 0, 0, 0);
   182→    if (to < from) return 1;
   183→    return Math.round((to.getTime() - from.getTime()) / 86400000) + 1;
   184→  }, [dateRange]);
   185→  const useMultiDayLayout = rangeDayCount >= 2;
   186→
   187→  // Hour labels for the left axis (08:00 → 21:00).
   188→  const hours = useMemo(() => {
   189→    const h: number[] = [];
   190→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
   191→    return h;
   192→  }, []);
   193→
   194→  // Fetch ALL active staff for the selected branch (so every staff has a column,
   195→  // not just those with bookings today). Mirrors the BookingDialog's staff query.
   196→  const [allStaff, setAllStaff] = useState<Staff[]>([]);
   197→  useEffect(() => {
   198→    let cancelled = false;
   199→    async function fetchStaff() {
   200→      if (!branchId) {
   201→        setAllStaff([]);
   202→        return;
   203→      }
   204→      try {
   205→        const res = await fetch(
   206→          `/api/supabase/staff?branch_id=${encodeURIComponent(branchId)}&active=true&limit=200`
   207→        );
   208→        const json = await res.json();
   209→        if (cancelled) return;
   210→        if (!json.ok) {
   211→          setAllStaff([]);
   212→          return;
   213→        }
   214→        // Only show hairdresser groups (same filter as BookingDialog).
   215→        const list = ((json.data as Array<Record<string, unknown>>) || [])
   216→          .filter((s) => {
   217→            const groupName = (s.group as { name?: string } | null)?.name;
   218→            return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   219→          })
   220→          .map((s) => ({
   221→            id: s.id as string,
   222→            name: s.name as string,
   223→            groupName: (s.group as { name?: string } | null)?.name,
   224→          }));
   225→        setAllStaff(list);
   226→      } catch {
   227→        if (!cancelled) setAllStaff([]);
   228→      }
   229→    }
   230→    fetchStaff();
   231→    return () => { cancelled = true; };
   232→  }, [branchId]);
   233→
   234→  // Build staff columns from the FULL staff list. Each booking is split into
   235→  // SEGMENTS where each segment = a run of same-staff services merged into one
   236→  // slot. PARALLEL model: every service in the booking starts at the SAME
   237→  // booking-level start time (each runs on a different staff, simultaneously),
   238→  // so every segment's startMin = the booking's startMin (NO cursor advance).
   239→  // A run of same-staff services still merges into one taller block (duration
   240→  // = sum of their durations). Each segment lands in the column of the staff
   241→  // who performs it. Segments whose services have no staff go to a
   242→  // "Chưa xếp nhân viên" column at the end.
   243→  const staffColumns = useMemo<StaffColumn[]>(() => {
   244→    const columns: StaffColumn[] = allStaff.map((staff) => ({
   245→      staff,
   246→      segments: [],
   247→    }));
   248→    const byStaffId = new Map<string, StaffColumn>();
   249→    for (const col of columns) {
   250→      if (col.staff) byStaffId.set(col.staff.id, col);
   251→    }
   252→
   253→    const unassigned: ServiceSegment[] = [];
   254→    for (const booking of bookings) {
   255→      // Services ordered by sort_order for stable segment ordering.
   256→      const services = getAllServices(booking)
   257→        .slice()
   258→        .sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
   259→      // Group CONSECUTIVE services performed by the SAME staff into one
   260→      // segment (a run of same-staff services merges into one slot).
   261→      const groups: BookingServiceRow[][] = [];
   262→      for (const s of services) {
   263→        const last = groups.length > 0 ? groups[groups.length - 1] : null;
   264→        const lastStaffId = last && last.length > 0 ? last[last.length - 1].staff_id : null;
   265→        if (last && lastStaffId && lastStaffId === s.staff_id) {
   266→          last.push(s);
   267→        } else {
   268→          groups.push([s]);
   269→        }
   270→      }
   271→      const totalSegments = groups.length;
   272→      const startMin = getBookingStartMinutes(booking);
   273→      // PARALLEL: every segment starts at the booking's startMin. Previously a
   274→      // cursor advanced by each segment's duration (consecutive model), making
   275→      // the 2nd service of a multi-staff booking appear at startMin + 1st
   276→      // service's duration (e.g. 11:00 instead of 09:30). Fixed to parallel.
   277→      groups.forEach((group, i) => {
   278→        const duration = group.reduce((sum, s) => sum + (s.service?.duration || 0), 0);
   279→        const staffId = group[0].staff_id || "";
   280→        const seg: ServiceSegment = {
   281→          booking,
   282→          services: group,
   283→          staffId,
   284→          staffName: group[0].staff?.name || "—",
   285→          segmentIndex: i,
   286→          totalSegments,
   287→          startMin: startMin ?? 0,
   288→          duration,
   289→        };
   290→        if (!staffId) {
   291→          unassigned.push(seg);
   292→          return;
   293→        }
   294→        const col = byStaffId.get(staffId);
   295→        if (col) {
   296→          col.segments.push(seg);
   297→        } else {
   298→          // Staff on the booking but not in allStaff (different branch / group).
   299→          // Add an extra column so the segment is still visible.
   300→          const newCol: StaffColumn = {
   301→            staff: { id: staffId, name: seg.staffName },
   302→            segments: [seg],
   303→          };
   304→          columns.push(newCol);
   305→          byStaffId.set(staffId, newCol);
   306→        }
   307→      });
   308→    }
   309→
   310→    // The "Chưa xếp nhân viên" (unassigned) column is placed FIRST (before the
   311→    // real staff columns) so it sits right after the "Giờ" column on the left,
   312→    // per the user's request. Previously it was appended at the end (rightmost)
   313→    // which made it easy to miss. Putting it first draws attention to bookings
   314→    // that still need a staff assignment — the cashier sees them immediately
   315→    // and can click "Xếp nhân viên" on each block to assign one.
   316→    if (unassigned.length > 0) {
   317→      columns.unshift({ staff: null, segments: unassigned });
   318→    }
   319→    // Sort each column's segments by PRIORITY so that overlapping segments
   320→    // stack with UNPAID bookings (pending/new/confirmed/checkin) ON TOP of
   321→    // cancelled/no_show ones. In the single-day staff view, segments are
   322→    // absolutely positioned and later-in-DOM renders on top — so we sort
   323→    // cancelled FIRST (bottom) and unpaid LAST (top). This implements the
   324→    // user's requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy"
   325→    // (a cancelled booking at 10:30-12:00 should NOT cover an unpaid booking
   326→    // at 10:00-11:30; the unpaid one must be fully visible).
   327→    const SEG_PRIORITY: Record<string, number> = {
   328→      // cancelled / no-show → priority 0 (rendered FIRST → at the BOTTOM)
   329→      cancelled: 0,
   330→      no_show: 0,
   331→      // paid → priority 1
   332→      checkout: 1,
   333→      // unpaid / active → priority 2 (rendered LAST → ON TOP)
   334→      pending: 2,
   335→      new: 2,
   336→      confirmed: 2,
   337→      checkin: 2,
   338→    };
   339→    for (const col of columns) {
   340→      col.segments.sort((a, b) => {
   341→        const pa = SEG_PRIORITY[a.booking.status] ?? 1;
   342→        const pb = SEG_PRIORITY[b.booking.status] ?? 1;
   343→        if (pa !== pb) return pa - pb; // lower priority first (rendered below)
   344→        // Same priority → keep chronological order (earlier start first).
   345→        return a.startMin - b.startMin;
   346→      });
   347→    }
   348→    return columns;
   349→  }, [allStaff, bookings]);
   350→
   351→  // Total timeline height in pixels. Uses the resizable pxPerHour (single-day)
   352→  // or the constant (multi-day DayRangeGrid uses its own DAYGRID_ROW_HEIGHT).
   353→  const timelineHeight = (END_HOUR - START_HOUR) * pxPerHour;
   354→
   355→  // ---- Column resizing (with persistence) ----
   356→  // columnWidths[0] = "Giờ" column width; columnWidths[1..N] = staff columns.
   357→  // State-driven so the user can drag column edges to resize (same UX as the
   358→  // global TableResizer, but adapted for this CSS-grid layout which isn't a
   359→  // <table> and therefore isn't handled by TableResizer).
   360→  //
   361→  // Persistence: widths are saved to localStorage keyed by branch + column
   362→  // count, so the user's column customizations survive page switches and
   363→  // browser refreshes. When the column count changes (e.g. branch switch
   364→  // loads a different number of staff), the saved widths for that exact
   365→  // column count are restored; if none match, defaults are used.
   366→  const STORAGE_PREFIX = "crm-staff-grid-widths:";
   367→
   368→  /** Build the localStorage key for the current branch + column count. */
   369→  const storageKey = `${STORAGE_PREFIX}${branchId || "all"}::n${staffColumns.length}`;
   370→
   371→  /** Read saved widths from localStorage. Returns null if none/invalid. */
   372→  function readSavedWidths(key: string): number[] | null {
   373→    try {
   374→      const raw = localStorage.getItem(key);
   375→      if (!raw) return null;
   376→      const parsed = JSON.parse(raw);
   377→      if (!Array.isArray(parsed) || parsed.length === 0) return null;
   378→      if (!parsed.every((n: unknown) => typeof n === "number" && n > 0)) return null;
   379→      return parsed as number[];
   380→    } catch {
   381→      return null;
   382→    }
   383→  }
   384→
   385→  /** Save widths to localStorage (best-effort). */
   386→  function writeSavedWidths(key: string, widths: number[]) {
   387→    try {
   388→      localStorage.setItem(key, JSON.stringify(widths));
   389→    } catch {
   390→      // best-effort — localStorage may be full or disabled
   391→    }
   392→  }
   393→
   394→  // Compute the initial widths: try saved widths for this branch+count, fall
   395→  // back to defaults.
   396→  const [columnWidths, setColumnWidths] = useState<number[]>(() => {
   397→    const saved = readSavedWidths(storageKey);
   398→    if (saved && saved.length === staffColumns.length + 1) return saved;
   399→    return [TIME_COL_WIDTH, ...Array(staffColumns.length).fill(STAFF_COL_WIDTH)];
   400→  });
   401→
   402→  // Reset widths whenever the column count changes (e.g. staff data finishes
   403→  // loading, or the branch changes). Uses the "adjust state during render"
   404→  // pattern (React docs) instead of an effect so the lint rule
   405→  // `react-hooks/set-state-in-effect` is satisfied. Restores saved widths
   406→  // for the new column count if available.
   407→  const [prevStorageKey, setPrevStorageKey] = useState(storageKey);
   408→  if (storageKey !== prevStorageKey) {
   409→    setPrevStorageKey(storageKey);
   410→    const saved = readSavedWidths(storageKey);
   411→    if (saved && saved.length === staffColumns.length + 1) {
   412→      setColumnWidths(saved);
   413→    } else {
   414→      setColumnWidths([
   415→        TIME_COL_WIDTH,
   416→        ...Array(staffColumns.length).fill(STAFF_COL_WIDTH),
   417→      ]);
   418→    }
   419→  }
   420→
   421→  // Permission gate: only staff whose group has "resize_table" can drag the
   422→  // column edges to resize. When the permission is OFF, the grid uses `1fr`
   423→  // for all staff columns (equal width, fills container, NOT resizable). When
   424→  // the permission is ON, the grid uses persisted pixel widths (resizable via
   425→  // the drag handles on each column's right edge; dragging one staff column
   426→  // resizes ALL staff columns synchronously so they stay uniform).
   427→  const canResizeTable = hasPermission("resize_table");
   428→
   429→  // Build the CSS grid-template-columns string.
   430→  // - canResizeTable = true: "Giờ" col (px) + staff cols (px, from
   431→  //   columnWidths). Dragging a staff column's edge resizes all staff columns
   432→  //   together (sync mode). The total may exceed the container → horizontal
   433→  //   scroll.
   434→  // - canResizeTable = false: "Giờ" col (px) + staff cols (1fr each). Always
   435→  //   equal width, fills the container, not draggable.
   436→  const gridTemplate = canResizeTable
   437→    ? columnWidths.map((w) => `${w}px`).join(" ")
   438→    : `${columnWidths[0] || TIME_COL_WIDTH}px repeat(${Math.max(staffColumns.length, 1)}, minmax(0, 1fr))`;
   439→  // Total scrollable width. Only meaningful when canResizeTable = true (the
   440→  // px-based grid may exceed the container). When 1fr, the container is 100%.
   441→  const totalWidth = columnWidths.reduce((s, w) => s + w, 0);
   442→
   443→  // Start dragging a column's right edge. `idx` is the column index in
   444→  // columnWidths (0 = "Giờ", 1..N = staff columns).
   445→  //
   446→  // SYNC MODE: dragging ANY staff column's edge resizes ALL staff columns
   447→  // together (the delta is applied to every staff column uniformly). This
   448→  // keeps the grid visually uniform — all staff columns always have the same
   449→  // width. The "Giờ" column (idx 0) is excluded from the sync: dragging its
   450→  // edge still resizes only it (since the time column has a different purpose
   451→  // and width). The user's request was specifically about staff columns, so
   452→  // we sync those.
   453→  const startColumnResize = useCallback(
   454→    (e: React.MouseEvent, idx: number) => {
   455→      e.preventDefault();
   456→      e.stopPropagation();
   457→      const startX = e.clientX;
   458→      const startWidth = columnWidths[idx] || STAFF_COL_WIDTH;
   459→      // For staff columns, capture every staff column's start width so we can
   460→      // apply the same delta to all of them synchronously.
   461→      const staffStartWidths = columnWidths.slice(1);
   462→      document.body.style.cursor = "col-resize";
   463→      document.body.style.userSelect = "none";
   464→      // Capture the current key so the save after drag uses the right one
   465→      // (in case the branch changes mid-drag — extremely unlikely but safe).
   466→      const dragKey = storageKey;
   467→      let saveTimer: ReturnType<typeof setTimeout> | null = null;
   468→      const onMove = (ev: MouseEvent) => {
   469→        const delta = ev.clientX - startX;
   470→        if (idx === 0) {
   471→          // Dragging the "Giờ" column's edge → resize only that column.
   472→          const newWidth = Math.max(MIN_COL_WIDTH, startWidth + delta);
   473→          setColumnWidths((prev) => {
   474→            const next = [...prev];
   475→            next[0] = newWidth;
   476→            return next;
   477→          });
   478→        } else {
   479→          // Dragging a STAFF column's edge → apply the delta to ALL staff
   480→          // columns synchronously so they stay uniform. Each staff column's
   481→          // new width is clamped to MIN_COL_WIDTH independently (in practice
   482→          // they all clamp together since they share the same start width +
   483→          // delta).

   831→/** A single service-segment block on a staff's timeline. The block occupies

   833→ *  THIS service runs for, on THIS staff's column. A multi-service / multi-staff
   834→ *  booking therefore appears as separate blocks in each staff's column, each
   835→ *  occupying its own time slice (services run back-to-back). Hover opens the
   836→ *  full booking popover (all services, invoice, actions). */
   837→function SegmentBlock({
   838→  segment,
   839→  onClick,
   840→  onStatusChange,
   841→  onEdit,
   842→  onDelete,
   843→  onAssignStaff,
   844→  pxPerHour = PX_PER_HOUR,
   845→}: {
   846→  segment: ServiceSegment;
   847→  onClick: () => void;
   848→  onStatusChange: (status: BookingStatusType) => void;
   849→  onEdit: () => void;
   850→  onDelete: () => void;
   851→  /** Open the booking edit dialog to assign a staff to this segment's
   852→      service(s). Called by the "Xếp nhân viên" button shown ONLY when this
   853→      segment has no staff assigned (staffName === "—"). Mirrors onEdit but is
   854→      a separate affordance so the user sees a clear "assign staff" CTA
   855→      directly on unassigned blocks. */

   858→  pxPerHour?: number;

   861→  const canCancelPayment = useAuthStore((s) => s.hasPermission("cancel_payment"));
   862→  const canViewCustomerPhone = useAuthStore((s) => s.hasPermission("view_customer_phone"));

   870→  const highlightBookingId = useBookingStore((s) => s.highlightBookingId);
   871→  const setHighlightBookingId = useBookingStore((s) => s.setHighlightBookingId);
   872→  const isHighlighted = highlightBookingId === booking.id;
   873→  // After the flash animation completes, clear the highlight so it doesn't
   874→  // re-flash on every re-render.
   875→  useEffect(() => {
   876→    if (!isHighlighted) return;
   877→    const timer = setTimeout(() => setHighlightBookingId(null), 1500);
   878→    return () => clearTimeout(timer);
   879→  }, [isHighlighted, setHighlightBookingId]);
   880→  const [hovered, setHovered] = useState(false);
   881→  const [selectOpen, setSelectOpen] = useState(false);
   882→
   883→  // Clamp the segment into the visible [08:00, 21:00] window.
   884→  const trackMinutes = (END_HOUR - START_HOUR) * 60;
   885→  // Skip only segments that start AFTER the visible window ends.
   886→  // Segments that end BEFORE the visible window (e.g. a booking at 03:30 AM
   887→  // ending at 05:30 — before 08:00) are NOT skipped: they're clamped to the
   888→  // top of the grid so the user can SEE them and click to fix the time.
   889→  // Without this, a booking at 03:30 VN (e.g. from a timezone bug) would be
   890→  // invisible in View nhân viên — the user would see 5 bookings in the list
   891→  // but only 4 in the staff view.
   892→  if (segment.startMin >= trackMinutes) return null;

   894→  // If the entire segment is before the visible window, give it a minimum
   895→  // visible height (30 min) at the top of the grid.
   896→  const rawEnd = segment.startMin + segment.duration;
   897→  const clampedEnd = rawEnd <= 0
   898→    ? Math.max(clampedStart + 30, Math.min(rawEnd, trackMinutes))
   899→    : Math.min(rawEnd, trackMinutes);
   900→  const topPx = (clampedStart / 60) * pxPerHour;
   901→  const heightPx = Math.max(24, ((clampedEnd - clampedStart) / 60) * pxPerHour);
   902→
   903→  // Time-range label "HH:MM - HH:MM" for THIS service's slice.
   904→  const startTotalMin = START_HOUR * 60 + segment.startMin;
   905→  const endTotalMin = startTotalMin + segment.duration;
   906→  const fmtTime = (totalMin: number) => {
   907→    const h = Math.floor(totalMin / 60);
   908→    const m = totalMin % 60;
   909→    return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`;
   910→  };
   911→  const timeRange = `${fmtTime(startTotalMin)} - ${fmtTime(endTotalMin)}`;
   912→
   913→  const dt = booking.date_time;
   914→  const d = dt ? new Date(dt) : null;
   915→  const dateLabel = d && !isNaN(d.getTime())
   916→    ? `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`
   917→    : "";
   918→
   919→  const staffName = segment.staffName;
   920→  const isMultiSegment = segment.totalSegments > 1;
   921→  const isMergedSlot = segment.services.length > 1;
   922→
   923→  // Block background color — groups the booking by payment/cancellation state
   924→  // so the cashier can tell at a glance which slots need attention:
   925→  // - Paid (checkout) → emerald tint (matches the status badge)
   926→  // - Unpaid active (new / confirmed / checkin) → sky tint (in-progress)
   927→  // - Cancelled / no-show → red tint (dead slots)
   928→  // We use the SAME color tokens as the status badge (BookingStatusBadgeColors)
   929→  // so the block and its popover badge stay visually consistent.
   930→  // Card background color — SAME scheme as View khách hàng > Khung giờ
   931→  // (user's color spec):
   932→  // - confirmed / new → XANH DƯƠNG (blue)
   933→  // - checkin, chưa bấm Thanh toán → XANH (green)
   934→  // - checkin + đang review (đã bấm Thanh toán, chưa Hoàn tất) → TÍM ĐẬM (darker purple)
   935→  // - no_show → yellow
   936→  // - cancelled → red
   937→  // - checkout (đã Hoàn tất) → TRẮNG (white)
   938→  //
   939→  // "đã bấm Thanh toán" is the shared payment-review flag (useIsReviewing),
   940→  // NOT the invoice's pending status — every checkin booking auto-creates a
   941→  // pending invoice, so pending-status can't tell the two states apart.
   942→  const isPaid = booking.status === "checkout";
   943→  const isCancelled = booking.status === "cancelled";
   944→  const isNoShow = booking.status === "no_show";
   945→  const isCheckin = booking.status === "checkin";
   946→
   947→  let blockBg: string;
   948→  let timeText: string;

  1165→/** Minutes from the timeline's START_HOUR (08:00) to the booking's start time.
  1166→ *  May be negative (booking starts before 08:00) — callers clamp into the
  1167→ *  visible window. Returns null when date_time is missing or unparseable. */
  1168→function getBookingStartMinutes(booking: Booking): number | null {
  1169→  const dt = booking.date_time;
  1170→  if (!dt) return null;
  1171→  // Use the timezone-safe Vietnam time helper. Supabase normalizes stored
  1172→  // date_time offsets to +00:00 (UTC), so the "THH:MM" segment is the UTC
  1173→  // time — NOT the Vietnam time the user entered. Parsing the segment directly
  1174→  // made bookings display at the wrong hour (and outside the visible window).
  1175→  const hhmm = toVietnamTime(dt);
  1176→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1177→  if (!m) return null;
  1178→  const hour = parseInt(m[1], 10);
  1179→  const minute = parseInt(m[2], 10);
  1180→  if (isNaN(hour) || isNaN(minute)) return null;
  1181→  return (hour - START_HOUR) * 60 + minute;
  1182→}

  1230→/** Build the list of calendar days (inclusive) between `from` and `to`.
  1231→ *  Each Date is normalized to local midnight so day-of-month arithmetic is
  1232→ *  stable regardless of the host timezone. */
  1233→function buildDaysInRange(from: Date, to: Date): Date[] {
  1234→  const days: Date[] = [];
  1235→  const start = new Date(from);
  1236→  start.setHours(0, 0, 0, 0);
  1237→  const end = new Date(to);
  1238→  end.setHours(0, 0, 0, 0);
  1239→  if (end < start) return [start];
  1240→  const cur = new Date(start);
  1241→  while (cur <= end) {
  1242→    days.push(new Date(cur));
  1243→    cur.setDate(cur.getDate() + 1);
  1244→  }
  1245→  return days;
  1246→}

  1248→/** Build a `YYYY-MM-DD` dayKey from a local Date object (no UTC shift). */
  1249→function dateToLocalDayKey(d: Date): string {
  1250→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
  1251→}

  1253→/** Extract { dayKey, hour, minute } from the booking's `date_time` as VIETNAM
  1254→ *  wall-clock values. Post-migration (Task 6), `date_time` is stored as a UTC
  1255→ *  ISO string (e.g. "2026-07-14T03:30:00+00:00" represents 10:30 VN), so the
  1256→ *  raw "THH:MM" segment is the UTC time, NOT the VN time the user entered.
  1257→ *  Using `toVietnamDay` + `toVietnamTime` ensures the booking lands in the
  1258→ *  correct row+column of the multi-day staff grid (a 10:30 VN booking lands in
  1259→ *  the "10:00" row of column "2026-07-14", not the skipped "03:00" row). */
  1260→function getBookingDayHour(booking: Booking): { dayKey: string; hour: number; minute: number } | null {
  1261→  const dt = booking.date_time;
  1262→  if (!dt || typeof dt !== "string") return null;
  1263→  const dayKey = toVietnamDay(dt);
  1264→  const hhmm = toVietnamTime(dt);
  1265→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1266→  if (!m) return null;
  1267→  const hour = parseInt(m[1], 10);
  1268→  const minute = parseInt(m[2], 10);
  1269→  if (isNaN(hour) || isNaN(minute)) return null;
  1270→  return { dayKey, hour, minute };
  1271→}

  1273→/** "dd/MM" label for a day. */
  1274→function dayLabel(d: Date): string {
  1275→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`;
  1276→}

  1278→/** Short weekday name (T2, T3, … T7, CN). */
  1279→function weekdayLabel(d: Date): string {
  1280→  const vi = ["CN", "T2", "T3", "T4", "T5", "T6", "T7"];

  1284→function DayRangeGrid({
  1285→  dateRange,
  1286→  bookings,
  1287→  onBookingClick,
  1288→  canViewCustomerPhone,
  1289→  onSlotClick,
  1290→  slotLocked,
  1291→  onEdit,
  1292→  onAssignStaff,
  1293→}: DayRangeGridProps) {
  1294→  const days = useMemo(() => buildDaysInRange(dateRange.from, dateRange.to), [dateRange.from, dateRange.to]);
  1295→  const hours = useMemo(() => {
  1296→    const h: number[] = [];
  1297→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
  1298→    return h;
  1299→  }, []);
  1300→
  1301→  // Measure the container width so we can compute the per-day column width
  1302→  // when 5+ days are selected. The "minimum readable" layout is defined as
  1303→  // 4 days filling the table; 5+ days keep the SAME column width as 4 days
  1304→  // would have had, and scroll horizontally. Without measuring we'd have to
  1305→  // hard-code a width that may not match the actual table width.
  1306→  const containerRef = useRef<HTMLDivElement>(null);
  1307→  const [containerWidth, setContainerWidth] = useState(0);
  1308→  useEffect(() => {
  1309→    const el = containerRef.current;
  1310→    if (!el) return;
  1311→    const update = () => setContainerWidth(el.getBoundingClientRect().width);
  1312→    update();
  1313→    const ro = new ResizeObserver(update);
  1314→    ro.observe(el);
  1315→    return () => ro.disconnect();
  1316→  }, []);
  1317→
  1318→  // Index bookings by `${dayKey}|${hour}` for O(1) cell lookup.
  1319→  const cellMap = useMemo(() => {
  1320→    const m = new Map<string, Booking[]>();
  1321→    for (const b of bookings) {
  1322→      const info = getBookingDayHour(b);
  1323→      if (!info) continue;
  1324→      // Skip bookings outside the visible hour range.
  1325→      if (info.hour < START_HOUR || info.hour > END_HOUR) continue;
  1326→      const key = `${info.dayKey}|${info.hour}`;
  1327→      const arr = m.get(key);
  1328→      if (arr) arr.push(b);
  1329→      else m.set(key, [b]);
  1330→    }
  1331→    // Sort each cell's bookings:
  1332→    // 1. UNPAID bookings (pending/new/confirmed/checkin) take priority and
  1333→    //    render ON TOP of cancelled/no_show ones. When a cancelled booking
  1334→    //    would occupy the same slot as an unpaid booking (same start minute),
  1335→    //    the unpaid one wins the visible position. This implements the user's
  1336→    //    requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy".
  1337→    // 2. Within the same priority, sort by start minute (earlier → top) so
  1338→    //    the cell still reads top-to-bottom chronologically.
  1339→    const PRIORITY: Record<string, number> = {
  1340→      // unpaid / active → priority 0 (top)
  1341→      pending: 0,
  1342→      new: 0,
  1343→      confirmed: 0,
  1344→      checkin: 0,
  1345→      // paid → priority 1
  1346→      checkout: 1,
  1347→      // cancelled / no-show → priority 2 (bottom, "hidden under" unpaid)
  1348→      cancelled: 2,
  1349→      no_show: 2,
  1350→    };
  1351→    for (const arr of m.values()) {
  1352→      arr.sort((a, b) => {
  1353→        const pa = PRIORITY[a.status] ?? 1;
  1354→        const pb = PRIORITY[b.status] ?? 1;
  1355→        if (pa !== pb) return pa - pb;
  1356→        const ai = getBookingDayHour(a);
  1357→        const bi = getBookingDayHour(b);
  1358→        return (ai?.minute ?? 0) - (bi?.minute ?? 0);
  1359→      });
  1360→      // When an UNPAID booking and a CANCELLED booking share the EXACT same
  1361→      // start minute (a true slot collision — e.g. a cancelled booking at
  1362→      // 10:30 and a new confirmed booking at 10:30 for the same staff), hide
  1363→      // the cancelled one entirely so the unpaid booking's chip is fully
  1364→      // visible (not stacked under/over the cancelled chip). Bookings with
  1365→      // different start minutes stay (they're different time slots within
  1366→      // the same hour cell).
  1367→      const hasUnpaid = arr.some(
  1368→        (b) => (PRIORITY[b.status] ?? 1) === 0
  1369→      );
  1370→      if (hasUnpaid) {
  1371→        // Build a set of start-minutes that have an unpaid booking.
  1372→        const unpaidMinutes = new Set<number>();
  1373→        for (const b of arr) {
  1374→          if ((PRIORITY[b.status] ?? 1) === 0) {
  1375→            const info = getBookingDayHour(b);
  1376→            if (info) unpaidMinutes.add(info.minute);
  1377→          }
  1378→        }
  1379→        // Remove cancelled/no_show bookings whose start minute collides with
  1380→        // an unpaid booking's start minute. Keep non-colliding cancelled ones
  1381→        // (different slot within the same hour) so the staff still sees them.
  1382→        for (let i = arr.length - 1; i >= 0; i--) {
  1383→          const b = arr[i];
  1384→          const isCancelled = b.status === "cancelled" || b.status === "no_show";
  1385→          if (!isCancelled) continue;
  1386→          const info = getBookingDayHour(b);
  1387→          if (info && unpaidMinutes.has(info.minute)) {
  1388→            arr.splice(i, 1);
  1389→          }
  1390→        }
  1391→      }
  1392→    }
  1393→    return m;
  1394→  }, [bookings]);
  1395→
  1396→  // Layout decision:
  1397→  //   - 1–4 days → columns fill the full table width (1fr each), no scroll.
  1398→  //   - 5+ days → each column keeps the SAME width it would have had when 4
  1399→  //     days filled the table ((containerWidth - timeCol) / 4), and the table
  1400→  //     scrolls horizontally. This guarantees cells never shrink below the
  1401→  //     4-day baseline width, no matter how many days are selected.
