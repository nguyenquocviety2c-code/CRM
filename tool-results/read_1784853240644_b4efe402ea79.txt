    39→// Fetch a customer's "Khách cũ" status from the API. A customer counts as
    40→// "old" (Khách cũ) when they have at least one completed invoice OR belong
    41→// to a customer group whose name contains "khách cũ". This mirrors the
    42→// booking dialog's determination so both modules filter service categories
    43→// consistently. Returns "new" on any error (safe default: show the
    44→// new-customer cut).
    45→async function fetchCustomerOldStatus(customerId: string): Promise<"old" | "new"> {
    46→  try {
    47→    const res = await fetch(
    48→      `/api/supabase/customers/${encodeURIComponent(customerId)}`
    49→    );
    50→    const json = await res.json();
    51→    if (!json.ok || !json.data) return "new";
    52→    const groupName = (json.data.group?.name || "").toLowerCase();
    53→    const isOld =
    54→      json.data.customer_type === "old" || groupName.includes("khách cũ");
    55→    return isOld ? "old" : "new";
    56→  } catch {
    57→    return "new";
    58→  }
    59→}

   445→  // Pick a day's booking → opens a customer tab AND adds the booking's services

   452→    if (!b.customer) return;

   455→    openCustomerTab({
   456→      customerId: tabId,

   461→    // decide same-booking (PUT, gap ≤ 15 min) vs new-booking (POST, gap > 15).
