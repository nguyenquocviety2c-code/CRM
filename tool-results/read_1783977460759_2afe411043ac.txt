   105→  // currently typing in so the dropdown anchors under it.

   107→    Array<{ id: string; name: string; phone: string | null; code: string | null }>

   131→  // --- Reference data: service categories ---------------------------------
   132→  const { data: categoriesData } = useQuery<ServiceCategory[]>({

   148→    queryFn: async () => {
   149→      const res = await fetch("/api/supabase/services?limit=200");

   153→  });
   154→
   155→  // --- Reference data: staff (hairdressers only) --------------------------
   156→  const { data: staffData } = useQuery<Staff[]>({
   157→    queryKey: ["dat-lich-staff", selectedBranchId],
   158→    queryFn: async () => {
   159→      if (!selectedBranchId) return [];
   160→      const res = await fetch(
   161→        `/api/supabase/staff?branch_id=${selectedBranchId}&active=true&limit=200`
   162→      );
   163→      const json = await res.json();
   164→      if (!json.ok) return [];
   165→      return (json.data as Array<Record<string, unknown>>)
   166→        .filter((s) => {
   167→          const groupName = (s.group as { name?: string } | null)?.name;
   168→          return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   169→        })
   170→        .map((s) => ({
   171→          id: s.id as string,
   172→          name: s.name as string,

   178→
   179→  // --- Phone lookup: determine if the customer is new or returning -------
   180→  // A "returning" (old) customer is one whose exact phone number already
   181→  // exists in the customers table. This drives the service-category filter
   182→  // (new customers see the "Dành cho khách hàng mới" cut; old customers see
   183→  // the regular "Dịch Vụ Cắt").
   184→  const trimmedPhone = customerPhone.trim();
   185→  const { data: phoneLookup } = useQuery<
   186→    { id: string; phone?: string | null }[] | null
   187→  >({
   188→    queryKey: ["dat-lich-phone-lookup", trimmedPhone],
   189→    queryFn: async () => {
   190→      if (!trimmedPhone) return null;
   191→      const res = await fetch(
   192→        `/api/supabase/customers?search=${encodeURIComponent(trimmedPhone)}&limit=20`
   193→      );
   194→      const json = await res.json();
   195→      if (!json.ok) return null;
   196→      return (json.data || []) as { id: string; phone?: string | null }[];
   197→    },
   198→    enabled: trimmedPhone.length >= 9,
   199→  });
   200→  const isOldCustomer = useMemo(() => {
   201→    if (!trimmedPhone || !phoneLookup) return false;
   202→    return phoneLookup.some((c) => c.phone === trimmedPhone);
   203→  }, [trimmedPhone, phoneLookup]);
   204→
   205→  // --- Customer autocomplete ----------------------------------------------
   206→  // Debounced search: when the user types ≥2 chars in EITHER the name or phone
   207→  // field, fetch matching customers (by name prefix OR phone prefix) and show
   208→  // a dropdown. Selecting a suggestion fills both name + phone.
   209→  useEffect(() => {
   210→    // Determine the search term based on which field the user is typing in.
   211→    const term =
   212→      activeField === "phone"
   213→        ? customerPhone.trim()
   214→        : activeField === "name"
   215→          ? customerName.trim()
   216→          : "";
   217→    let cancelled = false;
   218→    const t = setTimeout(async () => {
   219→      if (term.length < 2) {
   220→        if (!cancelled) {
   221→          setSuggestions([]);
   222→          setShowSuggestions(false);
   223→        }
   224→        return;
   225→      }
   226→      try {
   227→        const res = await fetch(
   228→          `/api/supabase/customers?search=${encodeURIComponent(term)}&limit=10`
   229→        );
   230→        const json = await res.json();
   231→        if (cancelled) return;
   232→        if (json.ok && Array.isArray(json.data)) {
   233→          // Filter to prefix matches on name OR phone (the API's `search`
   234→          // does an ilike contains; we refine to prefix for relevance).
   235→          const lower = term.toLowerCase();
   236→          const matches = (json.data as Array<{ id: string; name: string; phone?: string | null; code?: string | null }>)
   237→            .filter((c) => {
   238→              const nameMatch = (c.name || "").toLowerCase().startsWith(lower);
   239→              const phoneMatch = (c.phone || "").startsWith(term);
   240→              return nameMatch || phoneMatch;
   241→            })
   242→            .map((c) => ({ id: c.id, name: c.name, phone: c.phone ?? null, code: c.code ?? null }));
   243→          if (!cancelled) {
   244→            setSuggestions(matches);
   245→            setShowSuggestions(matches.length > 0);
   246→          }
   247→        }
   248→      } catch {
   249→        if (!cancelled) {
   250→          setSuggestions([]);
   251→          setShowSuggestions(false);
   252→        }
   253→      }
   254→    }, 250); // 250ms debounce
   255→    return () => {
   256→      cancelled = true;
   257→      clearTimeout(t);
   258→    };
   259→  }, [customerName, customerPhone, activeField]);
   260→
   261→  const selectSuggestion = (s: { name: string; phone: string | null }) => {
   262→    setCustomerName(s.name);
   263→    setCustomerPhone(s.phone || "");
   264→    setShowSuggestions(false);
   265→    setActiveField(null);
   266→  };
   267→  const customerType: "new" | "old" = isOldCustomer ? "old" : "new";
   268→
   269→  // Categories filtered by branch AND customer type (mirrors booking-dialog):
   270→  // - New customer (phone not in DB): hide "Dịch Vụ Cắt", keep new-customer cut.
   271→  // - Old customer (phone in DB): hide "Dành cho khách hàng mới - DV Cắt".
   272→  const categories = useMemo(() => {
   273→    const all = categoriesData || [];
   274→    const byBranch =
   275→      !selectedBranchId || selectedBranchId === "all"
   276→        ? all
   277→        : all.filter((c) => (c.branches || []).includes(selectedBranchId));
   278→    return byBranch.filter((c) => {
   279→      const name = c.name.toLowerCase();
   280→      const isNewCustomerCut = name.includes("dành cho khách hàng mới");
   281→      const isRegularCut =
   282→        name.includes("dịch vụ cắt") || name === "dịch vụ cắt";
   283→      if (customerType === "new") {
   284→        if (isRegularCut && !isNewCustomerCut) return false;
   285→      } else {
   286→        if (isNewCustomerCut) return false;
   287→      }
   288→      return true;
   289→    });
   290→  }, [categoriesData, selectedBranchId, customerType]);
   291→
   292→  const allServices = servicesData || [];
   293→  const allStaff = staffData || [];
   294→
   295→  // If a row's selected category is no longer visible after the customer-type
   296→  // filter changes (e.g. phone typed → old customer → "Dành cho KH mới" hidden),
   297→  // reset that row's service selection chain to avoid a stale hidden selection.
   298→  useEffect(() => {
   299→    setServiceRows((prev) => {
   300→      let changed = false;
   301→      const next = prev.map((row) => {
   302→        if (row.categoryId && !categories.some((c) => c.id === row.categoryId)) {
   303→          changed = true;
   304→          return { ...row, categoryId: "", serviceId: "", staffId: "" };
   305→        }
   306→        return row;
   307→      });
   308→      return changed ? next : prev;
   309→    });
   310→  }, [categories]);
   311→
   312→  // Per-row helpers: services filtered by the row's category, and staff
   313→  // filtered by the row's service (name-based narrowing mirrors booking-dialog).
   314→  const servicesForCategory = useCallback(
   315→    (catId: string) => allServices.filter((s) => !catId || s.category_id === catId),
   316→    [allServices]
   317→  );
   318→  const staffForService = useCallback(
   319→    (svcId: string) => {
   320→      if (!svcId) return allStaff;
   321→      const svc = allServices.find((s) => s.id === svcId);
   322→      if (!svc) return allStaff;
   323→      const n = (svc.name || "").toLowerCase();
   324→      if (n.includes("creative director")) {
   325→        return allStaff.filter((s) => s.groupName === "Creative Director");
   326→      }
   327→      if (n.includes("artist")) {
   328→        return allStaff.filter((s) => s.groupName === "Artist");
   329→      }
   330→      return allStaff;
   331→    },
   332→    [allStaff, allServices]
   333→  );
   334→
   335→  // --- Feasible-time computation (multi-service aware) --------------------
   336→  // Fetch the day's existing bookings (for the branch). We compute which
   337→  // HH:MM start times would cause a conflict for ANY of the selected staff.
   338→  // In the parallel model, all services start at the same time T; a time T
   339→  // is infeasible if, for ANY service row, the staff is busy during
   340→  // [T, T + service_duration]. So the hidden set = union across all rows.
   341→  const isoDay = useMemo(() => {
   342→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   343→    return m ? `${m[3]}-${m[2]}-${m[1]}` : null;
   344→  }, [bookingDate]);
   345→
   346→  // Collect the "active" rows: those with both a service and a staff chosen.
   347→  // These are the rows that participate in the feasibility computation.
   348→  const activeRows = useMemo(
   349→    () => serviceRows.filter((r) => r.serviceId && r.staffId),
   350→    [serviceRows]
   351→  );
   352→  const hasActiveRow = activeRows.length > 0;
   353→
   354→  const { data: dayBookings } = useQuery<ExistingBooking[]>({
   355→    queryKey: ["dat-lich-day-bookings", isoDay, selectedBranchId],
   356→    queryFn: async () => {
   357→      if (!isoDay) return [];
   358→      const params = new URLSearchParams();
   359→      params.set("page", "1");
   360→      params.set("limit", "200");
   361→      // Timezone-safe: isoDay is a Vietnam calendar day; convert to its UTC
   362→      // range so Supabase filters the correct window (no 7-hour shift).
   363→      const dl = localDayToUtcRange(isoDay);
   364→      params.set("date_from", dl.from);
   365→      params.set("date_to", dl.to);
   366→      if (selectedBranchId) params.set("branch_id", selectedBranchId);
   367→      const res = await fetch(`/api/supabase/bookings?${params.toString()}`);
   368→      const json = await res.json();
   369→      if (!json.ok) return [];
   370→      return (json.data || []) as ExistingBooking[];
   371→    },
   372→    enabled: !!isoDay && hasActiveRow,
   373→  });
   374→
   375→  // Compute hiddenHours + hiddenMinutes for the TimePicker across ALL active
   376→  // rows. A minute is hidden if it conflicts for ANY active row's staff.
   377→  const { hiddenHours, hiddenMinutes } = useMemo(() => {
   378→    const empty = { hiddenHours: new Set<string>(), hiddenMinutes: {} as Record<string, Set<string>> };
   379→    if (!isoDay || activeRows.length === 0) return empty;
   380→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   381→    if (!m) return empty;
   382→    // Use the +07:00 offset so dayBase is the correct UTC instant for VN
   383→    // midnight (same fix as the booking dialog + cashier service selector).
   384→    const dayBase = new Date(`${m[3]}-${m[2]}-${m[1]}T00:00:00+07:00`).getTime();
   385→    if (isNaN(dayBase)) return empty;
   386→
   387→    // Build per-row busy intervals + duration. Each row contributes its own
   388→    // staff's busy slots and its own service duration.
   389→    const rowSpecs: Array<{
   390→      intervals: { startMs: number; endMs: number }[];
   391→      durationMs: number;
   392→    }> = [];
   393→    for (const row of activeRows) {
   394→      const svc = allServices.find((s) => s.id === row.serviceId);
   395→      const durationMs = (svc?.duration || 60) * 60 * 1000;
   396→      if (durationMs <= 0) continue;
   397→      const intervals: { startMs: number; endMs: number }[] = [];
   398→      for (const b of dayBookings || []) {
   399→        if (b.status === "cancelled" || b.status === "no_show") continue;
   400→        const exStart = new Date(b.date_time).getTime();
   401→        if (isNaN(exStart)) continue;
   402→        for (const s of b.services || []) {
   403→          if (s.staff_id !== row.staffId) continue;
   404→          const dur =
   405→            (Number(s.duration) || Number(s.service?.duration) || 60) * 60 * 1000;
   406→          intervals.push({ startMs: exStart, endMs: exStart + dur });
   407→        }
   408→      }
   409→      rowSpecs.push({ intervals, durationMs });
   410→    }
   411→    if (rowSpecs.length === 0) return empty;
   412→
   413→    const hh = new Set<string>();
   414→    const hm: Record<string, Set<string>> = {};
   415→    const hourHasFeasible: Record<string, boolean> = {};
   416→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   417→    for (let min = 0; min < 60 * 24; min++) {
   418→      const startMs = dayBase + min * 60 * 1000;
   419→      // A minute conflicts if ANY row's staff is busy during [start, start+dur].
   420→      const overlaps = rowSpecs.some((spec) => {
   421→        const endMs = startMs + spec.durationMs;
   422→        return spec.intervals.some((iv) => startMs < iv.endMs && iv.startMs < endMs);
   423→      });
   424→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   425→      const mStr = String(min % 60).padStart(2, "0");
   426→      if (overlaps) {
   427→        if (!hm[hStr]) hm[hStr] = new Set<string>();
   428→        hm[hStr].add(mStr);
   429→      } else {
   430→        hourHasFeasible[hStr] = true;
   431→      }
   432→    }
   433→    for (let h = 0; h < 24; h++) {
   434→      const hStr = String(h).padStart(2, "0");
   435→      if (!hourHasFeasible[hStr]) hh.add(hStr);
   436→    }
   437→    return { hiddenHours: hh, hiddenMinutes: hm };
   438→  }, [activeRows, isoDay, dayBookings, bookingDate, allServices]);
   439→
   440→  // --- Row mutation helpers -----------------------------------------------
   441→  const updateRow = (id: string, patch: Partial<ServiceRow>) => {
   442→    setServiceRows((prev) => prev.map((r) => (r.id === id ? { ...r, ...patch } : r)));
   443→  };
   444→  const onRowCategoryChange = (id: string, catId: string) => {
   445→    // Changing category resets service + staff for that row.
   446→    updateRow(id, { categoryId: catId, serviceId: "", staffId: "" });
   447→    setBookingTime("");
   448→  };
   449→  const onRowServiceChange = (id: string, svcId: string) => {
   450→    // Changing service resets staff for that row.
   451→    updateRow(id, { serviceId: svcId, staffId: "" });
   452→    setBookingTime("");
   453→  };
   454→  const onRowStaffChange = (id: string, stId: string) => {
   455→    updateRow(id, { staffId: stId });
   456→    setBookingTime("");
   457→  };
   458→  const addRow = () => {
   459→    setServiceRows((prev) => [...prev, newRow()]);
   460→  };
   461→  const removeRow = (id: string) => {
   462→    setServiceRows((prev) => (prev.length > 1 ? prev.filter((r) => r.id !== id) : prev));
   463→    setBookingTime("");
   464→  };
   465→  // Changing the branch invalidates the whole service chain — staff and
   466→  // service availability are branch-specific. Reset all rows + time.
   467→  const onBranchChange = (id: string) => {
   468→    setSelectedBranchId(id);
   469→    setServiceRows([newRow()]);
   470→    setBookingTime("");
   471→  };
   472→
   473→  // Part 1 (customer info) must be fully filled before Part 2 (service) opens.
   474→  const part1Complete = !!(
   475→    customerName.trim() &&
   476→    customerPhone.trim() &&
   477→    bookingDate
   478→  );
   479→  // A row is "complete" when it has category + service + staff. At least one
   480→  // complete row is required before the time picker opens. Incomplete rows
   481→  // (empty or half-filled) are ignored at submit time.
   482→  const completeRows = useMemo(
   483→    () => serviceRows.filter((r) => r.categoryId && r.serviceId && r.staffId),
   484→    [serviceRows]
   485→  );
   486→  // Khung giờ requires at least 1 complete row (Nhóm DV + Dịch vụ + Kỹ thuật viên).
   487→  const timePickerReady = part1Complete && completeRows.length >= 1;
   488→
   489→  // Within-form staff conflict: two complete rows sharing the same staff would
   490→  // overlap in the parallel model (both start at T). The backend rejects this;
   491→  // we surface the conflict early so the customer can pick a different staff.
   492→  const duplicateStaff = useMemo(() => {
   493→    const ids = completeRows.map((r) => r.staffId);
   494→    return ids.some((id, i) => ids.indexOf(id) !== i);
   495→  }, [completeRows]);
   496→
   497→  const canSubmit =

   639→        }),

   644→      }

   646→      setSuccess(true);
   647→      setSkipExistingCheck(false);
   648→      setPendingExisting([]);

   657→        title: "Đặt lịch thất bại",

   659→        variant: "destructive",

   663→    }
