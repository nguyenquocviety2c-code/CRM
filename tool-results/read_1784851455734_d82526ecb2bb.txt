    44→    44→async function fetchCustomerOldStatus(customerId: string): Promise<"old" | "new"> {
    45→    45→  try {
    46→    46→    const res = await fetch(
    47→    47→      `/api/supabase/customers/${encodeURIComponent(customerId)}`
    48→    48→    );
    49→    49→    const json = await res.json();
    50→    50→    if (!json.ok || !json.data) return "new";
    51→    51→    const groupName = (json.data.group?.name || "").toLowerCase();
    52→    52→    const isOld =
    53→    53→      json.data.customer_type === "old" || groupName.includes("khách cũ");
    54→    54→    return isOld ? "old" : "new";
    55→    55→  } catch {
    56→    56→    return "new";
    57→    57→  }
