   108→/** Convert an Invoice (activity page shape) to a minimal Booking object so it
   109→ *  can be passed to InvoiceDialog (which expects a Booking). The InvoiceDialog
   110→ *  will fetch the invoice by booking_id, so we just need the id + status. */
   111→function invoiceToBooking(inv: Invoice): Booking {
   112→  return {
   113→    id: inv.booking_id || inv.id, // use booking_id if available, else invoice id
   114→    code: inv.code || "",
   115→    date: "",
   116→    time: "",
   117→    date_time: inv.created_at,
   118→    status: inv.status === "completed" ? "checkout" : "confirmed",
   119→    note: inv.note,
   120→    numberOfCustomers: 1,
   121→    customerSourceId: null,
   122→    customerChannelId: null,
   123→    createdBy: null,
   124→    customer: {
   125→      id: inv.customer?.id || "",
   126→      name: inv.customer?.name || "",
   127→      phone: inv.customer?.phone || "",
   128→      code: inv.customer?.code || "",
   129→    },
   130→    branch: inv.branch ? { id: inv.branch.id, name: inv.branch.name } : null,
   131→    services: [],
   132→    invoice: {
   133→      id: inv.id,
   134→      code: inv.code,
   135→      status: inv.status,
   136→      final_amount: inv.final_amount,
   137→      payment_method: inv.payment_method,
   138→    },
   139→  } as Booking;
   140→}

   476→          method, confirm button). For paid invoices, opens the full-page
   477→          PaidInvoiceView. */}
   478→      {detailInvoice && detailInvoice.status === "pending" ? (
   479→        <InvoiceDialog
   480→          booking={invoiceToBooking(detailInvoice)}
   481→          onClose={() => setDetailInvoice(null)}
   482→          onPaid={() => {
   483→            setDetailInvoice(null);
   484→            queryClient.invalidateQueries({ queryKey: ["supabase-invoice-history"] });
   485→          }}
   486→        />
   487→      ) : detailInvoice ? (
   488→        <PaidInvoiceView
   489→          invoiceId={detailInvoice.id}
   490→          customerName={(detailInvoice as any).customer?.name}
   491→          customerPhone={(detailInvoice as any).customer?.phone}
   492→          bookingCode={detailInvoice.code}
   493→          onClose={() => setDetailInvoice(null)}
   494→        />
   495→      ) : null}
   496→
   497→      {/* Paid invoice full-page view — also opened from InvoiceDetailDialog */}
   498→      {paidInvoiceViewData && (
   499→        <PaidInvoiceView
   500→          invoiceId={paidInvoiceViewData.invoiceId}
   501→          customerName={paidInvoiceViewData.customerName}
   502→          customerPhone={paidInvoiceViewData.customerPhone}
   503→          bookingCode={paidInvoiceViewData.code}
   504→          onClose={() => setPaidInvoiceViewData(null)}
   505→        />
   506→      )}
   507→
   508→      {/* Fallback: InvoiceDialog opened from InvoiceDetailDialog's button
   509→          (for paid invoices that the user wants to re-confirm). */}
   510→      {invoiceDialogInvoice && (
   511→        <InvoiceDialog
   512→          booking={invoiceToBooking(invoiceDialogInvoice)}
   513→          onClose={() => setInvoiceDialogInvoice(null)}
   514→          onPaid={() => {
   515→            setInvoiceDialogInvoice(null);
   516→            queryClient.invalidateQueries({ queryKey: ["supabase-invoice-history"] });
   517→          }}
   518→        />
   519→      )}
   520→    </div>
   521→  );
   522→}

   524→/**
   525→ * Read-only invoice detail dialog. Shows the full invoice: customer info,
   526→ * items, promotion, tip, total, payment method, status, creation date.
   527→ */
   528→function InvoiceDetailDialog({
   529→  invoice,
   530→  onClose,
   531→  canConfirmPayment,
   532→  onConfirmPayment,
   533→}: {
   534→  invoice: Invoice;
   535→  onClose: () => void;
   536→  canConfirmPayment: boolean;
   537→  onConfirmPayment: () => void;
   538→}) {
   539→  const fmt = (n: number) => new Intl.NumberFormat("vi-VN").format(n);
   540→  const [lightboxPhoto, setLightboxPhoto] = useState<string | null>(null);
   541→  const queryClient = useQueryClient();
   542→  // Permissions: "upload_photo" gates the device-upload feature. A single
   543→  // permission covers both unpaid and paid invoices.
   544→  // "delete_past_photos" gates deleting photos on PAID invoices.
   545→  // "view_customer_photo" gates the lightbox (click-to-zoom).
   546→  // "view_customer_phone" gates displaying the customer's phone number.
   547→  const { hasPermission } = useAuthStore();
   548→  const canUpload = hasPermission("upload_photo");
   549→  const canDeletePastPhotos = hasPermission("delete_past_photos");
   550→  const canViewCustomerPhoto = hasPermission("view_customer_photo");
   551→  const canViewCustomerPhone = hasPermission("view_customer_phone");
   552→  // Local copy of photos so uploads appear immediately without a refetch.
   553→  const [photos, setPhotos] = useState<string[]>(
   554→    Array.isArray(invoice.photos) ? invoice.photos : []
   555→  );
   556→  const [uploading, setUploading] = useState(false);
   557→  const [uploadError, setUploadError] = useState("");
   558→
   559→  const handlePhotoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   560→    const files = e.target.files;
   561→    if (!files || files.length === 0) return;
   562→    setUploading(true);
   563→    setUploadError("");
   564→    try {
   565→      const folder = invoice.code || invoice.id || "invoices";
   566→      const newUrls = await uploadImagesToR2(files, `invoices/${folder}`);
   567→      if (newUrls.length === 0) {
   568→        setUploadError("Không thể tải ảnh lên. Vui lòng thử lại.");
   569→        e.target.value = "";
   570→        return;
   571→      }
   572→      const updated = [...photos, ...newUrls];
   573→      // Persist to the invoice's note JSON via PUT.
   574→      const res = await fetch(`/api/supabase/invoices/${invoice.id}`, {
   575→        method: "PUT",
   576→        headers: { "Content-Type": "application/json" },
   577→        body: JSON.stringify({ photos: updated }),
   578→      });
   579→      const json = await res.json();
   580→      if (!json.ok) throw new Error(json.error || "Lưu ảnh thất bại");
   581→      setPhotos(updated);
   582→      // Refresh the invoice list so the new photos sync everywhere.
   583→      queryClient.invalidateQueries({ queryKey: ["supabase-invoice-history"] });
   584→    } catch (err) {
   585→      setUploadError(err instanceof Error ? err.message : "Tải ảnh lên thất bại");
   586→    } finally {
   587→      setUploading(false);
   588→      e.target.value = "";
   589→    }
   590→  };
   591→
   592→  const handleRemovePhoto = async (idx: number) => {
   593→    const updated = photos.filter((_, i) => i !== idx);
   594→    try {
   595→      await fetch(`/api/supabase/invoices/${invoice.id}`, {
   596→        method: "PUT",
   597→        headers: { "Content-Type": "application/json" },
   598→        body: JSON.stringify({ photos: updated }),
   599→      });
   600→      setPhotos(updated);
   601→      queryClient.invalidateQueries({ queryKey: ["supabase-invoice-history"] });
   602→    } catch {
   603→      /* best-effort */
   604→    }
   605→  };
   606→
   607→  const items = Array.isArray(invoice.items) ? invoice.items : [];
   608→  const subtotal = items.reduce(
   609→    (sum, it) => sum + (Number(it.price) || 0) * (Number(it.quantity) || 1),
   610→    0
   611→  );
   612→  const discount = Number(invoice.discount) || 0;
   613→  const tip = Number(invoice.tip) || 0;
   614→  const promotion = invoice.promotion;
   615→  const finalAmount = Number(invoice.final_amount) || 0;
   616→  const paymentMethod = invoice.payment_method;
   617→  // `photos` is now stateful (see handlePhotoUpload above) so uploads appear
   618→  // immediately. It is initialized from invoice.photos.
   619→
   620→  return (
   621→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   622→      <DialogContent className="order-dialog-dense !max-w-[747px] p-0 overflow-hidden">
   623→        <DialogHeader className="px-5 pt-4 pb-1.5">
   624→          <DialogTitle className="text-lg font-semibold">
   625→            Hóa đơn {invoice.code ? `· ${invoice.code}` : ""}
   626→          </DialogTitle>
   627→        </DialogHeader>
   628→        {/* Two-column layout: left = invoice detail, right = read-only photo gallery */}
   629→        <div className="grid grid-cols-1 md:grid-cols-2 gap-0">
   630→        {/* LEFT: invoice detail — scrollable independently */}
   631→        <div className="px-5 pb-4 space-y-2 md:border-r border-gray-100 max-h-[60vh] overflow-y-auto">
   632→          {/* Customer info */}
   633→          <div className="rounded-lg border bg-gray-50 p-2.5 space-y-0 text-sm">
   634→            <div className="flex justify-between">
   635→              <span className="text-gray-500">Khách hàng:</span>
   636→              <span className="font-medium text-gray-900">{invoice.customer?.name || "—"}</span>
   637→            </div>
   638→            <div className="flex justify-between">
   639→              <span className="text-gray-500">Số điện thoại:</span>
   640→              <span className="text-gray-900">{canViewCustomerPhone ? (invoice.customer?.phone || "—") : maskPhone(invoice.customer?.phone)}</span>
   641→            </div>
   642→            <div className="flex justify-between">
   643→              <span className="text-gray-500">Nguồn khách hàng:</span>
   644→              <span className="text-gray-900">{invoice.customer?.source?.name || "—"}</span>
   645→            </div>
   646→            <div className="flex justify-between">
   647→              <span className="text-gray-500">Kênh đặt lịch:</span>
   648→              <span className="text-gray-900">{invoice.customer?.channel?.name || "—"}</span>
   649→            </div>
   650→            {invoice.branch && (
   651→              <div className="flex justify-between">
   652→                <span className="text-gray-500">Chi nhánh:</span>
   653→                <span className="text-gray-900">{invoice.branch.name}</span>
   654→              </div>
   655→            )}
   656→            {invoice.staff && (
   657→              <div className="flex justify-between">
   658→                <span className="text-gray-500">Nhân viên:</span>
   659→                <span className="text-gray-900">{invoice.staff.name}</span>
   660→              </div>
   661→            )}
   662→            <div className="flex justify-between">
   663→              <span className="text-gray-500">Ngày tạo:</span>
   664→              <span className="text-gray-900">
   665→                {(() => {
   666→                  try {
   667→                    return format(new Date(invoice.created_at), "HH:mm dd/MM/yyyy", { locale: vi });
   668→                  } catch {
   669→                    return "—";
   670→                  }
   671→                })()}
   672→              </span>
   673→            </div>
   674→            <div className="flex justify-between">
   675→              <span className="text-gray-500">Trạng thái:</span>
   676→              <span
   677→                className={cn(
   678→                  "inline-flex rounded-full px-2 py-0.5 text-xs font-medium",
   679→                  invoice.status === "completed" ? "bg-emerald-100 text-emerald-700"
   680→                    : invoice.status === "pending" ? "bg-amber-100 text-amber-700"
   681→                    : invoice.status === "cancelled" ? "bg-red-100 text-red-700"
   682→                    : "bg-gray-100 text-gray-700"
   683→                )}
   684→              >
   685→                {invoice.status === "completed" ? "Đã thanh toán"
   686→                  : invoice.status === "pending" ? "Chưa thanh toán"
   687→                  : invoice.status === "cancelled" ? "Đã hủy"
   688→                  : invoice.status}
   689→              </span>
   690→            </div>
   691→          </div>
   692→
   693→          {/* Services + Tip + Total */}
   694→          <div className="rounded-lg border p-2.5">
   695→            <div className="text-xs font-semibold uppercase tracking-wide text-gray-500 mb-1.5">
   696→              Dịch vụ
   697→            </div>
   698→            <div className="space-y-1.5">
   699→              {items.length === 0 ? (
   700→                <div className="text-sm text-gray-400">Chưa có dịch vụ</div>
   701→              ) : (
   702→                items.map((s, idx) => (
   703→                  <div key={idx} className="flex items-start justify-between text-sm">
   704→                    <div>
   705→                      <div className="font-medium text-gray-900">{s.name || "Dịch vụ"}</div>
   706→                      {s.staffName && <div className="text-xs text-gray-500">NV: {s.staffName}</div>}
   707→                    </div>
   708→                    <div className="font-medium text-gray-900">
   709→                      {fmt(Number(s.total) || Number(s.price) || 0)}đ
   710→                    </div>
   711→                  </div>
   712→                ))
   713→              )}
   714→            </div>
   715→
   716→            {/* Thành tiền (subtotal) */}
   717→            <div className="mt-1.5 flex items-center justify-between text-sm">
   718→              <span className="text-gray-600">Thành tiền</span>
   719→              <span className="font-medium text-gray-900">{fmt(subtotal)}đ</span>
   720→            </div>
   721→
   722→            {/* Chương trình khuyến mãi */}
   723→            <div className="mt-2 flex items-center justify-between text-sm">
   724→              <span className="text-gray-600">Chương trình khuyến mãi</span>
   725→              <span className="font-medium text-gray-900 text-right">
   726→                {promotion
   727→                  ? `${promotion.name} (−${fmt(promotion.discountAmount)}đ)`
   728→                  : discount > 0
   729→                  ? `−${fmt(discount)}đ`
   730→                  : "—"}
   731→              </span>
   732→            </div>
   733→
   734→            {/* Tip */}
   735→            <div className="mt-2 flex items-center justify-between text-sm">
   736→              <span className="text-gray-600">Thường (thưởng thợ)</span>
   737→              <span className="font-medium text-gray-900">
   738→                {tip > 0 ? `+${fmt(tip)}đ` : "—"}
   739→              </span>
   740→            </div>
   741→
   742→            <div className="mt-1.5 flex justify-between border-t pt-1.5 text-sm">
   743→              <span className="font-medium text-gray-700">Tổng tiền</span>
   744→              <span className="font-bold text-emerald-700">{fmt(finalAmount)}đ</span>
   745→            </div>
   746→          </div>
   747→
   748→          {/* Payment method */}
   749→          <div className="space-y-1.5">
   750→            <div className="text-sm font-medium text-gray-700">Phương thức thanh toán</div>
   751→            <div className="rounded-md border border-gray-200 bg-gray-50 px-4 py-3 text-sm font-medium text-gray-900">
   752→              {paymentMethod === "transfer" ? "Chuyển khoản" : "Tiền mặt"}
   753→            </div>
   754→          </div>
   755→
   756→          {/* Note (if any) */}
   757→          {invoice.note && (
   758→            <div className="space-y-1.5">
   759→              <div className="text-sm font-medium text-gray-700">Ghi chú</div>
   760→              <div className="rounded-md border border-gray-200 bg-gray-50 px-4 py-3 text-sm text-gray-900">
   761→                {invoice.note}
   762→              </div>
   763→            </div>
   764→          )}
   765→
   766→          {/* Actions */}
   767→          <div className="flex justify-end gap-2 border-t pt-2">
   768→            {/* "Xác nhận thanh toán" button — only for unpaid invoices (status
   769→                "pending") when the user has the appropriate permission. Opens
   770→                the InvoiceDialog with editable UI (promotion, tip, payment). */}
   771→            {invoice.status === "pending" && canConfirmPayment && (
   772→              <Button
   773→                type="button"
   774→                onClick={onConfirmPayment}
   775→                className="bg-emerald-600 hover:bg-emerald-700"
   776→              >
   777→                Xác nhận thanh toán
   778→              </Button>
   779→            )}
   780→            <Button type="button" variant="outline" onClick={onClose}>
   781→              Đóng
   782→            </Button>
   783→          </div>
   784→        </div>
   785→
   786→        {/* RIGHT: photo gallery + upload (upload gated by upload_photo permission) */}
   787→        <div className="px-5 pb-4 space-y-3 bg-gray-50/50 max-h-[60vh] overflow-y-auto">
   788→          <div>
   789→            <div className="text-xs font-semibold uppercase tracking-wide text-gray-500 mb-1.5 flex items-center gap-2">
   790→              <Camera className="h-4 w-4" />
   791→              Ảnh đính kèm
   792→            </div>
   793→            {canUpload && (
   794→              <label className="mb-2 inline-flex items-center gap-2 rounded-lg border border-dashed border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-600 hover:bg-gray-50 cursor-pointer w-fit">
   795→                {uploading ? (
   796→                  <Loader2 className="h-4 w-4 animate-spin" />
   797→                ) : (
   798→                  <Camera className="h-4 w-4" />
   799→                )}
   800→                {uploading ? "Đang tải..." : "Tải ảnh lên"}
   801→                <input
   802→                  type="file"
   803→                  accept="image/*"
   804→                  multiple
   805→                  className="hidden"
   806→                  disabled={uploading}
   807→                  onChange={handlePhotoUpload}
   808→                />
   809→              </label>
   810→            )}
   811→            {uploadError && (
   812→              <div className="mb-2 rounded-md bg-red-50 px-3 py-2 text-xs text-red-600">
   813→                {uploadError}
   814→              </div>
   815→            )}
   816→            {photos.length === 0 ? (
   817→              <div className="text-sm text-gray-400">
   818→                {canUpload ? "Chưa có ảnh nào được tải lên" : "Chưa có ảnh nào được tải lên"}
   819→              </div>
   820→            ) : (
   821→              <div className="grid grid-cols-3 gap-1.5">
   822→                {photos.map((src, idx) => (
   823→                  <button
   824→                    key={idx}
   825→                    type="button"
   826→                    onClick={() => canViewCustomerPhoto && setLightboxPhoto(src)}
   827→                    disabled={!canViewCustomerPhoto}
   828→                    title={canViewCustomerPhoto ? undefined : "Bạn không có quyền xem ảnh"}
   829→                    className="relative aspect-square overflow-hidden rounded-lg border bg-white"
   830→                  >
   831→                    <img
   832→                      src={src}
   833→                      alt={`Ảnh ${idx + 1}`}
   834→                      className="h-full w-full object-cover"
   835→                    />
   836→                    {canDeletePastPhotos && (
   837→                      <span
   838→                        role="button"
   839→                        tabIndex={0}
   840→                        onClick={(e) => { e.stopPropagation(); handleRemovePhoto(idx); }}
   841→                        onKeyDown={(e) => { if (e.key === "Enter") { e.stopPropagation(); handleRemovePhoto(idx); } }}
   842→                        className="absolute right-1 top-1 inline-flex h-5 w-5 items-center justify-center rounded-full bg-black/60 text-white hover:bg-black/80"
   843→                        aria-label="Xóa ảnh"
   844→                      >
   845→                        <span className="text-xs">×</span>
   846→                      </span>
   847→                    )}
   848→                  </button>
   849→                ))}
   850→              </div>
   851→            )}
   852→          </div>
   853→        </div>
   854→        </div>
   855→      </DialogContent>
   856→
   857→      {/* Photo lightbox — clicking a thumbnail opens the full-size image. */}
   858→      {lightboxPhoto && (
   859→        <Dialog open onOpenChange={(v) => !v && setLightboxPhoto(null)}>
   860→          <DialogContent className="max-w-4xl">
   861→            <img
   862→              src={lightboxPhoto}
   863→              alt="Ảnh hóa đơn"
   864→              className="w-full h-auto"
   865→            />
   866→          </DialogContent>
   867→        </Dialog>
   868→      )}
   869→    </Dialog>
   870→  );
   871→}
