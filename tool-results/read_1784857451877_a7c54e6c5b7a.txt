   390→  // Compute hiddenHours + hiddenMinutes for the TimePicker in the dialog.

   404→    // with the correct +07:00 offset, so their epoch ms is 7 hours off from
   405→    // a Date.UTC-based dayBase).
   406→    const dayBase = new Date(`${m[3]}-${m[2]}-${m[1]}T00:00:00+07:00`).getTime();

   409→    const hiddenMinutes: Record<string, Set<string>> = {};
   410→    const hourHasFeasible: Record<string, boolean> = {};
   411→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   412→    for (let min = 0; min < 60 * 24; min++) {
   413→      const startMs = dayBase + min * 60 * 1000;
   414→      const endMs = startMs + durationMs;
   415→      const overlaps = intervals.some((iv) => startMs < iv.endMs && iv.startMs < endMs);
   416→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   417→      const mStr = String(min % 60).padStart(2, "0");
   418→      if (overlaps) {
   419→        if (!hiddenMinutes[hStr]) hiddenMinutes[hStr] = new Set<string>();
   420→        hiddenMinutes[hStr].add(mStr);
   421→      } else {
   422→        hourHasFeasible[hStr] = true;
   423→      }
   424→    }
   425→    for (let h = 0; h < 24; h++) {
   426→      const hStr = String(h).padStart(2, "0");
   427→      if (!hourHasFeasible[hStr]) hiddenHours.add(hStr);
   428→    }
   429→    return { hiddenHours, hiddenMinutes };
   430→  }, [feasibleKey, busyIntervals, selectedService, selectedDate]);
