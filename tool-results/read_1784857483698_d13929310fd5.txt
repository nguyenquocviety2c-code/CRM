    99→    Array<{ date: string; time: string; status: string; branchName: string; services: Array<{ name: string; staffName: string }> }>

   101→  const [skipExistingCheck, setSkipExistingCheck] = useState(false);
   102→  // Customer autocomplete: when the user types a name OR phone prefix (≥2
   103→  // chars), fetch matching customers and show a dropdown. Selecting a result
   104→  // fills BOTH name + phone. `activeField` tracks which input the user is
   105→  // currently typing in so the dropdown anchors under it.
   106→  const [suggestions, setSuggestions] = useState<
   107→    Array<{ id: string; name: string; phone: string | null; code: string | null }>

   128→    },
   129→  });
   130→
   131→  // --- Reference data: service categories ---------------------------------
   132→  const { data: categoriesData } = useQuery<ServiceCategory[]>({
   133→    queryKey: ["dat-lich-categories", selectedBranchId],

   136→      const json = await res.json();

   144→
   145→  // --- Reference data: services -------------------------------------------
   146→  const { data: servicesData } = useQuery<Service[]>({
   147→    queryKey: ["dat-lich-services"],
   148→    queryFn: async () => {
   149→      const res = await fetch("/api/supabase/services?limit=200");
   150→      const json = await res.json();
   151→      return (json.data || []) as Service[];
   152→    },
   153→  });
   154→
   155→  // --- Reference data: staff (hairdressers only) --------------------------
   156→  const { data: staffData } = useQuery<Staff[]>({
   157→    queryKey: ["dat-lich-staff", selectedBranchId],
   158→    queryFn: async () => {
   159→      if (!selectedBranchId) return [];
   160→      const res = await fetch(
   161→        `/api/supabase/staff?branch_id=${selectedBranchId}&active=true&limit=200`
   162→      );
   163→      const json = await res.json();
   164→      if (!json.ok) return [];
   165→      return (json.data as Array<Record<string, unknown>>)
   166→        .filter((s) => {
   167→          const groupName = (s.group as { name?: string } | null)?.name;
   168→          return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   169→        })
   170→        .map((s) => ({
   171→          id: s.id as string,
   172→          name: s.name as string,
   173→          groupName: (s.group as { name?: string } | null)?.name,
   174→        }));
   175→    },
   176→    enabled: !!selectedBranchId,
   177→  });
   178→
   179→  // --- Phone lookup: determine if the customer is new or returning -------
   180→  // A "returning" (old) customer is one whose exact phone number already
   181→  // exists in the customers table. This drives the service-category filter
   182→  // (new customers see the "Dành cho khách hàng mới" cut; old customers see
   183→  // the regular "Dịch Vụ Cắt").
   184→  const trimmedPhone = customerPhone.trim();
   185→  const { data: phoneLookup } = useQuery<
   186→    { id: string; phone?: string | null }[] | null
   187→  >({
   188→    queryKey: ["dat-lich-phone-lookup", trimmedPhone],
   189→    queryFn: async () => {
   190→      if (!trimmedPhone) return null;
   191→      const res = await fetch(
   192→        `/api/supabase/customers?search=${encodeURIComponent(trimmedPhone)}&limit=20`
   193→      );
   194→      const json = await res.json();
   195→      if (!json.ok) return null;
   196→      return (json.data || []) as { id: string; phone?: string | null }[];
   197→    },
   198→    enabled: trimmedPhone.length >= 9,
   199→  });
   200→  const isOldCustomer = useMemo(() => {
   201→    if (!trimmedPhone || !phoneLookup) return false;
   202→    return phoneLookup.some((c) => c.phone === trimmedPhone);
   203→  }, [trimmedPhone, phoneLookup]);
   204→
   205→  // --- Customer autocomplete ----------------------------------------------
   206→  // Debounced search: when the user types ≥2 chars in EITHER the name or phone
   207→  // field, fetch matching customers (by name prefix OR phone prefix) and show
   208→  // a dropdown. Selecting a suggestion fills both name + phone.
   209→  useEffect(() => {
   210→    // Determine the search term based on which field the user is typing in.
   211→    const term =
   212→      activeField === "phone"
   213→        ? customerPhone.trim()
   214→        : activeField === "name"
   215→          ? customerName.trim()
   216→          : "";
   217→    let cancelled = false;
   218→    const t = setTimeout(async () => {
   219→      if (term.length < 2) {
   220→        if (!cancelled) {
   221→          setSuggestions([]);
   222→          setShowSuggestions(false);
   223→        }
   224→        return;
   225→      }
   226→      try {
   227→        const res = await fetch(
   228→          `/api/supabase/customers?search=${encodeURIComponent(term)}&limit=10`
   229→        );
   230→        const json = await res.json();
   231→        if (cancelled) return;
   232→        if (json.ok && Array.isArray(json.data)) {
   233→          // Filter to prefix matches on name OR phone (the API's `search`
   234→          // does an ilike contains; we refine to prefix for relevance).
   235→          const lower = term.toLowerCase();
   236→          const matches = (json.data as Array<{ id: string; name: string; phone?: string | null; code?: string | null }>)
   237→            .filter((c) => {
   238→              const nameMatch = (c.name || "").toLowerCase().startsWith(lower);
   239→              const phoneMatch = (c.phone || "").startsWith(term);
   240→              return nameMatch || phoneMatch;
   241→            })
   242→            .map((c) => ({ id: c.id, name: c.name, phone: c.phone ?? null, code: c.code ?? null }));
   243→          if (!cancelled) {
   244→            setSuggestions(matches);
   245→            setShowSuggestions(matches.length > 0);
   246→          }
   247→        }
   248→      } catch {
   249→        if (!cancelled) {
   250→          setSuggestions([]);
   251→          setShowSuggestions(false);
   252→        }
   253→      }
   254→    }, 250); // 250ms debounce
   255→    return () => {
   256→      cancelled = true;
   257→      clearTimeout(t);
   258→    };
   259→  }, [customerName, customerPhone, activeField]);
   260→
   261→  const selectSuggestion = (s: { name: string; phone: string | null }) => {
   262→    setCustomerName(s.name);
   263→    setCustomerPhone(s.phone || "");
   264→    setShowSuggestions(false);
   265→    setActiveField(null);
   266→  };
   267→  const customerType: "new" | "old" = isOldCustomer ? "old" : "new";
   268→
   269→  // Categories filtered by branch AND customer type (mirrors booking-dialog):
   270→  // - New customer (phone not in DB): hide "Dịch Vụ Cắt", keep new-customer cut.
   271→  // - Old customer (phone in DB): hide "Dành cho khách hàng mới - DV Cắt".
   272→  const categories = useMemo(() => {
   273→    const all = categoriesData || [];
   274→    const byBranch =
   275→      !selectedBranchId || selectedBranchId === "all"
   276→        ? all
   277→        : all.filter((c) => (c.branches || []).includes(selectedBranchId));
   278→    return byBranch.filter((c) => {
   279→      const name = c.name.toLowerCase();
   280→      const isNewCustomerCut = name.includes("dành cho khách hàng mới");
   281→      const isRegularCut =
   282→        name.includes("dịch vụ cắt") || name === "dịch vụ cắt";
   283→      if (customerType === "new") {
   284→        if (isRegularCut && !isNewCustomerCut) return false;
   285→      } else {
   286→        if (isNewCustomerCut) return false;
   287→      }
   288→      return true;
   289→    });
   290→  }, [categoriesData, selectedBranchId, customerType]);
   291→
   292→  const allServices = servicesData || [];
   293→  const allStaff = staffData || [];
   294→
   295→  // If a row's selected category is no longer visible after the customer-type
   296→  // filter changes (e.g. phone typed → old customer → "Dành cho KH mới" hidden),
   297→  // reset that row's service selection chain to avoid a stale hidden selection.
   298→  // Implemented as a render-phase update (the React 19 idiom for "adjust state
   299→  // when a derived value changes") rather than setState-in-effect, to avoid
   300→  // cascading renders and the react-hooks/set-state-in-effect lint error.
   301→  const categoriesKey = categories.map((c) => c.id).join("|");
   302→  const [prevCategoriesKey, setPrevCategoriesKey] = useState(categoriesKey);
   303→  if (categoriesKey !== prevCategoriesKey) {
   304→    setPrevCategoriesKey(categoriesKey);
   305→    setServiceRows((prev) => {
   306→      let changed = false;
   307→      const next = prev.map((row) => {
   308→        if (row.categoryId && !categories.some((c) => c.id === row.categoryId)) {
   309→          changed = true;
   310→          return { ...row, categoryId: "", serviceId: "", staffId: "" };
   311→        }
   312→        return row;
   313→      });
   314→      return changed ? next : prev;
   315→    });
   316→  }
   317→
   318→  // Per-row helpers: services filtered by the row's category, and staff
   319→  // filtered by the row's service (name-based narrowing mirrors booking-dialog).
   320→  const servicesForCategory = useCallback(
   321→    (catId: string) => allServices.filter((s) => !catId || s.category_id === catId),
   322→    [allServices]
   323→  );
   324→  const staffForService = useCallback(
   325→    (svcId: string) => {
   326→      if (!svcId) return allStaff;
   327→      const svc = allServices.find((s) => s.id === svcId);
   328→      if (!svc) return allStaff;
   329→      const n = (svc.name || "").toLowerCase();
   330→      if (n.includes("creative director")) {
   331→        return allStaff.filter((s) => s.groupName === "Creative Director");
   332→      }
   333→      if (n.includes("artist")) {
   334→        return allStaff.filter((s) => s.groupName === "Artist");
   335→      }
   336→      return allStaff;
   337→    },
   338→    [allStaff, allServices]
   339→  );
   340→
   341→  // --- Feasible-time computation (multi-service aware) --------------------
   342→  // Fetch the day's existing bookings (for the branch). We compute which
   343→  // HH:MM start times would cause a conflict for ANY of the selected staff.
   344→  // In the parallel model, all services start at the same time T; a time T
   345→  // is infeasible if, for ANY service row, the staff is busy during
   346→  // [T, T + service_duration]. So the hidden set = union across all rows.
   347→  const isoDay = useMemo(() => {
   348→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   349→    return m ? `${m[3]}-${m[2]}-${m[1]}` : null;
   350→  }, [bookingDate]);
   351→
   352→  // Collect the "active" rows: those with both a service and a staff chosen.
   353→  // These are the rows that participate in the feasibility computation.
   354→  const activeRows = useMemo(
   355→    () => serviceRows.filter((r) => r.serviceId && r.staffId),
   356→    [serviceRows]
   357→  );
   358→  const hasActiveRow = activeRows.length > 0;
   359→
   360→  const { data: dayBookings } = useQuery<ExistingBooking[]>({
   361→    queryKey: ["dat-lich-day-bookings", isoDay, selectedBranchId],
   362→    queryFn: async () => {
   363→      if (!isoDay) return [];
   364→      const params = new URLSearchParams();
   365→      params.set("page", "1");
   366→      params.set("limit", "200");
   367→      // Timezone-safe: isoDay is a Vietnam calendar day; convert to its UTC
   368→      // range so Supabase filters the correct window (no 7-hour shift).
   369→      const dl = localDayToUtcRange(isoDay);
   370→      params.set("date_from", dl.from);
   371→      params.set("date_to", dl.to);
   372→      if (selectedBranchId) params.set("branch_id", selectedBranchId);
   373→      const res = await fetch(`/api/supabase/bookings?${params.toString()}`);
   374→      const json = await res.json();
   375→      if (!json.ok) return [];
   376→      return (json.data || []) as ExistingBooking[];
   377→    },
   378→    enabled: !!isoDay && hasActiveRow,
   379→  });
   380→
   381→  // Compute hiddenHours + hiddenMinutes for the TimePicker across ALL active
   382→  // rows. A minute is hidden if it conflicts for ANY active row's staff.
   383→  const { hiddenHours, hiddenMinutes } = useMemo(() => {
   384→    const empty = { hiddenHours: new Set<string>(), hiddenMinutes: {} as Record<string, Set<string>> };
   385→    if (!isoDay || activeRows.length === 0) return empty;
   386→    const m = bookingDate.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   387→    if (!m) return empty;
   388→    // Use the +07:00 offset so dayBase is the correct UTC instant for VN
   389→    // midnight (same fix as the booking dialog + cashier service selector).
   390→    const dayBase = new Date(`${m[3]}-${m[2]}-${m[1]}T00:00:00+07:00`).getTime();
   391→    if (isNaN(dayBase)) return empty;
   392→
   393→    // Build per-row busy intervals + duration. Each row contributes its own
   394→    // staff's busy slots and its own service duration.
   395→    const rowSpecs: Array<{
   396→      intervals: { startMs: number; endMs: number }[];
   397→      durationMs: number;
   398→    }> = [];
   399→    for (const row of activeRows) {
   400→      const svc = allServices.find((s) => s.id === row.serviceId);
   401→      const durationMs = (svc?.duration || 60) * 60 * 1000;
   402→      if (durationMs <= 0) continue;
   403→      const intervals: { startMs: number; endMs: number }[] = [];
   404→      for (const b of dayBookings || []) {
   405→        if (b.status === "cancelled" || b.status === "no_show") continue;
   406→        const exStart = new Date(b.date_time).getTime();
   407→        if (isNaN(exStart)) continue;
   408→        for (const s of b.services || []) {
   409→          if (s.staff_id !== row.staffId) continue;
   410→          const dur =
   411→            (Number(s.duration) || Number(s.service?.duration) || 60) * 60 * 1000;
   412→          intervals.push({ startMs: exStart, endMs: exStart + dur });
   413→        }
   414→      }
   415→      rowSpecs.push({ intervals, durationMs });
   416→    }
   417→    if (rowSpecs.length === 0) return empty;
   418→
   419→    const hh = new Set<string>();
   420→    const hm: Record<string, Set<string>> = {};
   421→    const hourHasFeasible: Record<string, boolean> = {};
   422→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   423→    for (let min = 0; min < 60 * 24; min++) {
   424→      const startMs = dayBase + min * 60 * 1000;
   425→      // A minute conflicts if ANY row's staff is busy during [start, start+dur].
   426→      const overlaps = rowSpecs.some((spec) => {
   427→        const endMs = startMs + spec.durationMs;
   428→        return spec.intervals.some((iv) => startMs < iv.endMs && iv.startMs < endMs);
   429→      });
   430→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   431→      const mStr = String(min % 60).padStart(2, "0");
   432→      if (overlaps) {
   433→        if (!hm[hStr]) hm[hStr] = new Set<string>();
   434→        hm[hStr].add(mStr);
   435→      } else {
   436→        hourHasFeasible[hStr] = true;
   437→      }
   438→    }
   439→    for (let h = 0; h < 24; h++) {
   440→      const hStr = String(h).padStart(2, "0");
   441→      if (!hourHasFeasible[hStr]) hh.add(hStr);
   442→    }
   443→    return { hiddenHours: hh, hiddenMinutes: hm };
   444→  }, [activeRows, isoDay, dayBookings, bookingDate, allServices]);
   445→
   446→  // --- Row mutation helpers -----------------------------------------------
   447→  const updateRow = (id: string, patch: Partial<ServiceRow>) => {
   448→    setServiceRows((prev) => prev.map((r) => (r.id === id ? { ...r, ...patch } : r)));
   449→  };
   450→  const onRowCategoryChange = (id: string, catId: string) => {
   451→    // Changing category resets service + staff for that row.
   452→    updateRow(id, { categoryId: catId, serviceId: "", staffId: "" });
   453→    setBookingTime("");
   454→  };
   455→  const onRowServiceChange = (id: string, svcId: string) => {
   456→    // Changing service resets staff for that row.
   457→    updateRow(id, { serviceId: svcId, staffId: "" });
   458→    setBookingTime("");
   459→  };
   460→  const onRowStaffChange = (id: string, stId: string) => {
   461→    updateRow(id, { staffId: stId });
   462→    setBookingTime("");
   463→  };
   464→  const addRow = () => {
   465→    setServiceRows((prev) => [...prev, newRow()]);
   466→  };
   467→  const removeRow = (id: string) => {
   468→    setServiceRows((prev) => (prev.length > 1 ? prev.filter((r) => r.id !== id) : prev));
   469→    setBookingTime("");
   470→  };
   471→  // Changing the branch invalidates the whole service chain — staff and
   472→  // service availability are branch-specific. Reset all rows + time.
   473→  const onBranchChange = (id: string) => {
   474→    setSelectedBranchId(id);
   475→    setServiceRows([newRow()]);
   476→    setBookingTime("");
   477→  };
   478→
   479→  // Part 1 (customer info) must be fully filled before Part 2 (service) opens.
   480→  const part1Complete = !!(
   481→    customerName.trim() &&
   482→    customerPhone.trim() &&
   483→    bookingDate
   484→  );
   485→  // A row is "complete" when it has category + service + staff. At least one
   486→  // complete row is required before the time picker opens. Incomplete rows
   487→  // (empty or half-filled) are ignored at submit time.
   488→  const completeRows = useMemo(
   489→    () => serviceRows.filter((r) => r.categoryId && r.serviceId && r.staffId),
   490→    [serviceRows]
   491→  );
   492→  // Khung giờ requires at least 1 complete row (Nhóm DV + Dịch vụ + Kỹ thuật viên).
   493→  const timePickerReady = part1Complete && completeRows.length >= 1;
   494→
