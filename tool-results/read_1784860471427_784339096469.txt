   778→        }

   788→  // Compute hiddenHours + hiddenMinutes for a given entry.
   789→  const computeFeasible = (
   790→    entry: { staffId?: string; date?: string; serviceId?: string }
   791→  ): { hiddenHours: Set<string>; hiddenMinutes: Record<string, Set<string>> } => {
   792→    const empty = { hiddenHours: new Set<string>(), hiddenMinutes: {} };
   793→    if (!entry.staffId || !entry.date || !entry.serviceId) return empty;
   794→    const m = entry.date.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
   795→    if (!m) return empty;

   797→    const intervals = busyIntervals[`${entry.staffId}|${isoDay}`];
   798→    if (!intervals || intervals.length === 0) return empty;
   799→    const svc = services.find((s) => s.id === entry.serviceId);
   800→    const durationMs = (svc?.duration || 60) * 60 * 1000;
   801→    if (durationMs <= 0) return empty;

   803→    // advances `min` minutes from dayBase to enumerate every VN wall-clock
   804→    // minute of the day; min=540 → 09:00 VN (epoch). The busy intervals are
   805→    // also UTC-ms epochs (from `new Date(b.date_time).getTime()`), so the
   806→    // overlap comparison is apples-to-apples. Previously dayBase used
   807→    // Date.UTC(...,0,0,0) which is UTC midnight — that mapped min=540 to
   808→    // 09:00 UTC (= 16:00 VN), so a 16:00-VN booking (stored as 09:00 UTC)
   809→    // collided with the user's intended 09:00 VN slot. Fixed to VN midnight.
   810→    const dayBase = new Date(localDayStartUtc(isoDay)).getTime();
   811→    if (isNaN(dayBase)) return empty;
   812→    const hiddenHours = new Set<string>();
   813→    const hiddenMinutes: Record<string, Set<string>> = {};
   814→    const hourHasFeasible: Record<string, boolean> = {};
   815→    for (let h = 0; h < 24; h++) hourHasFeasible[String(h).padStart(2, "0")] = false;
   816→    for (let min = 0; min < 60 * 24; min++) {
   817→      const startMs = dayBase + min * 60 * 1000;
   818→      const endMs = startMs + durationMs;
   819→      const overlaps = intervals.some(
   820→        (iv) => startMs < iv.endMs && iv.startMs < endMs
   821→      );
   822→      const hStr = String(Math.floor(min / 60)).padStart(2, "0");
   823→      const mStr = String(min % 60).padStart(2, "0");
   824→      if (overlaps) {
   825→        if (!hiddenMinutes[hStr]) hiddenMinutes[hStr] = new Set<string>();
   826→        hiddenMinutes[hStr].add(mStr);
   827→      } else {
   828→        hourHasFeasible[hStr] = true;
   829→      }
   830→    }
   831→    for (let h = 0; h < 24; h++) {
   832→      const hStr = String(h).padStart(2, "0");
   833→      if (!hourHasFeasible[hStr]) hiddenHours.add(hStr);
   834→    }
   835→    return { hiddenHours, hiddenMinutes };
   836→  };

  1444→        }

  1446→        /* best-effort — don't block on network errors */
  1447→      }
  1448→    }

  1450→    // Parse the booking-level start (dd/MM/yyyy + HH:mm) into ms (UTC).
  1451→    const startM = data.date.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);

  1457→    // The form's date/time is a VIETNAM wall-clock value (the user enters 09:00

  1469→    }

  1472→    // running consecutively from the booking-level start.

  1474→      staffId: string;
  1475→      staffName: string;
  1476→      start: number; // epoch ms
  1477→      end: number; // epoch ms

  1479→      timeLabel: string;
  1480→      dateLabel: string;
  1481→    };

  1483→    // PARALLEL model (matches Task 18's submit + server): every service starts
  1484→    // at the SAME booking-level start time. `cursor` here is just the booking

  1486→    for (const entry of entries) {

  1490→      const duration = (svc?.duration || 60) * 60 * 1000; // ms
  1491→      const end = start + duration;
  1492→

  1497→      }
  1498→      // Format this slot's time label in VIETNAM time (the user entered a VN
  1499→      // time; `start` is the true UTC epoch of that VN instant, so convert back

  1501→      const slotDate = data.date;
  1502→      const slotTime = toVietnamTime(start);
  1503→      newSlots.push({

  1516→    // overlap, so this only fires for genuinely overlapping same-staff slots.)

  1518→      for (let j = i + 1; j < newSlots.length; j++) {
  1519→        const a = newSlots[i];
  1520→        const b = newSlots[j];

  1528→          );
  1529→        }
  1530→      }
  1531→    }
  1532→

  1534→    // All services share the booking-level date, so fetch that one day.

  1536→    const params = new URLSearchParams();
  1537→    params.set("page", "1");
  1538→    params.set("limit", "200");

  1543→    params.set("date_from", vd.from);
  1544→    params.set("date_to", vd.to);

  1547→      const res = await fetch(`/api/supabase/bookings?${params.toString()}`);
  1548→      const json = await res.json();

  1552→          // Skip cancelled / no_show bookings.
  1553→          const status = existing.status as string;
  1554→          if (status === "cancelled" || status === "no_show") continue;
  1555→          // When editing, skip the booking being edited itself.
  1556→          if (booking && existing.id === booking.id) continue;
  1557→          const exDateTime = existing.date_time as string;
  1558→          if (!exDateTime) continue;
  1559→          const exStart = new Date(exDateTime).getTime();
  1560→          if (isNaN(exStart)) continue;
  1561→          const exServices = (existing.services || []) as Array<{
  1562→            staff_id?: string | null;
  1563→            staff?: { name?: string } | null;
  1564→            service?: { duration?: number; name?: string } | null;
  1565→            service_id?: string;
  1566→          }>;
  1567→          for (const exSvc of exServices) {
  1568→            const exStaffId = exSvc.staff_id;

  1571→            const exEnd = exStart + exDuration;

  1573→            for (const ns of newSlots) {

  1575→                // Build a detailed conflict message that identifies the

  1577→                // service, staff, the FULL time range (start → end) of the

  1580→                // the staff can't tell how long the existing appointment runs,
  1581→                // which is the exact scenario the user described (a 9:30 90-min
  1582→                // service would overlap a 10:30-12:00 booking).
  1583→                const stf = staffList.find((s) => s.id === exStaffId);
  1584→                const staffName = stf?.name || exSvc.staff?.name || exStaffId;
  1585→                const svcName = exSvc.service?.name || "Dịch vụ";
  1586→                const exDurationMin = Math.round(exDuration / 60000);
  1587→                const exTimeStr = toVietnamTime(exStart);
  1588→                const exEndTimeStr = toVietnamTime(exEnd);
  1589→                const nsTimeStr = toVietnamTime(ns.start);
  1590→                const nsEndTimeStr = toVietnamTime(ns.end);
  1591→                const exDateStr = isoDay.split("-").reverse().join("/");
  1592→                const exCode = (existing.code as string) || "";
  1593→                const exCustName = (existing.customer as { name?: string } | null)?.name || "";
  1594→                const exBranchName = (existing.branch as { name?: string } | null)?.name || "";
  1595→                const statusLabel: Record<string, string> = {
  1596→                  pending: "Chờ xác nhận",
  1597→                  confirmed: "Đã xác nhận",
  1598→                  checkin: "Đang phục vụ",
  1599→                  checkout: "Đã thanh toán",
  1600→                  cancelled: "Đã huỷ",
  1601→                  no_show: "Không đến",
  1602→                };
  1603→                const exStatusLabel = status
  1604→                  ? statusLabel[status] || status
  1605→                  : "";
  1606→                const codeLine = exCode ? `Lịch ${exCode}` : "Một lịch đã đặt trước đó";
  1607→                const custLine = exCustName ? `• Khách: ${exCustName}\n` : "";
  1608→                const branchLine = exBranchName ? `• Chi nhánh: ${exBranchName}\n` : "";
  1609→                const statusLine = exStatusLabel ? `• Trạng thái: ${exStatusLabel}\n` : "";
  1610→                return (
  1611→                  `Không thể đặt lịch vì trùng thời gian với một lịch đã đặt trước đó.\n` +
  1612→                  `${codeLine}:\n` +
  1613→                  custLine +
  1614→                  `• Thợ: ${staffName}\n` +
  1615→                  `• Dịch vụ: ${svcName} (${exDurationMin} phút)\n` +
  1616→                  `• Thời gian: ${exTimeStr} - ${exEndTimeStr} ngày ${exDateStr}\n` +
  1617→                  branchLine +
  1618→                  statusLine +
  1619→                  `→ Trùng với dịch vụ mới bạn đang đặt (${nsTimeStr} - ${nsEndTimeStr} ngày ${exDateStr}). ` +
  1620→                  `Vui lòng chọn khung giờ hoặc thợ khác.`
  1621→                );
  1622→              }
  1623→            }
  1624→          }
  1625→        }
  1626→      }
  1627→    } catch {
  1628→      // Network/parse error — don't block the submit; the server may still reject.
  1629→    }
  1630→
  1631→    return "";
  1632→  };

  1634→  // A booking is LOCKED (read-only) when it's in a terminal state:
  1635→  // - checkin: customer is currently being served → locked until checkout.
  1636→  // - checkout: already paid → immutable historical record.

  1639→  // The dialog still opens so the user can review the details, but every

  1641→  // (Terminal statuses also free up the time slot — a NEW booking can be

  1643→  // bookings in the conflict check.)

  1652→      : booking?.status === "checkout"
  1653→        ? "Lịch hẹn đã thanh toán không thể chỉnh sửa."

  1655→          ? "Lịch hẹn đã hủy không thể chỉnh sửa. Bạn có thể tạo lịch hẹn mới cho khung giờ này vì slot đã được giải phóng."
  1656→          : booking?.status === "no_show"
  1657→            ? "Lịch hẹn đánh dấu 'Không đến' không thể chỉnh sửa. Bạn có thể tạo lịch hẹn mới cho khung giờ này vì slot đã được giải phóng."
  1658→            : "";

  1660→  // ------------------------------------------------------------------
  1661→  // resolveCustomerId — shared customer-resolution logic used by BOTH the
  1662→  // single-customer flow (top-level phone/name) and the multi-customer flow
  1663→  // (per-slot phone/name). Returns the resolved customer_id, or an error.
  1664→  //   1. If both phone+name empty → create a "Khách vãng lai" guest record.
  1665→  //   2. If phone (or name) typed → match an existing customer by exact
  1666→  //      phone (or exact name when no phone), else create a new customer.
  1667→  // ------------------------------------------------------------------
  1668→  const resolveCustomerId = async (
  1669→    phone: string,
  1670→    name: string
  1671→  ): Promise<{ ok: boolean; customerId?: string; error?: string }> => {
  1672→    const typedPhone = phone.trim();
  1673→    const typedName = name.trim();
  1674→    if (!typedPhone && !typedName) {
  1675→      // Create a guest record (same as the walk-in flow).
  1676→      try {
  1677→        const res = await fetch("/api/supabase/customers", {
  1678→          method: "POST",
  1679→          headers: { "Content-Type": "application/json" },
  1680→          body: JSON.stringify({
  1681→            name: WALKIN_SOURCE_NAME,
  1682→            phone: "",
  1683→            customer_type: "guest",
  1684→            source_id: WALKIN_SOURCE_ID,
  1685→            branch_id: selectedBranchId || null,
  1686→          }),
  1687→        });
  1688→        const json = await res.json();
  1689→        if (json.ok && json.data?.id) {
  1690→          return { ok: true, customerId: json.data.id };
  1691→        }
  1692→        return { ok: false, error: json.error || "Không thể tạo khách vãng lai" };
  1693→      } catch {
  1694→        return { ok: false, error: "Không thể tạo khách vãng lai" };
  1695→      }
  1696→    }
  1697→    try {
  1698→      const params = new URLSearchParams();
  1699→      params.set("limit", "20");
  1700→      if (typedPhone) params.set("search", typedPhone);
  1701→      else params.set("search", typedName);
  1702→      const matchRes = await fetch(`/api/supabase/customers?${params.toString()}`);
  1703→      const matchJson = await matchRes.json();
  1704→      let matchedId: string | null = null;
  1705→      let matchedName: string | null = null;
  1706→      if (matchJson.ok && Array.isArray(matchJson.data)) {
  1707→        const exact = (matchJson.data as Array<{ id: string; phone?: string | null; name?: string }>).find(
  1708→          (c) =>
  1709→            typedPhone
  1710→              ? (c.phone || "") === typedPhone
  1711→              : (c.name || "") === typedName
  1712→        );
  1713→        if (exact) {
  1714→          matchedId = exact.id;
  1715→          matchedName = (exact.name as string) || "";
  1716→        }
  1717→      }
  1718→      if (matchedId) {
  1719→        // If the cashier typed a name that differs from the existing
  1720→        // customer's current name (and the typed name is non-empty), update
  1721→        // the customer profile so it reflects the latest info. This fulfils
  1722→        // the requirement that entered info is saved to the customer profile.
  1723→        if (typedName && typedName !== matchedName) {
  1724→          try {
  1725→            await fetch(`/api/supabase/customers/${matchedId}`, {
  1726→              method: "PUT",
  1727→              headers: { "Content-Type": "application/json" },
  1728→              body: JSON.stringify({ name: typedName }),
  1729→            });
  1730→            queryClient.invalidateQueries({ queryKey: queryKeys.customers.all });
  1731→          } catch {
  1732→            // Best-effort — the booking still proceeds with the matched id.
  1733→          }
  1734→        }
  1735→        return { ok: true, customerId: matchedId };
  1736→      }
  1737→      // No match → create a new customer with the typed phone + name.
  1738→      const createRes = await fetch("/api/supabase/customers", {
  1739→        method: "POST",
  1740→        headers: { "Content-Type": "application/json" },
  1741→        body: JSON.stringify({
  1742→          name: typedName || typedPhone,
  1743→          phone: typedPhone || undefined,
  1744→          branch_id: selectedBranchId || null,
  1745→        }),
  1746→      });
  1747→      const createJson = await createRes.json();
  1748→      if (createJson.ok && createJson.data?.id) {
  1749→        queryClient.invalidateQueries({ queryKey: queryKeys.customers.all });
  1750→        return { ok: true, customerId: createJson.data.id };
  1751→      }
  1752→      // Edge case: the POST returned 409 because a customer with the same
  1753→      // phone was created in the meantime (race condition or the search
  1754→      // missed it). Use the existing customer returned in the 409 body.
  1755→      if (createJson.existing_customer?.id) {
  1756→        // Update the name if a different name was typed.
  1757→        if (typedName && typedName !== (createJson.existing_customer.name || "")) {
  1758→          try {
  1759→            await fetch(`/api/supabase/customers/${createJson.existing_customer.id}`, {
  1760→              method: "PUT",
  1761→              headers: { "Content-Type": "application/json" },
  1762→              body: JSON.stringify({ name: typedName }),
  1763→            });
  1764→          } catch {
  1765→            // best-effort
  1766→          }
  1767→        }
  1768→        queryClient.invalidateQueries({ queryKey: queryKeys.customers.all });
  1769→        return { ok: true, customerId: createJson.existing_customer.id };
  1770→      }
  1771→      return { ok: false, error: createJson.error || "Không thể tạo khách hàng" };
  1772→    } catch {
  1773→      return { ok: false, error: "Không thể tạo khách hàng" };
  1774→    }
  1775→  };

  1777→  // ------------------------------------------------------------------
  1778→  // Multi-customer submit handler (CREATE mode, numberOfCustomers >= 2).
  1779→  //   - scheduleMode "same": 1 POST with all N services, customer_id =
  1780→  //     slot 0's resolved id. The other slots' customer info is NOT sent
  1781→  //     to the API (the booking API only accepts one customer_id per
  1782→  //     booking — a documented limitation).
  1783→  //   - scheduleMode "different": N separate POSTs, each with 1 service
  1784→  //     and that slot's resolved customer_id. All N bookings share the
  1785→  //     same date_time + branch_id + status. Failures are reported but
  1786→  //     already-created bookings stay.
  1787→  // ------------------------------------------------------------------
  1788→  const handleMultiCustomerSubmit = async (data: BookingFormValues) => {
  1789→    setIsMultiSubmitting(true);
  1790→    try {
  1791→      // 1. Resolve customer_id for each slot.
  1792→      const resolvedIds: string[] = [];
  1793→      for (let i = 0; i < data.services.length; i++) {
  1794→        const svc = data.services[i];
  1795→        const result = await resolveCustomerId(
  1796→          svc.customerPhone || "",
  1797→          svc.customerName || ""
  1798→        );
  1799→        if (!result.ok || !result.customerId) {
  1800→          setConflictMessage(
  1801→            `Khách #${i + 1}: ${result.error || "Không thể xác định khách hàng"}`
  1802→          );
  1803→          return;
  1804→        }
  1805→        resolvedIds.push(result.customerId);
  1806→      }
  1807→
  1808→      // 2. Validate the booking (conflict check). Works for both modes —
  1809→      //    within-form staff uniqueness + existing-booking overlap.
  1810→      const error = await validateBooking(data);
  1811→      if (error) {
  1812→        setConflictMessage(error);
  1813→        return;
  1814→      }
  1815→
  1816→      // 3. Build the shared date_time + booking-level fields.
  1817→      const firstDate = data.date.split("/").reverse().join("-");
  1818→      const dateTime = `${firstDate}T${data.time}:00+07:00`;
  1819→
  1820→      if (scheduleMode === "same") {
  1821→        // --- Cùng lịch: 1 POST with all services ---
  1822→        // The booking API stores only ONE customer_id per booking (slot 0's
  1823→        // resolved id). To preserve the per-slot customer mapping so display
  1824→        // sites (Staff View, Cashier) can show each service's own customer,
  1825→        // we persist a structured `[[MULTI]]` JSON block in the booking's
  1826→        // `note` field. Slots with no phone/name → walkin:true (rendered as
  1827→        // "Khách vãng lai"). The cashier's own typed note is kept inside the
