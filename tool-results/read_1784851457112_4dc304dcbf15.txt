    68→    68→
    69→    69→interface StaffColumn {
    70→    70→  staff: Staff | null;
    71→    71→  segments: ServiceSegment[];
    72→    72→}
    73→    73→
    74→    74→/** A service-segment of a booking, assigned to one staff's timeline. A
    75→    75→ *  segment = one or more CONSECUTIVE services performed by the SAME staff,
    76→    76→ *  merged into a single slot. Services run back-to-back (cursor advances by
    77→    77→ *  each service's duration), so consecutive in sort_order = adjacent in time.
    78→    78→ *  A run of same-staff services becomes one merged slot whose height ∝ the
    79→    79→ *  sum of their durations; different-staff services stay in separate slots
    80→    80→ *  (each in its own staff's column). Matches the server-side conflict check in
    81→    81→ *  /api/supabase/bookings/route.ts. */
    82→    82→interface ServiceSegment {
    83→    83→  booking: Booking;
    84→    84→  /** One or more consecutive same-staff services merged into this slot. */
    85→    85→  services: BookingServiceRow[];
    86→    86→  staffId: string;
    87→    87→  staffName: string;
    88→    88→  /** 0-based index of this segment (group) within the booking's segment list. */
    89→    89→  segmentIndex: number;
    90→    90→  /** Total number of segments (groups) in the booking (for the "1/2" badge). */
    91→    91→  totalSegments: number;
    92→    92→  /** Minutes from the timeline's START_HOUR (08:00) to this segment's start.
    93→    93→   *  May be negative when the booking starts before 08:00 — callers clamp
    94→    94→   *  into the visible window. */
    95→    95→  startMin: number;
    96→    96→  /** This segment's total duration = Σ of its services' durations (minutes). */
    97→    97→  duration: number;
    98→    98→}
    99→    99→
   100→   100→/** Pixels per hour on the vertical timeline. Slot height is proportional to
   101→   101→ *  duration. 1.5× the previous value (60 → 90) so each hour band is taller and
   102→   102→ *  multi-service segment blocks have more room to show their content. */
   103→   103→const PX_PER_HOUR = 90;
   104→   104→/** Timeline starts at 08:00 and ends at 21:00 (13 hours). */
   105→   105→const START_HOUR = 8;
   106→   106→const END_HOUR = 21;
   107→   107→/** Width of each staff column. The timeline scrolls horizontally when there

   109→   109→const STAFF_COL_WIDTH = 180;
   110→   110→/** Width of the fixed "Giờ" column on the left. */
   111→   111→const TIME_COL_WIDTH = 64;
   112→   112→/** Minimum column width when dragging (prevents columns from vanishing). */
   113→   113→const MIN_COL_WIDTH = 50;
   114→   114→/** Width of the drag handle on each header cell's right edge. */
   115→   115→const RESIZER_WIDTH = 8;

   117→   117→// Hairdresser groups — same filter as the booking dialog. Only these groups

   119→   119→const HAIRDRESSER_GROUPS = ["Artist", "Creative Director", "Master", "Junior"];

   121→   121→export function BookingStaffView({
   122→   122→  bookings,
   123→   123→  currentDate,
   124→   124→  dateRange,
   125→   125→  onBookingClick,
   126→   126→  onStatusChange,
   127→   127→  onEdit,
   128→   128→  onAssignStaff,
   129→   129→  onDelete,
   130→   130→  onShowInvoice,
   131→   131→  onSlotClick,
   132→   132→  branchId,
   133→   133→}: BookingStaffViewProps) {
   134→   134→  const { hasPermission } = useAuthStore();
   135→   135→  const canViewCustomerPhone = hasPermission("view_customer_phone");
   136→   136→  const canBookPastDate = hasPermission("book_past_date");
   137→   137→  // Whether the viewed date is in the past (slot creation blocked when
   138→   138→  // !canBookPastDate).
   139→   139→  const isPastDate = (() => {
   140→   140→    if (!currentDate) return false;
   141→   141→    const today = new Date();
   142→   142→    today.setHours(0, 0, 0, 0);
   143→   143→    const d = new Date(currentDate);
   144→   144→    d.setHours(0, 0, 0, 0);
   145→   145→    return d < today;
   146→   146→  })();
   147→   147→  const slotLocked = isPastDate && !canBookPastDate;
   148→   148→
   149→   149→  // Hour-band resize (single-day timeline only). The user drags a horizontal
   150→   150→  // handle on the "Giờ" axis to grow/shrink every hour band. Persisted.
   151→   151→  const MIN_PX_PER_HOUR = 40;
   152→   152→  const MAX_PX_PER_HOUR = 200;
   153→   153→  const [pxPerHour, setPxPerHour] = useState<number>(() => {
   154→   154→    try {
   155→   155→      const raw = localStorage.getItem("crm-staff-timegrid-pxph");
   156→   156→      const n = raw ? parseInt(raw, 10) : NaN;
   157→   157→      if (!isNaN(n) && n >= MIN_PX_PER_HOUR && n <= MAX_PX_PER_HOUR) return n;
   158→   158→    } catch { /* ignore */ }
   159→   159→    return PX_PER_HOUR;
   160→   160→  });
   161→   161→  const setPxPerHourPersisted = (n: number) => {
   162→   162→    const clamped = Math.max(MIN_PX_PER_HOUR, Math.min(MAX_PX_PER_HOUR, n));
   163→   163→    setPxPerHour(clamped);
   164→   164→    try { localStorage.setItem("crm-staff-timegrid-pxph", String(clamped)); } catch { /* ignore */ }
   165→   165→  };
   166→   166→
   167→   167→  // Tracks which (column, hour) slot the mouse is hovering over in the
   168→   168→  // single-day staff-column layout. Used to highlight ONLY that slot (not
   169→   169→  // the entire column) with an orange overlay on hover.
   170→   170→  const [hoveredSlot, setHoveredSlot] = useState<{ col: number; hour: number } | null>(null);
   171→   171→
   172→   172→  // Number of days in the selected range. When 1 day → use the legacy
   173→   173→  // per-staff column layout (one column per staff). When 2+ days → use the
   174→   174→  // DayRangeGrid (one column per day). Capped at 5 visible days; 6+ scroll.
   175→   175→  const rangeDayCount = useMemo(() => {
   176→   176→    if (!dateRange) return 0;
   177→   177→    const from = new Date(dateRange.from);
   178→   178→    from.setHours(0, 0, 0, 0);
   179→   179→    const to = new Date(dateRange.to);
   180→   180→    to.setHours(0, 0, 0, 0);
   181→   181→    if (to < from) return 1;
   182→   182→    return Math.round((to.getTime() - from.getTime()) / 86400000) + 1;
   183→   183→  }, [dateRange]);
   184→   184→  const useMultiDayLayout = rangeDayCount >= 2;
   185→   185→
   186→   186→  // Hour labels for the left axis (08:00 → 21:00).
   187→   187→  const hours = useMemo(() => {
   188→   188→    const h: number[] = [];
   189→   189→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
   190→   190→    return h;
   191→   191→  }, []);
   192→   192→
   193→   193→  // Fetch ALL active staff for the selected branch (so every staff has a column,
   194→   194→  // not just those with bookings today). Mirrors the BookingDialog's staff query.
   195→   195→  const [allStaff, setAllStaff] = useState<Staff[]>([]);
   196→   196→  useEffect(() => {
   197→   197→    let cancelled = false;
   198→   198→    async function fetchStaff() {
   199→   199→      if (!branchId) {
   200→   200→        setAllStaff([]);
   201→   201→        return;
   202→   202→      }
   203→   203→      try {
   204→   204→        const res = await fetch(
   205→   205→          `/api/supabase/staff?branch_id=${encodeURIComponent(branchId)}&active=true&limit=200`
   206→   206→        );
   207→   207→        const json = await res.json();
   208→   208→        if (cancelled) return;
   209→   209→        if (!json.ok) {
   210→   210→          setAllStaff([]);
   211→   211→          return;
   212→   212→        }
   213→   213→        // Only show hairdresser groups (same filter as BookingDialog).
   214→   214→        const list = ((json.data as Array<Record<string, unknown>>) || [])
   215→   215→          .filter((s) => {
   216→   216→            const groupName = (s.group as { name?: string } | null)?.name;
   217→   217→            return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   218→   218→          })
   219→   219→          .map((s) => ({
   220→   220→            id: s.id as string,
   221→   221→            name: s.name as string,
   222→   222→            groupName: (s.group as { name?: string } | null)?.name,
   223→   223→          }));
   224→   224→        setAllStaff(list);
   225→   225→      } catch {
   226→   226→        if (!cancelled) setAllStaff([]);
   227→   227→      }
   228→   228→    }
   229→   229→    fetchStaff();
   230→   230→    return () => { cancelled = true; };
   231→   231→  }, [branchId]);
   232→   232→
   233→   233→  // Build staff columns from the FULL staff list. Each booking is split into
   234→   234→  // SEGMENTS where each segment = a run of same-staff services merged into one
   235→   235→  // slot. PARALLEL model: every service in the booking starts at the SAME
   236→   236→  // booking-level start time (each runs on a different staff, simultaneously),
   237→   237→  // so every segment's startMin = the booking's startMin (NO cursor advance).
   238→   238→  // A run of same-staff services still merges into one taller block (duration
   239→   239→  // = sum of their durations). Each segment lands in the column of the staff
   240→   240→  // who performs it. Segments whose services have no staff go to a
   241→   241→  // "Chưa xếp nhân viên" column at the end.
   242→   242→  const staffColumns = useMemo<StaffColumn[]>(() => {
   243→   243→    const columns: StaffColumn[] = allStaff.map((staff) => ({
   244→   244→      staff,
   245→   245→      segments: [],
   246→   246→    }));
   247→   247→    const byStaffId = new Map<string, StaffColumn>();
   248→   248→    for (const col of columns) {
   249→   249→      if (col.staff) byStaffId.set(col.staff.id, col);
   250→   250→    }
   251→   251→
   252→   252→    const unassigned: ServiceSegment[] = [];
   253→   253→    for (const booking of bookings) {
   254→   254→      // Services ordered by sort_order for stable segment ordering.
   255→   255→      const services = getAllServices(booking)
   256→   256→        .slice()
   257→   257→        .sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
   258→   258→      // Group CONSECUTIVE services performed by the SAME staff into one
   259→   259→      // segment (a run of same-staff services merges into one slot).
   260→   260→      const groups: BookingServiceRow[][] = [];
   261→   261→      for (const s of services) {
   262→   262→        const last = groups.length > 0 ? groups[groups.length - 1] : null;
   263→   263→        const lastStaffId = last && last.length > 0 ? last[last.length - 1].staff_id : null;
   264→   264→        if (last && lastStaffId && lastStaffId === s.staff_id) {
   265→   265→          last.push(s);
   266→   266→        } else {
   267→   267→          groups.push([s]);
   268→   268→        }
   269→   269→      }
   270→   270→      const totalSegments = groups.length;
   271→   271→      const startMin = getBookingStartMinutes(booking);
   272→   272→      // PARALLEL: every segment starts at the booking's startMin. Previously a
   273→   273→      // cursor advanced by each segment's duration (consecutive model), making
   274→   274→      // the 2nd service of a multi-staff booking appear at startMin + 1st
   275→   275→      // service's duration (e.g. 11:00 instead of 09:30). Fixed to parallel.
   276→   276→      groups.forEach((group, i) => {
   277→   277→        const duration = group.reduce((sum, s) => sum + (s.service?.duration || 0), 0);
   278→   278→        const staffId = group[0].staff_id || "";
   279→   279→        const seg: ServiceSegment = {
   280→   280→          booking,
   281→   281→          services: group,
   282→   282→          staffId,
   283→   283→          staffName: group[0].staff?.name || "—",
   284→   284→          segmentIndex: i,
   285→   285→          totalSegments,
   286→   286→          startMin: startMin ?? 0,
   287→   287→          duration,
   288→   288→        };
   289→   289→        if (!staffId) {
   290→   290→          unassigned.push(seg);
   291→   291→          return;
   292→   292→        }
   293→   293→        const col = byStaffId.get(staffId);
   294→   294→        if (col) {
   295→   295→          col.segments.push(seg);
   296→   296→        } else {
   297→   297→          // Staff on the booking but not in allStaff (different branch / group).
   298→   298→          // Add an extra column so the segment is still visible.
   299→   299→          const newCol: StaffColumn = {
   300→   300→            staff: { id: staffId, name: seg.staffName },
   301→   301→            segments: [seg],
   302→   302→          };
   303→   303→          columns.push(newCol);
   304→   304→          byStaffId.set(staffId, newCol);
   305→   305→        }
   306→   306→      });
   307→   307→    }
   308→   308→
   309→   309→    // The "Chưa xếp nhân viên" (unassigned) column is placed FIRST (before the
   310→   310→    // real staff columns) so it sits right after the "Giờ" column on the left,
   311→   311→    // per the user's request. Previously it was appended at the end (rightmost)
   312→   312→    // which made it easy to miss. Putting it first draws attention to bookings
   313→   313→    // that still need a staff assignment — the cashier sees them immediately
   314→   314→    // and can click "Xếp nhân viên" on each block to assign one.
   315→   315→    if (unassigned.length > 0) {
   316→   316→      columns.unshift({ staff: null, segments: unassigned });
   317→   317→    }
   318→   318→    // Sort each column's segments by PRIORITY so that overlapping segments
   319→   319→    // stack with UNPAID bookings (pending/new/confirmed/checkin) ON TOP of
   320→   320→    // cancelled/no_show ones. In the single-day staff view, segments are
   321→   321→    // absolutely positioned and later-in-DOM renders on top — so we sort
   322→   322→    // cancelled FIRST (bottom) and unpaid LAST (top). This implements the
   323→   323→    // user's requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy"
   324→   324→    // (a cancelled booking at 10:30-12:00 should NOT cover an unpaid booking
   325→   325→    // at 10:00-11:30; the unpaid one must be fully visible).
   326→   326→    const SEG_PRIORITY: Record<string, number> = {
   327→   327→      // cancelled / no-show → priority 0 (rendered FIRST → at the BOTTOM)
   328→   328→      cancelled: 0,
   329→   329→      no_show: 0,
   330→   330→      // paid → priority 1
   331→   331→      checkout: 1,
   332→   332→      // unpaid / active → priority 2 (rendered LAST → ON TOP)
   333→   333→      pending: 2,
   334→   334→      new: 2,
   335→   335→      confirmed: 2,
   336→   336→      checkin: 2,
   337→   337→    };
   338→   338→    for (const col of columns) {
   339→   339→      col.segments.sort((a, b) => {
   340→   340→        const pa = SEG_PRIORITY[a.booking.status] ?? 1;
   341→   341→        const pb = SEG_PRIORITY[b.booking.status] ?? 1;
   342→   342→        if (pa !== pb) return pa - pb; // lower priority first (rendered below)
   343→   343→        // Same priority → keep chronological order (earlier start first).
   344→   344→        return a.startMin - b.startMin;
   345→   345→      });
   346→   346→    }
   347→   347→    return columns;
   348→   348→  }, [allStaff, bookings]);
   349→   349→
   350→   350→  // Total timeline height in pixels. Uses the resizable pxPerHour (single-day)
   351→   351→  // or the constant (multi-day DayRangeGrid uses its own DAYGRID_ROW_HEIGHT).
   352→   352→  const timelineHeight = (END_HOUR - START_HOUR) * pxPerHour;
   353→   353→
   354→   354→  // ---- Column resizing (with persistence) ----
   355→   355→  // columnWidths[0] = "Giờ" column width; columnWidths[1..N] = staff columns.
   356→   356→  // State-driven so the user can drag column edges to resize (same UX as the
   357→   357→  // global TableResizer, but adapted for this CSS-grid layout which isn't a
   358→   358→  // <table> and therefore isn't handled by TableResizer).
   359→   359→  //
   360→   360→  // Persistence: widths are saved to localStorage keyed by branch + column
   361→   361→  // count, so the user's column customizations survive page switches and
   362→   362→  // browser refreshes. When the column count changes (e.g. branch switch
   363→   363→  // loads a different number of staff), the saved widths for that exact
   364→   364→  // column count are restored; if none match, defaults are used.
   365→   365→  const STORAGE_PREFIX = "crm-staff-grid-widths:";
   366→   366→
   367→   367→  /** Build the localStorage key for the current branch + column count. */
   368→   368→  const storageKey = `${STORAGE_PREFIX}${branchId || "all"}::n${staffColumns.length}`;
   369→   369→
   370→   370→  /** Read saved widths from localStorage. Returns null if none/invalid. */
   371→   371→  function readSavedWidths(key: string): number[] | null {
   372→   372→    try {
   373→   373→      const raw = localStorage.getItem(key);
   374→   374→      if (!raw) return null;
   375→   375→      const parsed = JSON.parse(raw);
   376→   376→      if (!Array.isArray(parsed) || parsed.length === 0) return null;
   377→   377→      if (!parsed.every((n: unknown) => typeof n === "number" && n > 0)) return null;
   378→   378→      return parsed as number[];
   379→   379→    } catch {
   380→   380→      return null;
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  /** Save widths to localStorage (best-effort). */
   385→   385→  function writeSavedWidths(key: string, widths: number[]) {
   386→   386→    try {
   387→   387→      localStorage.setItem(key, JSON.stringify(widths));
   388→   388→    } catch {
   389→   389→      // best-effort — localStorage may be full or disabled
   390→   390→    }
   391→   391→  }
   392→   392→
   393→   393→  // Compute the initial widths: try saved widths for this branch+count, fall
   394→   394→  // back to defaults.
   395→   395→  const [columnWidths, setColumnWidths] = useState<number[]>(() => {
   396→   396→    const saved = readSavedWidths(storageKey);
   397→   397→    if (saved && saved.length === staffColumns.length + 1) return saved;
   398→   398→    return [TIME_COL_WIDTH, ...Array(staffColumns.length).fill(STAFF_COL_WIDTH)];
   399→   399→  });
   400→   400→
   401→   401→  // Reset widths whenever the column count changes (e.g. staff data finishes
   402→   402→  // loading, or the branch changes). Uses the "adjust state during render"
   403→   403→  // pattern (React docs) instead of an effect so the lint rule
   404→   404→  // `react-hooks/set-state-in-effect` is satisfied. Restores saved widths
   405→   405→  // for the new column count if available.
   406→   406→  const [prevStorageKey, setPrevStorageKey] = useState(storageKey);
   407→   407→  if (storageKey !== prevStorageKey) {
   408→   408→    setPrevStorageKey(storageKey);
   409→   409→    const saved = readSavedWidths(storageKey);
   410→   410→    if (saved && saved.length === staffColumns.length + 1) {
   411→   411→      setColumnWidths(saved);
   412→   412→    } else {
   413→   413→      setColumnWidths([
   414→   414→        TIME_COL_WIDTH,
   415→   415→        ...Array(staffColumns.length).fill(STAFF_COL_WIDTH),
   416→   416→      ]);
   417→   417→    }
   418→   418→  }

  1151→  1151→/** Minutes from the timeline's START_HOUR (08:00) to the booking's start time.
  1152→  1152→ *  May be negative (booking starts before 08:00) — callers clamp into the
  1153→  1153→ *  visible window. Returns null when date_time is missing or unparseable. */
  1154→  1154→function getBookingStartMinutes(booking: Booking): number | null {
  1155→  1155→  const dt = booking.date_time;
  1156→  1156→  if (!dt) return null;
  1157→  1157→  // Use the timezone-safe Vietnam time helper. Supabase normalizes stored
  1158→  1158→  // date_time offsets to +00:00 (UTC), so the "THH:MM" segment is the UTC
  1159→  1159→  // time — NOT the Vietnam time the user entered. Parsing the segment directly
  1160→  1160→  // made bookings display at the wrong hour (and outside the visible window).
  1161→  1161→  const hhmm = toVietnamTime(dt);
  1162→  1162→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1163→  1163→  if (!m) return null;
  1164→  1164→  const hour = parseInt(m[1], 10);
  1165→  1165→  const minute = parseInt(m[2], 10);
  1166→  1166→  if (isNaN(hour) || isNaN(minute)) return null;
  1167→  1167→  return (hour - START_HOUR) * 60 + minute;
  1168→  1168→}

  1216→  1216→/** Build the list of calendar days (inclusive) between `from` and `to`.
  1217→  1217→ *  Each Date is normalized to local midnight so day-of-month arithmetic is
  1218→  1218→ *  stable regardless of the host timezone. */
  1219→  1219→function buildDaysInRange(from: Date, to: Date): Date[] {
  1220→  1220→  const days: Date[] = [];
  1221→  1221→  const start = new Date(from);
  1222→  1222→  start.setHours(0, 0, 0, 0);
  1223→  1223→  const end = new Date(to);
  1224→  1224→  end.setHours(0, 0, 0, 0);
  1225→  1225→  if (end < start) return [start];
  1226→  1226→  const cur = new Date(start);
  1227→  1227→  while (cur <= end) {
  1228→  1228→    days.push(new Date(cur));
  1229→  1229→    cur.setDate(cur.getDate() + 1);
  1230→  1230→  }
  1231→  1231→  return days;
  1232→  1232→}

  1234→  1234→/** Build a `YYYY-MM-DD` dayKey from a local Date object (no UTC shift). */
  1235→  1235→function dateToLocalDayKey(d: Date): string {
  1236→  1236→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
  1237→  1237→}
  1238→  1238→
  1239→  1239→/** Extract { dayKey, hour, minute } from the booking's `date_time` as VIETNAM
  1240→  1240→ *  wall-clock values. Post-migration (Task 6), `date_time` is stored as a UTC
  1241→  1241→ *  ISO string (e.g. "2026-07-14T03:30:00+00:00" represents 10:30 VN), so the
  1242→  1242→ *  raw "THH:MM" segment is the UTC time, NOT the VN time the user entered.
  1243→  1243→ *  Using `toVietnamDay` + `toVietnamTime` ensures the booking lands in the
  1244→  1244→ *  correct row+column of the multi-day staff grid (a 10:30 VN booking lands in
  1245→  1245→ *  the "10:00" row of column "2026-07-14", not the skipped "03:00" row). */
  1246→  1246→function getBookingDayHour(booking: Booking): { dayKey: string; hour: number; minute: number } | null {
  1247→  1247→  const dt = booking.date_time;
  1248→  1248→  if (!dt || typeof dt !== "string") return null;
  1249→  1249→  const dayKey = toVietnamDay(dt);
  1250→  1250→  const hhmm = toVietnamTime(dt);
  1251→  1251→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1252→  1252→  if (!m) return null;
  1253→  1253→  const hour = parseInt(m[1], 10);
  1254→  1254→  const minute = parseInt(m[2], 10);
  1255→  1255→  if (isNaN(hour) || isNaN(minute)) return null;
  1256→  1256→  return { dayKey, hour, minute };
  1257→  1257→}
  1258→  1258→
  1259→  1259→/** "dd/MM" label for a day. */
  1260→  1260→function dayLabel(d: Date): string {
  1261→  1261→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`;
  1262→  1262→}
  1263→  1263→
  1264→  1264→/** Short weekday name (T2, T3, … T7, CN). */
  1265→  1265→function weekdayLabel(d: Date): string {
  1266→  1266→  const vi = ["CN", "T2", "T3", "T4", "T5", "T6", "T7"];
  1267→  1267→  return vi[d.getDay()];
  1268→  1268→}

  1270→  1270→function DayRangeGrid({
  1271→  1271→  dateRange,
  1272→  1272→  bookings,
  1273→  1273→  onBookingClick,
  1274→  1274→  canViewCustomerPhone,
  1275→  1275→  onSlotClick,
  1276→  1276→  slotLocked,
  1277→  1277→  onEdit,
  1278→  1278→  onAssignStaff,
  1279→  1279→}: DayRangeGridProps) {
  1280→  1280→  const days = useMemo(() => buildDaysInRange(dateRange.from, dateRange.to), [dateRange.from, dateRange.to]);
  1281→  1281→  const hours = useMemo(() => {
  1282→  1282→    const h: number[] = [];
  1283→  1283→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
  1284→  1284→    return h;
  1285→  1285→  }, []);
  1286→  1286→
  1287→  1287→  // Measure the container width so we can compute the per-day column width
  1288→  1288→  // when 5+ days are selected. The "minimum readable" layout is defined as
  1289→  1289→  // 4 days filling the table; 5+ days keep the SAME column width as 4 days
  1290→  1290→  // would have had, and scroll horizontally. Without measuring we'd have to
  1291→  1291→  // hard-code a width that may not match the actual table width.
  1292→  1292→  const containerRef = useRef<HTMLDivElement>(null);
  1293→  1293→  const [containerWidth, setContainerWidth] = useState(0);
  1294→  1294→  useEffect(() => {
  1295→  1295→    const el = containerRef.current;
  1296→  1296→    if (!el) return;
  1297→  1297→    const update = () => setContainerWidth(el.getBoundingClientRect().width);
  1298→  1298→    update();
  1299→  1299→    const ro = new ResizeObserver(update);
  1300→  1300→    ro.observe(el);
  1301→  1301→    return () => ro.disconnect();
  1302→  1302→  }, []);
  1303→  1303→
  1304→  1304→  // Index bookings by `${dayKey}|${hour}` for O(1) cell lookup.
  1305→  1305→  const cellMap = useMemo(() => {
  1306→  1306→    const m = new Map<string, Booking[]>();
  1307→  1307→    for (const b of bookings) {
  1308→  1308→      const info = getBookingDayHour(b);
  1309→  1309→      if (!info) continue;
  1310→  1310→      // Skip bookings outside the visible hour range.
  1311→  1311→      if (info.hour < START_HOUR || info.hour > END_HOUR) continue;
  1312→  1312→      const key = `${info.dayKey}|${info.hour}`;
  1313→  1313→      const arr = m.get(key);
  1314→  1314→      if (arr) arr.push(b);
  1315→  1315→      else m.set(key, [b]);
  1316→  1316→    }
  1317→  1317→    // Sort each cell's bookings:
  1318→  1318→    // 1. UNPAID bookings (pending/new/confirmed/checkin) take priority and
  1319→  1319→    //    render ON TOP of cancelled/no_show ones. When a cancelled booking
  1320→  1320→    //    would occupy the same slot as an unpaid booking (same start minute),
  1321→  1321→    //    the unpaid one wins the visible position. This implements the user's
  1322→  1322→    //    requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy".
  1323→  1323→    // 2. Within the same priority, sort by start minute (earlier → top) so
  1324→  1324→    //    the cell still reads top-to-bottom chronologically.
  1325→  1325→    const PRIORITY: Record<string, number> = {
  1326→  1326→      // unpaid / active → priority 0 (top)
  1327→  1327→      pending: 0,
  1328→  1328→      new: 0,
  1329→  1329→      confirmed: 0,
  1330→  1330→      checkin: 0,
  1331→  1331→      // paid → priority 1
  1332→  1332→      checkout: 1,
  1333→  1333→      // cancelled / no-show → priority 2 (bottom, "hidden under" unpaid)
  1334→  1334→      cancelled: 2,
  1335→  1335→      no_show: 2,
  1336→  1336→    };
  1337→  1337→    for (const arr of m.values()) {
  1338→  1338→      arr.sort((a, b) => {
  1339→  1339→        const pa = PRIORITY[a.status] ?? 1;
  1340→  1340→        const pb = PRIORITY[b.status] ?? 1;
  1341→  1341→        if (pa !== pb) return pa - pb;
  1342→  1342→        const ai = getBookingDayHour(a);
  1343→  1343→        const bi = getBookingDayHour(b);
  1344→  1344→        return (ai?.minute ?? 0) - (bi?.minute ?? 0);
  1345→  1345→      });
  1346→  1346→      // When an UNPAID booking and a CANCELLED booking share the EXACT same
  1347→  1347→      // start minute (a true slot collision — e.g. a cancelled booking at
  1348→  1348→      // 10:30 and a new confirmed booking at 10:30 for the same staff), hide
  1349→  1349→      // the cancelled one entirely so the unpaid booking's chip is fully
  1350→  1350→      // visible (not stacked under/over the cancelled chip). Bookings with
  1351→  1351→      // different start minutes stay (they're different time slots within
  1352→  1352→      // the same hour cell).
  1353→  1353→      const hasUnpaid = arr.some(
  1354→  1354→        (b) => (PRIORITY[b.status] ?? 1) === 0
  1355→  1355→      );
  1356→  1356→      if (hasUnpaid) {
  1357→  1357→        // Build a set of start-minutes that have an unpaid booking.
  1358→  1358→        const unpaidMinutes = new Set<number>();
  1359→  1359→        for (const b of arr) {
  1360→  1360→          if ((PRIORITY[b.status] ?? 1) === 0) {
  1361→  1361→            const info = getBookingDayHour(b);
  1362→  1362→            if (info) unpaidMinutes.add(info.minute);
  1363→  1363→          }
  1364→  1364→        }
  1365→  1365→        // Remove cancelled/no_show bookings whose start minute collides with
  1366→  1366→        // an unpaid booking's start minute. Keep non-colliding cancelled ones
  1367→  1367→        // (different slot within the same hour) so the staff still sees them.
  1368→  1368→        for (let i = arr.length - 1; i >= 0; i--) {
  1369→  1369→          const b = arr[i];
  1370→  1370→          const isCancelled = b.status === "cancelled" || b.status === "no_show";
  1371→  1371→          if (!isCancelled) continue;
  1372→  1372→          const info = getBookingDayHour(b);
  1373→  1373→          if (info && unpaidMinutes.has(info.minute)) {
  1374→  1374→            arr.splice(i, 1);
  1375→  1375→          }
  1376→  1376→        }
  1377→  1377→      }
  1378→  1378→    }
  1379→  1379→    return m;
  1380→  1380→  }, [bookings]);
  1381→  1381→
  1382→  1382→  // Layout decision:
  1383→  1383→  //   - 1–4 days → columns fill the full table width (1fr each), no scroll.
  1384→  1384→  //   - 5+ days → each column keeps the SAME width it would have had when 4
  1385→  1385→  //     days filled the table ((containerWidth - timeCol) / 4), and the table

  1387→  1387→  //     4-day baseline width, no matter how many days are selected.
  1388→  1388→  const dayCount = days.length;
  1389→  1389→  const needsScroll = dayCount > DAYGRID_BASELINE_DAYS;
  1390→  1390→  // Per-column width when scrolling. Derived from the measured container
  1391→  1391→  // width so 5+ days look exactly like 4 days did (just with more columns
  1392→  1392→  // off-screen to the right). Fall back to a sane default before the
  1393→  1393→  // ResizeObserver fires (containerWidth === 0 on first render).
  1394→  1394→  const scrollColWidth = containerWidth > 0
  1395→  1395→    ? Math.max(DAYGRID_DAY_COL_MIN_WIDTH, (containerWidth - DAYGRID_TIME_COL_WIDTH) / DAYGRID_BASELINE_DAYS)
  1396→  1396→    : 240;
  1397→  1397→  const totalWidth = needsScroll
  1398→  1398→    ? DAYGRID_TIME_COL_WIDTH + dayCount * scrollColWidth
  1399→  1399→    : undefined; // undefined → 100% width (fills the table)
  1400→  1400→
  1401→  1401→  const handleCellClick = (day: Date, hour: number, e: React.MouseEvent<HTMLDivElement>) => {
  1402→  1402→    if (!onSlotClick || slotLocked) return;
  1403→  1403→    if (e.target !== e.currentTarget) return; // only empty area
  1404→  1404→    const dd = String(day.getDate()).padStart(2, "0");
  1405→  1405→    const mo = String(day.getMonth() + 1).padStart(2, "0");
  1406→  1406→    const yyyy = day.getFullYear();
  1407→  1407→    onSlotClick({
  1408→  1408→      date: `${dd}/${mo}/${yyyy}`,
  1409→  1409→      time: `${String(hour).padStart(2, "0")}:00`,
  1410→  1410→      staffId: null,
  1411→  1411→    });
  1412→  1412→  };
  1413→  1413→
  1414→  1414→  return (
  1415→  1415→    <div className="border bg-white flex flex-col" ref={containerRef} style={{ maxHeight: "calc(100vh - 200px)" }}>
  1416→  1416→      <div className="flex-1 min-h-0 overflow-auto" data-grid-resizable>
  1417→  1417→        <div
  1418→  1418→          style={
  1419→  1419→            needsScroll
  1420→  1420→              ? { width: `${totalWidth}px`, minWidth: `${totalWidth}px` }
  1421→  1421→              : { width: "100%" }
  1422→  1422→          }
  1423→  1423→        >
  1424→  1424→          {/* Header: "Giờ" + day columns — sticky at top so it stays during vertical scroll */}
  1425→  1425→          <div
  1426→  1426→            data-grid-header
  1427→  1427→            className="grid border-b bg-gray-50 sticky top-0 z-10"
  1428→  1428→            style={{
  1429→  1429→              gridTemplateColumns: needsScroll
  1430→  1430→                ? `${DAYGRID_TIME_COL_WIDTH}px repeat(${dayCount}, ${scrollColWidth}px)`
  1431→  1431→                : `${DAYGRID_TIME_COL_WIDTH}px repeat(${dayCount}, minmax(0, 1fr))`,
  1432→  1432→            }}
  1433→  1433→          >
  1434→  1434→            <div className="border-r p-2 text-center text-xs font-semibold text-gray-600">
  1435→  1435→              Giờ
  1436→  1436→            </div>
  1437→  1437→            {days.map((day, idx) => {
  1438→  1438→              const isToday = (() => {
  1439→  1439→                const t = new Date();
  1440→  1440→                t.setHours(0, 0, 0, 0);
  1441→  1441→                const d = new Date(day);
  1442→  1442→                d.setHours(0, 0, 0, 0);
  1443→  1443→                return t.getTime() === d.getTime();
  1444→  1444→              })();
  1445→  1445→              return (
  1446→  1446→                <div
  1447→  1447→                  key={idx}
  1448→  1448→                  className={`border-r p-2 text-center ${isToday ? "bg-emerald-50" : ""}`}
  1449→  1449→                >
  1450→  1450→                  <div className="text-[11px] font-medium text-gray-500">
  1451→  1451→                    {weekdayLabel(day)}
  1452→  1452→                  </div>
  1453→  1453→                  <div className={`text-sm font-semibold ${isToday ? "text-emerald-700" : "text-gray-800"}`}>
  1454→  1454→                    {dayLabel(day)}

  1460→  1460→
  1461→  1461→          {/* Body: one row per hour. Uses minHeight (not height) so rows expand

  1463→  1463→          <div data-grid-body>

  1468→  1468→                style={{
  1469→  1469→                  gridTemplateColumns: needsScroll
  1470→  1470→                    ? `${DAYGRID_TIME_COL_WIDTH}px repeat(${dayCount}, ${scrollColWidth}px)`
  1471→  1471→                    : `${DAYGRID_TIME_COL_WIDTH}px repeat(${dayCount}, minmax(0, 1fr))`,

  1477→  1477→                  {String(hour).padStart(2, "0")}:00

  1528→  1528→/** Compact booking chip for the multi-day table cells. Non-overlapping: each
  1529→  1529→ *  chip is a flex child in its cell so multiple bookings stack vertically.
  1530→  1530→ *  Hovering opens a rich HoverCard popover showing full booking + invoice
  1531→  1531→ *  details (date/time, customer name+phone, services, products, staff,
  1532→  1532→ *  promotion, tip, total). */
  1533→  1533→function BookingChip({
  1534→  1534→  booking,
  1535→  1535→  canViewCustomerPhone,
  1536→  1536→  onClick,
  1537→  1537→  onEdit,
  1538→  1538→  onAssignStaff,
  1539→  1539→}: {
  1540→  1540→  booking: Booking;
  1541→  1541→  canViewCustomerPhone: boolean;
  1542→  1542→  onClick: () => void;
  1543→  1543→  /** Open the booking edit dialog (used by the "Xếp nhân viên" button in the
  1544→  1544→      hover popover for services with no staff). Optional — when omitted, the
  1545→  1545→      assign-staff button is hidden in the popover. */
  1546→  1546→  onEdit?: () => void;
  1547→  1547→  /** Open the dedicated "Xếp nhân viên" dialog (preferred over onEdit for the
  1548→  1548→      assign-staff button). When provided, the popover's "Xếp nhân viên" button
  1549→  1549→      calls this; falls back to onEdit when not provided. */
  1550→  1550→  onAssignStaff?: () => void;
  1551→  1551→}) {
  1552→  1552→  const serviceRows = getAllServices(booking);
  1553→  1553→  const svc = serviceRows[0] || null;
  1554→  1554→  const serviceName = svc?.service?.name || "Dịch vụ";
  1555→  1555→  const staffName = svc?.staff?.name || "—";
  1556→  1556→  const totalDuration = serviceRows.reduce((sum, s) => sum + (s.service?.duration || 0), 0);
  1557→  1557→  const isPaid = booking.status === "checkout";
  1558→  1558→  const isCancelled = booking.status === "cancelled";
  1559→  1559→  const isNoShow = booking.status === "no_show";
  1560→  1560→  const isCheckin = booking.status === "checkin";
  1561→  1561→  const isReviewing = useIsReviewing(booking.id);
  1562→  1562→  // SAME scheme as SegmentBlock above (user's color spec):
  1563→  1563→  // confirmed → blue, checkin (chưa bấm TT) → green, checkin + đang review (đã bấm TT) → darker purple, checkout → white.
