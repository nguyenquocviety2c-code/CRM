    68→
    69→interface StaffColumn {
    70→  staff: Staff | null;
    71→  segments: ServiceSegment[];
    72→}
    73→
    74→/** A service-segment of a booking, assigned to one staff's timeline. A
    75→ *  segment = one or more CONSECUTIVE services performed by the SAME staff,
    76→ *  merged into a single slot. Services run back-to-back (cursor advances by
    77→ *  each service's duration), so consecutive in sort_order = adjacent in time.
    78→ *  A run of same-staff services becomes one merged slot whose height ∝ the
    79→ *  sum of their durations; different-staff services stay in separate slots
    80→ *  (each in its own staff's column). Matches the server-side conflict check in
    81→ *  /api/supabase/bookings/route.ts. */
    82→interface ServiceSegment {
    83→  booking: Booking;
    84→  /** One or more consecutive same-staff services merged into this slot. */
    85→  services: BookingServiceRow[];
    86→  staffId: string;
    87→  staffName: string;
    88→  /** 0-based index of this segment (group) within the booking's segment list. */
    89→  segmentIndex: number;
    90→  /** Total number of segments (groups) in the booking (for the "1/2" badge). */
    91→  totalSegments: number;
    92→  /** Minutes from the timeline's START_HOUR (08:00) to this segment's start.
    93→   *  May be negative when the booking starts before 08:00 — callers clamp
    94→   *  into the visible window. */
    95→  startMin: number;
    96→  /** This segment's total duration = Σ of its services' durations (minutes). */
    97→  duration: number;
    98→}
    99→
   100→/** Pixels per hour on the vertical timeline. Slot height is proportional to
   101→ *  duration. 1.5× the previous value (60 → 90) so each hour band is taller and
   102→ *  multi-service segment blocks have more room to show their content. */
   103→const PX_PER_HOUR = 90;
   104→/** Timeline starts at 08:00 and ends at 21:00 (13 hours). */
   105→const START_HOUR = 8;
   106→const END_HOUR = 21;
   107→/** Width of each staff column. The timeline scrolls horizontally when there
   108→ *  are more staff than fit on screen. */
   109→const STAFF_COL_WIDTH = 180;
   110→/** Width of the fixed "Giờ" column on the left. */
   111→const TIME_COL_WIDTH = 64;
   112→/** Minimum column width when dragging (prevents columns from vanishing). */
   113→const MIN_COL_WIDTH = 50;
   114→/** Width of the drag handle on each header cell's right edge. */
   115→const RESIZER_WIDTH = 8;

   117→// Hairdresser groups — same filter as the booking dialog. Only these groups

   119→const HAIRDRESSER_GROUPS = ["Artist", "Creative Director", "Master", "Junior"];

   121→export function BookingStaffView({
   122→  bookings,
   123→  currentDate,
   124→  dateRange,
   125→  onBookingClick,
   126→  onStatusChange,
   127→  onEdit,
   128→  onAssignStaff,
   129→  onDelete,
   130→  onShowInvoice,
   131→  onSlotClick,
   132→  branchId,
   133→}: BookingStaffViewProps) {
   134→  const { hasPermission } = useAuthStore();
   135→  const canViewCustomerPhone = hasPermission("view_customer_phone");
   136→  const canBookPastDate = hasPermission("book_past_date");
   137→  // Whether the viewed date is in the past (slot creation blocked when
   138→  // !canBookPastDate).
   139→  const isPastDate = (() => {
   140→    if (!currentDate) return false;
   141→    const today = new Date();
   142→    today.setHours(0, 0, 0, 0);
   143→    const d = new Date(currentDate);
   144→    d.setHours(0, 0, 0, 0);
   145→    return d < today;
   146→  })();
   147→  const slotLocked = isPastDate && !canBookPastDate;
   148→
   149→  // Hour-band resize (single-day timeline only). The user drags a horizontal
   150→  // handle on the "Giờ" axis to grow/shrink every hour band. Persisted.
   151→  const MIN_PX_PER_HOUR = 40;
   152→  const MAX_PX_PER_HOUR = 200;
   153→  const [pxPerHour, setPxPerHour] = useState<number>(() => {
   154→    try {
   155→      const raw = localStorage.getItem("crm-staff-timegrid-pxph");
   156→      const n = raw ? parseInt(raw, 10) : NaN;
   157→      if (!isNaN(n) && n >= MIN_PX_PER_HOUR && n <= MAX_PX_PER_HOUR) return n;
   158→    } catch { /* ignore */ }
   159→    return PX_PER_HOUR;
   160→  });
   161→  const setPxPerHourPersisted = (n: number) => {
   162→    const clamped = Math.max(MIN_PX_PER_HOUR, Math.min(MAX_PX_PER_HOUR, n));
   163→    setPxPerHour(clamped);
   164→    try { localStorage.setItem("crm-staff-timegrid-pxph", String(clamped)); } catch { /* ignore */ }
   165→  };
   166→
   167→  // Tracks which (column, hour) slot the mouse is hovering over in the
   168→  // single-day staff-column layout. Used to highlight ONLY that slot (not
   169→  // the entire column) with an orange overlay on hover.
   170→  const [hoveredSlot, setHoveredSlot] = useState<{ col: number; hour: number } | null>(null);
   171→
   172→  // Number of days in the selected range. When 1 day → use the legacy
   173→  // per-staff column layout (one column per staff). When 2+ days → use the
   174→  // DayRangeGrid (one column per day). Capped at 5 visible days; 6+ scroll.
   175→  const rangeDayCount = useMemo(() => {
   176→    if (!dateRange) return 0;
   177→    const from = new Date(dateRange.from);
   178→    from.setHours(0, 0, 0, 0);
   179→    const to = new Date(dateRange.to);
   180→    to.setHours(0, 0, 0, 0);
   181→    if (to < from) return 1;
   182→    return Math.round((to.getTime() - from.getTime()) / 86400000) + 1;
   183→  }, [dateRange]);
   184→  const useMultiDayLayout = rangeDayCount >= 2;
   185→
   186→  // Hour labels for the left axis (08:00 → 21:00).
   187→  const hours = useMemo(() => {
   188→    const h: number[] = [];
   189→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
   190→    return h;
   191→  }, []);
   192→
   193→  // Fetch ALL active staff for the selected branch (so every staff has a column,
   194→  // not just those with bookings today). Mirrors the BookingDialog's staff query.
   195→  const [allStaff, setAllStaff] = useState<Staff[]>([]);
   196→  useEffect(() => {
   197→    let cancelled = false;
   198→    async function fetchStaff() {
   199→      if (!branchId) {
   200→        setAllStaff([]);
   201→        return;
   202→      }
   203→      try {
   204→        const res = await fetch(
   205→          `/api/supabase/staff?branch_id=${encodeURIComponent(branchId)}&active=true&limit=200`
   206→        );
   207→        const json = await res.json();
   208→        if (cancelled) return;
   209→        if (!json.ok) {
   210→          setAllStaff([]);
   211→          return;
   212→        }
   213→        // Only show hairdresser groups (same filter as BookingDialog).
   214→        const list = ((json.data as Array<Record<string, unknown>>) || [])
   215→          .filter((s) => {
   216→            const groupName = (s.group as { name?: string } | null)?.name;
   217→            return groupName && HAIRDRESSER_GROUPS.includes(groupName);
   218→          })
   219→          .map((s) => ({
   220→            id: s.id as string,
   221→            name: s.name as string,
   222→            groupName: (s.group as { name?: string } | null)?.name,
   223→          }));
   224→        setAllStaff(list);
   225→      } catch {
   226→        if (!cancelled) setAllStaff([]);
   227→      }
   228→    }
   229→    fetchStaff();
   230→    return () => { cancelled = true; };
   231→  }, [branchId]);
   232→
   233→  // Build staff columns from the FULL staff list. Each booking is split into
   234→  // SEGMENTS where each segment = a run of same-staff services merged into one
   235→  // slot. PARALLEL model: every service in the booking starts at the SAME
   236→  // booking-level start time (each runs on a different staff, simultaneously),
   237→  // so every segment's startMin = the booking's startMin (NO cursor advance).
   238→  // A run of same-staff services still merges into one taller block (duration
   239→  // = sum of their durations). Each segment lands in the column of the staff
   240→  // who performs it. Segments whose services have no staff go to a
   241→  // "Chưa xếp nhân viên" column at the end.
   242→  const staffColumns = useMemo<StaffColumn[]>(() => {
   243→    const columns: StaffColumn[] = allStaff.map((staff) => ({
   244→      staff,
   245→      segments: [],
   246→    }));
   247→    const byStaffId = new Map<string, StaffColumn>();
   248→    for (const col of columns) {
   249→      if (col.staff) byStaffId.set(col.staff.id, col);
   250→    }
   251→
   252→    const unassigned: ServiceSegment[] = [];
   253→    for (const booking of bookings) {
   254→      // Services ordered by sort_order for stable segment ordering.
   255→      const services = getAllServices(booking)
   256→        .slice()
   257→        .sort((a, b) => (a.sort_order ?? 0) - (b.sort_order ?? 0));
   258→      // Group CONSECUTIVE services performed by the SAME staff into one
   259→      // segment (a run of same-staff services merges into one slot).
   260→      const groups: BookingServiceRow[][] = [];
   261→      for (const s of services) {
   262→        const last = groups.length > 0 ? groups[groups.length - 1] : null;
   263→        const lastStaffId = last && last.length > 0 ? last[last.length - 1].staff_id : null;
   264→        if (last && lastStaffId && lastStaffId === s.staff_id) {
   265→          last.push(s);
   266→        } else {
   267→          groups.push([s]);
   268→        }
   269→      }
   270→      const totalSegments = groups.length;
   271→      const startMin = getBookingStartMinutes(booking);
   272→      // PARALLEL: every segment starts at the booking's startMin. Previously a
   273→      // cursor advanced by each segment's duration (consecutive model), making
   274→      // the 2nd service of a multi-staff booking appear at startMin + 1st
   275→      // service's duration (e.g. 11:00 instead of 09:30). Fixed to parallel.
   276→      groups.forEach((group, i) => {
   277→        const duration = group.reduce((sum, s) => sum + (s.service?.duration || 0), 0);
   278→        const staffId = group[0].staff_id || "";
   279→        const seg: ServiceSegment = {
   280→          booking,
   281→          services: group,
   282→          staffId,
   283→          staffName: group[0].staff?.name || "—",
   284→          segmentIndex: i,
   285→          totalSegments,
   286→          startMin: startMin ?? 0,
   287→          duration,
   288→        };
   289→        if (!staffId) {
   290→          unassigned.push(seg);
   291→          return;
   292→        }
   293→        const col = byStaffId.get(staffId);
   294→        if (col) {
   295→          col.segments.push(seg);
   296→        } else {
   297→          // Staff on the booking but not in allStaff (different branch / group).
   298→          // Add an extra column so the segment is still visible.
   299→          const newCol: StaffColumn = {
   300→            staff: { id: staffId, name: seg.staffName },
   301→            segments: [seg],
   302→          };
   303→          columns.push(newCol);
   304→          byStaffId.set(staffId, newCol);
   305→        }
   306→      });
   307→    }
   308→
   309→    // The "Chưa xếp nhân viên" (unassigned) column is placed FIRST (before the
   310→    // real staff columns) so it sits right after the "Giờ" column on the left,
   311→    // per the user's request. Previously it was appended at the end (rightmost)
   312→    // which made it easy to miss. Putting it first draws attention to bookings
   313→    // that still need a staff assignment — the cashier sees them immediately
   314→    // and can click "Xếp nhân viên" on each block to assign one.
   315→    if (unassigned.length > 0) {
   316→      columns.unshift({ staff: null, segments: unassigned });
   317→    }
   318→    // Sort each column's segments by PRIORITY so that overlapping segments
   319→    // stack with UNPAID bookings (pending/new/confirmed/checkin) ON TOP of
   320→    // cancelled/no_show ones. In the single-day staff view, segments are
   321→    // absolutely positioned and later-in-DOM renders on top — so we sort
   322→    // cancelled FIRST (bottom) and unpaid LAST (top). This implements the
   323→    // user's requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy"
   324→    // (a cancelled booking at 10:30-12:00 should NOT cover an unpaid booking
   325→    // at 10:00-11:30; the unpaid one must be fully visible).
   326→    const SEG_PRIORITY: Record<string, number> = {
   327→      // cancelled / no-show → priority 0 (rendered FIRST → at the BOTTOM)
   328→      cancelled: 0,
   329→      no_show: 0,
   330→      // paid → priority 1
   331→      checkout: 1,
   332→      // unpaid / active → priority 2 (rendered LAST → ON TOP)
   333→      pending: 2,
   334→      new: 2,
   335→      confirmed: 2,
   336→      checkin: 2,
   337→    };
   338→    for (const col of columns) {
   339→      col.segments.sort((a, b) => {
   340→        const pa = SEG_PRIORITY[a.booking.status] ?? 1;
   341→        const pb = SEG_PRIORITY[b.booking.status] ?? 1;
   342→        if (pa !== pb) return pa - pb; // lower priority first (rendered below)
   343→        // Same priority → keep chronological order (earlier start first).
   344→        return a.startMin - b.startMin;
   345→      });
   346→    }
   347→    return columns;
   348→  }, [allStaff, bookings]);
   349→
   350→  // Total timeline height in pixels. Uses the resizable pxPerHour (single-day)
   351→  // or the constant (multi-day DayRangeGrid uses its own DAYGRID_ROW_HEIGHT).
   352→  const timelineHeight = (END_HOUR - START_HOUR) * pxPerHour;
   353→
   354→  // ---- Column resizing (with persistence) ----
   355→  // columnWidths[0] = "Giờ" column width; columnWidths[1..N] = staff columns.
   356→  // State-driven so the user can drag column edges to resize (same UX as the
   357→  // global TableResizer, but adapted for this CSS-grid layout which isn't a
   358→  // <table> and therefore isn't handled by TableResizer).
   359→  //
   360→  // Persistence: widths are saved to localStorage keyed by branch + column
   361→  // count, so the user's column customizations survive page switches and
   362→  // browser refreshes. When the column count changes (e.g. branch switch
   363→  // loads a different number of staff), the saved widths for that exact
   364→  // column count are restored; if none match, defaults are used.
   365→  const STORAGE_PREFIX = "crm-staff-grid-widths:";
   366→
   367→  /** Build the localStorage key for the current branch + column count. */
   368→  const storageKey = `${STORAGE_PREFIX}${branchId || "all"}::n${staffColumns.length}`;
   369→
   370→  /** Read saved widths from localStorage. Returns null if none/invalid. */
   371→  function readSavedWidths(key: string): number[] | null {
   372→    try {
   373→      const raw = localStorage.getItem(key);
   374→      if (!raw) return null;
   375→      const parsed = JSON.parse(raw);
   376→      if (!Array.isArray(parsed) || parsed.length === 0) return null;
   377→      if (!parsed.every((n: unknown) => typeof n === "number" && n > 0)) return null;
   378→      return parsed as number[];
   379→    } catch {
   380→      return null;
   381→    }
   382→  }
   383→
   384→  /** Save widths to localStorage (best-effort). */
   385→  function writeSavedWidths(key: string, widths: number[]) {
   386→    try {
   387→      localStorage.setItem(key, JSON.stringify(widths));
   388→    } catch {
   389→      // best-effort — localStorage may be full or disabled
   390→    }
   391→  }
   392→
   393→  // Compute the initial widths: try saved widths for this branch+count, fall
   394→  // back to defaults.
   395→  const [columnWidths, setColumnWidths] = useState<number[]>(() => {
   396→    const saved = readSavedWidths(storageKey);
   397→    if (saved && saved.length === staffColumns.length + 1) return saved;
   398→    return [TIME_COL_WIDTH, ...Array(staffColumns.length).fill(STAFF_COL_WIDTH)];
   399→  });
   400→
   401→  // Reset widths whenever the column count changes (e.g. staff data finishes
   402→  // loading, or the branch changes). Uses the "adjust state during render"
   403→  // pattern (React docs) instead of an effect so the lint rule
   404→  // `react-hooks/set-state-in-effect` is satisfied. Restores saved widths
   405→  // for the new column count if available.
   406→  const [prevStorageKey, setPrevStorageKey] = useState(storageKey);
   407→  if (storageKey !== prevStorageKey) {
   408→    setPrevStorageKey(storageKey);
   409→    const saved = readSavedWidths(storageKey);
   410→    if (saved && saved.length === staffColumns.length + 1) {
   411→      setColumnWidths(saved);
   412→    } else {
   413→      setColumnWidths([
   414→        TIME_COL_WIDTH,
   415→        ...Array(staffColumns.length).fill(STAFF_COL_WIDTH),
   416→      ]);
   417→    }
   418→  }
   419→
   420→  // Permission gate: only staff whose group has "resize_table" can drag the
   421→  // column edges to resize. When the permission is OFF, the grid uses `1fr`
   422→  // for all staff columns (equal width, fills container, NOT resizable). When
   423→  // the permission is ON, the grid uses persisted pixel widths (resizable via
   424→  // the drag handles on each column's right edge; dragging one staff column
   425→  // resizes ALL staff columns synchronously so they stay uniform).
   426→  const canResizeTable = hasPermission("resize_table");
   427→
   428→  // Build the CSS grid-template-columns string.
   429→  // - canResizeTable = true: "Giờ" col (px) + staff cols (px, from
   430→  //   columnWidths). Dragging a staff column's edge resizes all staff columns
   431→  //   together (sync mode). The total may exceed the container → horizontal
   432→  //   scroll.
   433→  // - canResizeTable = false: "Giờ" col (px) + staff cols (1fr each). Always
   434→  //   equal width, fills the container, not draggable.
   435→  const gridTemplate = canResizeTable
   436→    ? columnWidths.map((w) => `${w}px`).join(" ")
   437→    : `${columnWidths[0] || TIME_COL_WIDTH}px repeat(${Math.max(staffColumns.length, 1)}, minmax(0, 1fr))`;
   438→  // Total scrollable width. Only meaningful when canResizeTable = true (the
   439→  // px-based grid may exceed the container). When 1fr, the container is 100%.
   440→  const totalWidth = columnWidths.reduce((s, w) => s + w, 0);
   441→
   442→  // Start dragging a column's right edge. `idx` is the column index in
   443→  // columnWidths (0 = "Giờ", 1..N = staff columns).
   444→  //
   445→  // SYNC MODE: dragging ANY staff column's edge resizes ALL staff columns
   446→  // together (the delta is applied to every staff column uniformly). This
   447→  // keeps the grid visually uniform — all staff columns always have the same
   448→  // width. The "Giờ" column (idx 0) is excluded from the sync: dragging its
   449→  // edge still resizes only it (since the time column has a different purpose
   450→  // and width). The user's request was specifically about staff columns, so
   451→  // we sync those.
   452→  const startColumnResize = useCallback(
   453→    (e: React.MouseEvent, idx: number) => {
   454→      e.preventDefault();
   455→      e.stopPropagation();
   456→      const startX = e.clientX;
   457→      const startWidth = columnWidths[idx] || STAFF_COL_WIDTH;
   458→      // For staff columns, capture every staff column's start width so we can
   459→      // apply the same delta to all of them synchronously.
   460→      const staffStartWidths = columnWidths.slice(1);
   461→      document.body.style.cursor = "col-resize";
   462→      document.body.style.userSelect = "none";
   463→      // Capture the current key so the save after drag uses the right one
   464→      // (in case the branch changes mid-drag — extremely unlikely but safe).
   465→      const dragKey = storageKey;
   466→      let saveTimer: ReturnType<typeof setTimeout> | null = null;
   467→      const onMove = (ev: MouseEvent) => {
   468→        const delta = ev.clientX - startX;
   469→        if (idx === 0) {
   470→          // Dragging the "Giờ" column's edge → resize only that column.
   471→          const newWidth = Math.max(MIN_COL_WIDTH, startWidth + delta);
   472→          setColumnWidths((prev) => {
   473→            const next = [...prev];
   474→            next[0] = newWidth;
   475→            return next;
   476→          });
   477→        } else {
   478→          // Dragging a STAFF column's edge → apply the delta to ALL staff
   479→          // columns synchronously so they stay uniform. Each staff column's
   480→          // new width is clamped to MIN_COL_WIDTH independently (in practice
   481→          // they all clamp together since they share the same start width +
   482→          // delta).
   483→          setColumnWidths((prev) => {
   484→            const next = [...prev];

   830→/** A single service-segment block on a staff's timeline. The block occupies

   832→ *  THIS service runs for, on THIS staff's column. A multi-service / multi-staff
   833→ *  booking therefore appears as separate blocks in each staff's column, each
   834→ *  occupying its own time slice (services run back-to-back). Hover opens the
   835→ *  full booking popover (all services, invoice, actions). */
   836→function SegmentBlock({
   837→  segment,
   838→  onClick,
   839→  onStatusChange,
   840→  onEdit,
   841→  onDelete,
   842→  onAssignStaff,
   843→  pxPerHour = PX_PER_HOUR,
   844→}: {
   845→  segment: ServiceSegment;
   846→  onClick: () => void;
   847→  onStatusChange: (status: BookingStatusType) => void;
   848→  onEdit: () => void;
   849→  onDelete: () => void;
   850→  /** Open the booking edit dialog to assign a staff to this segment's
   851→      service(s). Called by the "Xếp nhân viên" button shown ONLY when this
   852→      segment has no staff assigned (staffName === "—"). Mirrors onEdit but is
   853→      a separate affordance so the user sees a clear "assign staff" CTA
   854→      directly on unassigned blocks. */
   855→  onAssignStaff?: () => void;
   856→  /** Hour-band height (single-day resizable timeline). Defaults to PX_PER_HOUR. */
   857→  pxPerHour?: number;
   858→}) {
   859→  const booking = segment.booking;
   860→  const canCancelPayment = useAuthStore((s) => s.hasPermission("cancel_payment"));
   861→  const canViewCustomerPhone = useAuthStore((s) => s.hasPermission("view_customer_phone"));
   862→  // Payment-review flag — must be called BEFORE any early return (rules of
   863→  // hooks). Shared across Booking + Cashier via sessionStorage; true when the
   864→  // cashier has pressed "Thanh toán" (review mode), cleared on Hủy/Hoàn tất.
   865→  const isReviewing = useIsReviewing(booking.id);
   866→  const [hovered, setHovered] = useState(false);
   867→  const [selectOpen, setSelectOpen] = useState(false);
   868→
   869→  // Clamp the segment into the visible [08:00, 21:00] window.
   870→  const trackMinutes = (END_HOUR - START_HOUR) * 60;
   871→  // Skip only segments that start AFTER the visible window ends.
   872→  // Segments that end BEFORE the visible window (e.g. a booking at 03:30 AM
   873→  // ending at 05:30 — before 08:00) are NOT skipped: they're clamped to the
   874→  // top of the grid so the user can SEE them and click to fix the time.
   875→  // Without this, a booking at 03:30 VN (e.g. from a timezone bug) would be
   876→  // invisible in View nhân viên — the user would see 5 bookings in the list
   877→  // but only 4 in the staff view.
   878→  if (segment.startMin >= trackMinutes) return null;
   879→  const clampedStart = Math.max(segment.startMin, 0);
   880→  // If the entire segment is before the visible window, give it a minimum
   881→  // visible height (30 min) at the top of the grid.
   882→  const rawEnd = segment.startMin + segment.duration;
   883→  const clampedEnd = rawEnd <= 0
   884→    ? Math.max(clampedStart + 30, Math.min(rawEnd, trackMinutes))
   885→    : Math.min(rawEnd, trackMinutes);
   886→  const topPx = (clampedStart / 60) * pxPerHour;
   887→  const heightPx = Math.max(24, ((clampedEnd - clampedStart) / 60) * pxPerHour);
   888→
   889→  // Time-range label "HH:MM - HH:MM" for THIS service's slice.
   890→  const startTotalMin = START_HOUR * 60 + segment.startMin;
   891→  const endTotalMin = startTotalMin + segment.duration;
   892→  const fmtTime = (totalMin: number) => {
   893→    const h = Math.floor(totalMin / 60);
   894→    const m = totalMin % 60;
   895→    return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`;
   896→  };
   897→  const timeRange = `${fmtTime(startTotalMin)} - ${fmtTime(endTotalMin)}`;
   898→
   899→  const dt = booking.date_time;
   900→  const d = dt ? new Date(dt) : null;
   901→  const dateLabel = d && !isNaN(d.getTime())
   902→    ? `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`
   903→    : "";
   904→
   905→  const staffName = segment.staffName;
   906→  const isMultiSegment = segment.totalSegments > 1;
   907→  const isMergedSlot = segment.services.length > 1;
   908→
   909→  // Block background color — groups the booking by payment/cancellation state
   910→  // so the cashier can tell at a glance which slots need attention:
   911→  // - Paid (checkout) → emerald tint (matches the status badge)
   912→  // - Unpaid active (new / confirmed / checkin) → sky tint (in-progress)
   913→  // - Cancelled / no-show → red tint (dead slots)
   914→  // We use the SAME color tokens as the status badge (BookingStatusBadgeColors)
   915→  // so the block and its popover badge stay visually consistent.
   916→  // Card background color — SAME scheme as View khách hàng > Khung giờ
   917→  // (user's color spec):
   918→  // - confirmed / new → XANH DƯƠNG (blue)
   919→  // - checkin, chưa bấm Thanh toán → XANH (green)
   920→  // - checkin + đang review (đã bấm Thanh toán, chưa Hoàn tất) → TÍM ĐẬM (darker purple)
   921→  // - no_show → yellow
   922→  // - cancelled → red
   923→  // - checkout (đã Hoàn tất) → TRẮNG (white)
   924→  //
   925→  // "đã bấm Thanh toán" is the shared payment-review flag (useIsReviewing),
   926→  // NOT the invoice's pending status — every checkin booking auto-creates a
   927→  // pending invoice, so pending-status can't tell the two states apart.
   928→  const isPaid = booking.status === "checkout";
   929→  const isCancelled = booking.status === "cancelled";
   930→  const isNoShow = booking.status === "no_show";
   931→  const isCheckin = booking.status === "checkin";
   932→
   933→  let blockBg: string;
   934→  let timeText: string;
   935→  if (isPaid) {
   936→    blockBg = "bg-white border-gray-300";
   937→    timeText = "text-gray-700";
   938→  } else if (isCancelled) {
   939→    blockBg = "bg-red-50 border-red-200";
   940→    timeText = "text-red-700";
   941→  } else if (isNoShow) {
   942→    blockBg = "bg-amber-50 border-amber-300";
   943→    timeText = "text-amber-700";
   944→  } else if (isCheckin && isReviewing) {
   945→    blockBg = "bg-purple-300 border-purple-600";
   946→    timeText = "text-purple-900";
   947→  } else if (isCheckin) {
   948→    blockBg = "bg-green-200 border-green-500";
   949→    timeText = "text-green-900";
   950→  } else {
   951→    blockBg = "bg-blue-100 border-blue-500";
   952→    timeText = "text-blue-800";
   953→  }

  1151→/** Minutes from the timeline's START_HOUR (08:00) to the booking's start time.
  1152→ *  May be negative (booking starts before 08:00) — callers clamp into the
  1153→ *  visible window. Returns null when date_time is missing or unparseable. */
  1154→function getBookingStartMinutes(booking: Booking): number | null {
  1155→  const dt = booking.date_time;
  1156→  if (!dt) return null;
  1157→  // Use the timezone-safe Vietnam time helper. Supabase normalizes stored
  1158→  // date_time offsets to +00:00 (UTC), so the "THH:MM" segment is the UTC
  1159→  // time — NOT the Vietnam time the user entered. Parsing the segment directly
  1160→  // made bookings display at the wrong hour (and outside the visible window).
  1161→  const hhmm = toVietnamTime(dt);
  1162→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1163→  if (!m) return null;
  1164→  const hour = parseInt(m[1], 10);
  1165→  const minute = parseInt(m[2], 10);
  1166→  if (isNaN(hour) || isNaN(minute)) return null;
  1167→  return (hour - START_HOUR) * 60 + minute;
  1168→}

  1216→/** Build the list of calendar days (inclusive) between `from` and `to`.
  1217→ *  Each Date is normalized to local midnight so day-of-month arithmetic is
  1218→ *  stable regardless of the host timezone. */
  1219→function buildDaysInRange(from: Date, to: Date): Date[] {
  1220→  const days: Date[] = [];
  1221→  const start = new Date(from);
  1222→  start.setHours(0, 0, 0, 0);
  1223→  const end = new Date(to);
  1224→  end.setHours(0, 0, 0, 0);
  1225→  if (end < start) return [start];
  1226→  const cur = new Date(start);
  1227→  while (cur <= end) {
  1228→    days.push(new Date(cur));
  1229→    cur.setDate(cur.getDate() + 1);
  1230→  }
  1231→  return days;
  1232→}

  1234→/** Build a `YYYY-MM-DD` dayKey from a local Date object (no UTC shift). */
  1235→function dateToLocalDayKey(d: Date): string {
  1236→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
  1237→}
  1238→
  1239→/** Extract { dayKey, hour, minute } from the booking's `date_time` as VIETNAM
  1240→ *  wall-clock values. Post-migration (Task 6), `date_time` is stored as a UTC
  1241→ *  ISO string (e.g. "2026-07-14T03:30:00+00:00" represents 10:30 VN), so the
  1242→ *  raw "THH:MM" segment is the UTC time, NOT the VN time the user entered.
  1243→ *  Using `toVietnamDay` + `toVietnamTime` ensures the booking lands in the
  1244→ *  correct row+column of the multi-day staff grid (a 10:30 VN booking lands in
  1245→ *  the "10:00" row of column "2026-07-14", not the skipped "03:00" row). */
  1246→function getBookingDayHour(booking: Booking): { dayKey: string; hour: number; minute: number } | null {
  1247→  const dt = booking.date_time;
  1248→  if (!dt || typeof dt !== "string") return null;
  1249→  const dayKey = toVietnamDay(dt);
  1250→  const hhmm = toVietnamTime(dt);
  1251→  const m = hhmm.match(/^(\d{2}):(\d{2})$/);
  1252→  if (!m) return null;
  1253→  const hour = parseInt(m[1], 10);
  1254→  const minute = parseInt(m[2], 10);
  1255→  if (isNaN(hour) || isNaN(minute)) return null;
  1256→  return { dayKey, hour, minute };
  1257→}

  1259→/** "dd/MM" label for a day. */
  1260→function dayLabel(d: Date): string {
  1261→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`;
  1262→}

  1264→/** Short weekday name (T2, T3, … T7, CN). */
  1265→function weekdayLabel(d: Date): string {
  1266→  const vi = ["CN", "T2", "T3", "T4", "T5", "T6", "T7"];
  1267→  return vi[d.getDay()];
  1268→}

  1270→function DayRangeGrid({
  1271→  dateRange,
  1272→  bookings,
  1273→  onBookingClick,
  1274→  canViewCustomerPhone,
  1275→  onSlotClick,
  1276→  slotLocked,
  1277→  onEdit,
  1278→  onAssignStaff,
  1279→}: DayRangeGridProps) {
  1280→  const days = useMemo(() => buildDaysInRange(dateRange.from, dateRange.to), [dateRange.from, dateRange.to]);
  1281→  const hours = useMemo(() => {
  1282→    const h: number[] = [];
  1283→    for (let i = START_HOUR; i <= END_HOUR; i++) h.push(i);
  1284→    return h;
  1285→  }, []);
  1286→
  1287→  // Measure the container width so we can compute the per-day column width
  1288→  // when 5+ days are selected. The "minimum readable" layout is defined as
  1289→  // 4 days filling the table; 5+ days keep the SAME column width as 4 days
  1290→  // would have had, and scroll horizontally. Without measuring we'd have to
  1291→  // hard-code a width that may not match the actual table width.
  1292→  const containerRef = useRef<HTMLDivElement>(null);
  1293→  const [containerWidth, setContainerWidth] = useState(0);
  1294→  useEffect(() => {
  1295→    const el = containerRef.current;
  1296→    if (!el) return;
  1297→    const update = () => setContainerWidth(el.getBoundingClientRect().width);
  1298→    update();
  1299→    const ro = new ResizeObserver(update);
  1300→    ro.observe(el);
  1301→    return () => ro.disconnect();
  1302→  }, []);
  1303→
  1304→  // Index bookings by `${dayKey}|${hour}` for O(1) cell lookup.
  1305→  const cellMap = useMemo(() => {
  1306→    const m = new Map<string, Booking[]>();
  1307→    for (const b of bookings) {
  1308→      const info = getBookingDayHour(b);
  1309→      if (!info) continue;
  1310→      // Skip bookings outside the visible hour range.
  1311→      if (info.hour < START_HOUR || info.hour > END_HOUR) continue;
  1312→      const key = `${info.dayKey}|${info.hour}`;
  1313→      const arr = m.get(key);
  1314→      if (arr) arr.push(b);
  1315→      else m.set(key, [b]);
  1316→    }
  1317→    // Sort each cell's bookings:
  1318→    // 1. UNPAID bookings (pending/new/confirmed/checkin) take priority and
  1319→    //    render ON TOP of cancelled/no_show ones. When a cancelled booking
  1320→    //    would occupy the same slot as an unpaid booking (same start minute),
  1321→    //    the unpaid one wins the visible position. This implements the user's
  1322→    //    requirement: "lịch chưa thanh toán hiển thị đè lên lịch đã hủy".
  1323→    // 2. Within the same priority, sort by start minute (earlier → top) so
  1324→    //    the cell still reads top-to-bottom chronologically.
  1325→    const PRIORITY: Record<string, number> = {
  1326→      // unpaid / active → priority 0 (top)
  1327→      pending: 0,
  1328→      new: 0,
  1329→      confirmed: 0,
  1330→      checkin: 0,
  1331→      // paid → priority 1
  1332→      checkout: 1,
  1333→      // cancelled / no-show → priority 2 (bottom, "hidden under" unpaid)
  1334→      cancelled: 2,
  1335→      no_show: 2,
  1336→    };
  1337→    for (const arr of m.values()) {
  1338→      arr.sort((a, b) => {
  1339→        const pa = PRIORITY[a.status] ?? 1;
  1340→        const pb = PRIORITY[b.status] ?? 1;
  1341→        if (pa !== pb) return pa - pb;
  1342→        const ai = getBookingDayHour(a);
  1343→        const bi = getBookingDayHour(b);
  1344→        return (ai?.minute ?? 0) - (bi?.minute ?? 0);
  1345→      });
  1346→      // When an UNPAID booking and a CANCELLED booking share the EXACT same
  1347→      // start minute (a true slot collision — e.g. a cancelled booking at
  1348→      // 10:30 and a new confirmed booking at 10:30 for the same staff), hide
  1349→      // the cancelled one entirely so the unpaid booking's chip is fully
  1350→      // visible (not stacked under/over the cancelled chip). Bookings with
  1351→      // different start minutes stay (they're different time slots within
  1352→      // the same hour cell).
  1353→      const hasUnpaid = arr.some(
  1354→        (b) => (PRIORITY[b.status] ?? 1) === 0
  1355→      );
  1356→      if (hasUnpaid) {
  1357→        // Build a set of start-minutes that have an unpaid booking.
  1358→        const unpaidMinutes = new Set<number>();
  1359→        for (const b of arr) {
  1360→          if ((PRIORITY[b.status] ?? 1) === 0) {
  1361→            const info = getBookingDayHour(b);
  1362→            if (info) unpaidMinutes.add(info.minute);
  1363→          }
  1364→        }
  1365→        // Remove cancelled/no_show bookings whose start minute collides with
  1366→        // an unpaid booking's start minute. Keep non-colliding cancelled ones
  1367→        // (different slot within the same hour) so the staff still sees them.
  1368→        for (let i = arr.length - 1; i >= 0; i--) {
  1369→          const b = arr[i];
  1370→          const isCancelled = b.status === "cancelled" || b.status === "no_show";
  1371→          if (!isCancelled) continue;
  1372→          const info = getBookingDayHour(b);
  1373→          if (info && unpaidMinutes.has(info.minute)) {
  1374→            arr.splice(i, 1);
  1375→          }
  1376→        }
  1377→      }
  1378→    }
  1379→    return m;
  1380→  }, [bookings]);
  1381→
  1382→  // Layout decision:
  1383→  //   - 1–4 days → columns fill the full table width (1fr each), no scroll.
  1384→  //   - 5+ days → each column keeps the SAME width it would have had when 4
  1385→  //     days filled the table ((containerWidth - timeCol) / 4), and the table
  1386→  //     scrolls horizontally. This guarantees cells never shrink below the
  1387→  //     4-day baseline width, no matter how many days are selected.
  1388→  const dayCount = days.length;
  1389→  const needsScroll = dayCount > DAYGRID_BASELINE_DAYS;
  1390→  // Per-column width when scrolling. Derived from the measured container
  1391→  // width so 5+ days look exactly like 4 days did (just with more columns
  1392→  // off-screen to the right). Fall back to a sane default before the
  1393→  // ResizeObserver fires (containerWidth === 0 on first render).
  1394→  const scrollColWidth = containerWidth > 0
  1395→    ? Math.max(DAYGRID_DAY_COL_MIN_WIDTH, (containerWidth - DAYGRID_TIME_COL_WIDTH) / DAYGRID_BASELINE_DAYS)
  1396→    : 240;
  1397→  const totalWidth = needsScroll
  1398→    ? DAYGRID_TIME_COL_WIDTH + dayCount * scrollColWidth
  1399→    : undefined; // undefined → 100% width (fills the table)
  1400→
  1401→  const handleCellClick = (day: Date, hour: number, e: React.MouseEvent<HTMLDivElement>) => {
  1402→    if (!onSlotClick || slotLocked) return;
