    53→/**
    54→ * Read a File as a base64 data URL (for storing photos in the invoice note JSON).
    55→ */
    56→const fileToDataUrl = (file: File): Promise<string> => {
    57→  return new Promise((resolve, reject) => {
    58→    const reader = new FileReader();
    59→    reader.onload = () => resolve(reader.result as string);
    60→    reader.onerror = reject;
    61→    reader.readAsDataURL(file);
    62→  });
    63→};
