   107→/** Convert a BookingOrder (cashier invoices page shape) to a minimal Booking
   108→ *  object so it can be passed to InvoiceDialog (which expects a Booking). */
   109→function orderToBooking(order: BookingOrder): Booking {
   110→  return {
   111→    id: order.id,
   112→    code: order.code || "",
   113→    date: "",
   114→    time: "",
   115→    date_time: order.date_time || undefined,
   116→    status: order.status as BookingStatusType,
   117→    note: order.note ?? null,
   118→    numberOfCustomers: order.number_of_customers || 1,
   119→    customerSourceId: null,
   120→    customerChannelId: null,
   121→    // Pass the enriched creator so InvoiceDialog + its activity history can
   122→    // show the staff name (not null). Matches the bookings API shape.
   123→    createdBy: order.createdBy ?? null,
   124→    customer: {
   125→      id: order.customer?.id || "",
   126→      name: order.customer?.name || "",
   127→      phone: order.customer?.phone || "",
   128→      code: order.customer?.code || "",
   129→    },
   130→    branch: order.branch ? { id: order.branch.id, name: order.branch.name } : null,
   131→    // InvoiceDialog reads booking.services as BookingServiceRow[] — map the
   132→    // cashier page's BookingService shape into the fields it actually uses.
   133→    services: (order.services || []).map((s, i) => ({
   134→      id: `svc-${i}`,
   135→      booking_id: order.id,
   136→      service_id: "",
   137→      staff_id: null,
   138→      service_category_id: null,
   139→      sort_order: i,
   140→      service: s.service ? { id: "", name: s.service.name || "", code: "", price: s.service.price || 0, duration: 60 } : null,
   141→      category: s.category ? { id: "", name: s.category.name || "" } : null,
   142→      staff: s.staff ? { id: "", name: s.staff.name || "" } : null,
   143→    })),
   144→    invoice: order.invoice
   145→      ? {
   146→          id: order.invoice.id,
   147→          code: order.invoice.code,
   148→          status: order.invoice.status,
   149→          final_amount: order.invoice.final_amount,
   150→          payment_method: order.invoice.payment_method,
   151→        }
   152→      : null,
   153→  } as Booking;
   154→}
