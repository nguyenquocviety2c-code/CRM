     6→     6→// Note: the `bookings` table does not have FK constraints registered
     7→     7→// for `customer_id` -> `customers.id` or `customer_source_id` -> `customer_sources.id`
     8→     8→// in this Supabase schema, and the `customer_channels` table does not exist.
     9→     9→// So we only join booking_services and branches here, and enrich the customer
    10→    10→// (and source) data manually after fetching.
    11→    11→const BOOKING_SELECT =
    12→    12→  "*, branch:branches!branch_id(id, name), services:booking_services!booking_id(id, booking_id, service_id, staff_id, service_category_id, sort_order, service:services!service_id(id, name, code, price, duration), category:service_categories!service_category_id(id, name))";

    14→    14→/**
    15→    15→ * Generate a booking code: "LH" + 6-digit zero-padded sequence.
    16→    16→ * Tries RPC generate_code first, then falls back to JS counting.
    17→    17→ */
    18→    18→async function generateBookingCode(): Promise<string> {
    19→    19→  // Try RPC first
    20→    20→  try {
    21→    21→    const { data: rpcData, error: rpcError } = await supabaseAdmin.rpc(
    22→    22→      "generate_code",
    23→    23→      { prefix: "LH", table_name: "bookings" }
    24→    24→    );
    25→    25→    if (!rpcError && rpcData) {
    26→    26→      return String(rpcData);
    27→    27→    }
    28→    28→  } catch {
    29→    29→    // ignore and fallback
    30→    30→  }
    31→    31→
    32→    32→  // JS fallback: count existing rows with code starting with "LH"
    33→    33→  const { data: existing, error: countError } = await supabaseAdmin
    34→    34→    .from("bookings")
    35→    35→    .select("code")
    36→    36→    .like("code", "LH%")
    37→    37→    .order("code", { ascending: false })
    38→    38→    .limit(1);
    39→    39→
    40→    40→  if (countError) {
    41→    41→    const ts = Date.now().toString().slice(-6);
    42→    42→    return `LH${ts.padStart(6, "0")}`;
    43→    43→  }
    44→    44→
    45→    45→  let next = 1;
    46→    46→  if (existing && existing.length > 0) {
    47→    47→    const lastCode = String(existing[0].code || "");
    48→    48→    const numPart = lastCode.replace(/^LH/, "");
    49→    49→    const parsed = parseInt(numPart, 10);
    50→    50→    if (!isNaN(parsed)) {
    51→    51→      next = parsed + 1;
    52→    52→    }
    53→    53→  }
    54→    54→  return `LH${String(next).padStart(6, "0")}`;
    55→    55→}

    57→    57→/**
    58→    58→ * Enrich a list of booking rows with customer, customer source, and staff data.
    59→    59→ * Done as manual batch lookups because the FKs are not registered in PostgREST.
    60→    60→ */
    61→    61→async function enrichBookings(
    62→    62→  rows: Array<Record<string, unknown>>
    63→    63→): Promise<void> {
    64→    64→  const customerIds = new Set<string>();
    65→    65→  const sourceIds = new Set<string>();
    66→    66→  const channelIds = new Set<string>();
    67→    67→  const staffIds = new Set<string>();
    68→    68→  const bookingIds = new Set<string>();
    69→    69→  for (const r of rows) {
    70→    70→    if (r.customer_id) customerIds.add(String(r.customer_id));
    71→    71→    if (r.customer_source_id) sourceIds.add(String(r.customer_source_id));
    72→    72→    if (r.customer_channel_id) channelIds.add(String(r.customer_channel_id));
    73→    73→    if (r.id) bookingIds.add(String(r.id));
    74→    74→    // Collect the booking's creator staff id (created_by) so we can attach
    75→    75→    // their name for the "Tạo bởi" display. Null created_by = kiosk/customer.
    76→    76→    if (r.created_by) staffIds.add(String(r.created_by));
    77→    77→    // Collect staff_id from each booking_service entry.
    78→    78→    const services = Array.isArray(r.services) ? r.services : [];
    79→    79→    for (const s of services) {
    80→    80→      const sid = (s as Record<string, unknown>)?.staff_id;
    81→    81→      if (sid) staffIds.add(String(sid));
    82→    82→    }
    83→    83→  }
    84→    84→
    85→    85→  const customerMap = new Map<string, Record<string, unknown>>();
    86→    86→  const sourceMap = new Map<string, Record<string, unknown>>();
    87→    87→  const channelMap = new Map<string, Record<string, unknown>>();
    88→    88→  const staffMap = new Map<string, Record<string, unknown>>();
    89→    89→  // Fetch linked invoices (id, code, status, final_amount, payment_method) by
    90→    90→  // booking_id so the booking list can show the paid amount inline.
    91→    91→  const invoiceByBooking = new Map<string, Record<string, unknown>>();
    92→    92→
    93→    93→  // Run all enrichment queries IN PARALLEL. The old code awaited each query
    94→    94→  // sequentially (5 round-trips back-to-back), which made /api/supabase/bookings
    95→    95→  // noticeably slow. Promise.all fires them all at once — total latency ≈ the
    96→    96→  // slowest single query instead of the sum.
    97→    97→  const [
    98→    98→    customersResult,
    99→    99→    sourcesResult,
   100→   100→    channelsResult,
   101→   101→    staffResult,
   102→   102→    invoicesResult,
   103→   103→  ] = await Promise.all([
   104→   104→    customerIds.size > 0
   105→   105→      ? supabaseAdmin.from("customers").select("id, code, name, phone, email").in("id", Array.from(customerIds))
   106→   106→      : Promise.resolve({ data: null, error: null }),
   107→   107→    sourceIds.size > 0
   108→   108→      ? supabaseAdmin.from("customer_sources").select("id, name").in("id", Array.from(sourceIds))
   109→   109→      : Promise.resolve({ data: null, error: null }),
   110→   110→    channelIds.size > 0
   111→   111→      ? supabaseAdmin.from("booking_channels").select("id, name").in("id", Array.from(channelIds))
   112→   112→      : Promise.resolve({ data: null, error: null }),
   113→   113→    staffIds.size > 0
   114→   114→      ? supabaseAdmin.from("staff").select("id, name").in("id", Array.from(staffIds))
   115→   115→      : Promise.resolve({ data: null, error: null }),
   116→   116→    bookingIds.size > 0
   117→   117→      ? supabaseAdmin.from("invoices").select("id, code, status, final_amount, payment_method, booking_id").in("booking_id", Array.from(bookingIds))
   118→   118→      : Promise.resolve({ data: null, error: null }),
   119→   119→  ]);
   120→   120→
   121→   121→  if (customersResult.data) {
   122→   122→    for (const c of customersResult.data as Array<Record<string, unknown>>) {
   123→   123→      customerMap.set(String(c.id), c);
   124→   124→    }
   125→   125→  }
   126→   126→  if (sourcesResult.data) {
   127→   127→    for (const s of sourcesResult.data as Array<Record<string, unknown>>) {
   128→   128→      sourceMap.set(String(s.id), s);
   129→   129→    }
   130→   130→  }
   131→   131→  if (channelsResult.data) {
   132→   132→    for (const c of channelsResult.data as Array<Record<string, unknown>>) {
   133→   133→      channelMap.set(String(c.id), c);
   134→   134→    }
   135→   135→  }
   136→   136→  if (staffResult.data) {
   137→   137→    for (const s of staffResult.data as Array<Record<string, unknown>>) {
   138→   138→      staffMap.set(String(s.id), s);
   139→   139→    }
   140→   140→  }
   141→   141→  if (invoicesResult.data) {
   142→   142→    for (const inv of invoicesResult.data as Array<Record<string, unknown>>) {
   143→   143→      const bid = inv.booking_id;
   144→   144→      if (bid) invoiceByBooking.set(String(bid), inv);
   145→   145→    }
   146→   146→  }
   147→   147→
   148→   148→  for (const r of rows) {
   149→   149→    const cid = r.customer_id ? String(r.customer_id) : null;
   150→   150→    r.customer = cid ? customerMap.get(cid) ?? null : null;
   151→   151→    const sid = r.customer_source_id ? String(r.customer_source_id) : null;
   152→   152→    r.source = sid ? sourceMap.get(sid) ?? null : null;
   153→   153→    const chid = r.customer_channel_id ? String(r.customer_channel_id) : null;
   154→   154→    r.channel = chid ? channelMap.get(chid) ?? null : null;
   155→   155→    // Attach the linked invoice (paid amount) for the Thanh toán column.
   156→   156→    const bid = r.id ? String(r.id) : null;
   157→   157→    r.invoice = bid ? invoiceByBooking.get(bid) ?? null : null;
   158→   158→    // Attach the creator staff ({ id, name }) for the "Tạo bởi" display.
   159→   159→    // created_by null = booking placed by a customer via the /dat-lich kiosk →
   160→   160→    // createdBy stays null and the frontend shows "Khách hàng".
   161→   161→    const cby = r.created_by ? String(r.created_by) : null;
   162→   162→    r.createdBy = cby ? staffMap.get(cby) ?? null : null;
   163→   163→    // Attach staff name to each booking_service entry.
   164→   164→    const services = Array.isArray(r.services) ? r.services : [];
   165→   165→    for (const s of services) {
   166→   166→      const svc = s as Record<string, unknown>;
   167→   167→      const stid = svc.staff_id ? String(svc.staff_id) : null;
   168→   168→      svc.staff = stid ? staffMap.get(stid) ?? null : null;
   169→   169→    }
   170→   170→  }
   171→   171→}

   173→   173→export async function GET(request: NextRequest) {
   174→   174→  try {
   175→   175→    const { searchParams } = new URL(request.url);
   176→   176→    const branchId = searchParams.get("branch_id");
   177→   177→    const customerId = searchParams.get("customer_id");
   178→   178→    const status = searchParams.get("status");
   179→   179→    const dateFrom = searchParams.get("date_from");
   180→   180→    const dateTo = searchParams.get("date_to");
   181→   181→    const search = searchParams.get("search");
   182→   182→    const pageStr = searchParams.get("page");
   183→   183→    const limitStr = searchParams.get("limit");
   184→   184→
   185→   185→    const page = pageStr ? Math.max(1, parseInt(pageStr, 10)) : 1;
   186→   186→    const limit = limitStr ? Math.max(1, parseInt(limitStr, 10)) : 50;
   187→   187→    const from = (page - 1) * limit;
   188→   188→    const to = from + limit - 1;
   189→   189→
   190→   190→    let query = supabaseAdmin
   191→   191→      .from("bookings")
   192→   192→      .select(BOOKING_SELECT, { count: "exact" });
   193→   193→
   194→   194→    if (branchId) query = query.eq("branch_id", branchId);
   195→   195→    if (customerId) query = query.eq("customer_id", customerId);
   196→   196→    if (status) query = query.eq("status", status);
   197→   197→    if (dateFrom) query = query.gte("date_time", dateFrom);
   198→   198→    if (dateTo) query = query.lte("date_time", dateTo);
   199→   199→    if (search) {
   200→   200→      // The placeholder says "Tìm theo tên hoặc sđt" (search by name or phone),
   201→   201→      // but the bookings table doesn't join customers (no FK constraint), so we
   202→   202→      // first fetch customer IDs whose name or phone matches the search term,
   203→   203→      // then filter bookings by those IDs (in addition to code/note matches).
   204→   204→      // This makes the search box actually work as advertised.
   205→   205→      let matchingCustomerIds: string[] = [];
   206→   206→      try {
   207→   207→        const { data: matchingCustomers, error: custErr } = await supabaseAdmin
   208→   208→          .from("customers")
   209→   209→          .select("id")
   210→   210→          .or(`name.ilike.%${search}%,phone.ilike.%${search}%`)
   211→   211→          .limit(500);
   212→   212→        if (!custErr && matchingCustomers) {
   213→   213→          matchingCustomerIds = matchingCustomers.map(
   214→   214→            (c: Record<string, unknown>) => String(c.id)
   215→   215→          );
   216→   216→        }
   217→   217→      } catch {
   218→   218→        // best-effort — fall through to code/note-only search
   219→   219→      }
   220→   220→
   221→   221→      if (matchingCustomerIds.length > 0) {
   222→   222→        // Match booking code OR note OR any of the matching customer IDs.
   223→   223→        // The customer_id IN (...) is expressed as repeated `customer_id.eq.X` ORs
   224→   224→        // because Supabase's `.or()` filter string doesn't support `in()` directly
   225→   225→        // mixed with column filters — but `.in()` works as a separate filter.
   226→   226→        // We use `.or()` for code/note + a chained `.in()` for customer_id, joined
   227→   227→        // by ORing the two via a nested `.or()` filter.
   228→   228→        const customerFilter = matchingCustomerIds
   229→   229→          .map((id) => `customer_id.eq.${id}`)
   230→   230→          .join(",");
   231→   231→        query = query.or(
   232→   232→          `code.ilike.%${search}%,note.ilike.%${search}%,${customerFilter}`
   233→   233→        );
   234→   234→      } else {
   235→   235→        // No customer matches — only search code + note.
   236→   236→        query = query.or(`code.ilike.%${search}%,note.ilike.%${search}%`);
   237→   237→      }
   238→   238→    }
   239→   239→
   240→   240→    query = query.order("date_time", { ascending: false }).range(from, to);
   241→   241→
   242→   242→    const { data, error, count } = await query;
   243→   243→
   244→   244→    if (error) {
   245→   245→      return NextResponse.json(
   246→   246→        { ok: false, error: error.message },
   247→   247→        { status: 500 }
   248→   248→      );
   249→   249→    }
   250→   250→
   251→   251→    const rows = (data ?? []) as Array<Record<string, unknown>>;
   252→   252→    await enrichBookings(rows);
   253→   253→
   254→   254→    const total = count ?? 0;
   255→   255→    const totalPages = Math.max(1, Math.ceil(total / limit));
   256→   256→
   257→   257→    return NextResponse.json({
   258→   258→      ok: true,
   259→   259→      data: rows,
   260→   260→      pagination: {
   261→   261→        page,
   262→   262→        limit,
   263→   263→        total,
   264→   264→        totalPages,
   265→   265→      },
   266→   266→    });
   267→   267→  } catch (error: unknown) {
   268→   268→    return NextResponse.json(
   269→   269→      { ok: false, error: error instanceof Error ? error.message : "Failed" },
   270→   270→      { status: 500 }
   271→   271→    );
   272→   272→  }
   273→   273→}

   275→   275→export async function POST(request: NextRequest) {
   276→   276→  try {
   277→   277→    const body = await request.json();
   278→   278→    const {
   279→   279→      date_time,
   280→   280→      duration,
   281→   281→      status,
   282→   282→      note,
   283→   283→      number_of_customers,
   284→   284→      customer_source_id,
   285→   285→      customer_channel_id,
   286→   286→      customer_id,
   287→   287→      branch_id,
   288→   288→      created_by,
   289→   289→      code,
   290→   290→      services,
   291→   291→    } = body;
   292→   292→
   293→   293→    if (!date_time) {
   294→   294→      return NextResponse.json(
   295→   295→        { ok: false, error: "Booking date_time is required" },
   296→   296→        { status: 400 }
   297→   297→      );
   298→   298→    }
   299→   299→
   300→   300→    if (!customer_id) {
   301→   301→      return NextResponse.json(
   302→   302→        { ok: false, error: "customer_id is required" },
   303→   303→        { status: 400 }
   304→   304→      );
   305→   305→    }
   306→   306→
   307→   307→    // A booking MUST have at least 1 service — "không có dịch vụ được hẹn thì
   308→   308→    // lấy đâu ra lịch hẹn" (no services = no booking). This prevents orphan
   309→   309→    // bookings from cluttering Lịch hẹn with empty slots that can't be clicked
   310→   310→    // or edited.
   311→   311→    if (!Array.isArray(services) || services.length === 0) {
   312→   312→      return NextResponse.json(
   313→   313→        { ok: false, error: "Vui lòng chọn ít nhất 1 dịch vụ" },
   314→   314→        { status: 400 }
   315→   315→      );
   316→   316→    }
   317→   317→
   318→   318→    // Business-hours validation: the salon accepts bookings from 08:30 to
   319→   319→    // 19:30 (Vietnam time). Reject any booking whose start time falls outside
   320→   320→    // this window — the TimePicker dropdown only shows 08-19, but the user can
   321→   321→    // TYPE any time directly into the input. If the TimePicker's blur handler
   322→   322→    // doesn't fire (e.g., the user clicks OK without blurring), the raw typed
   323→   323→    // value (e.g. "23:30") would be sent as-is. This server-side check is the
   324→   324→    // last line of defense against out-of-hours bookings.
   325→   325→    try {
   326→   326→      const dt = new Date(date_time);
   327→   327→      if (!isNaN(dt.getTime())) {
   328→   328→        // Convert to Vietnam time (UTC+7) to check the wall-clock hour.
   329→   329→        const vnMs = dt.getTime() + 7 * 60 * 60 * 1000;
   330→   330→        const vnDate = new Date(vnMs);
   331→   331→        const vnHour = vnDate.getUTCHours();
   332→   332→        const vnMinute = vnDate.getUTCMinutes();
   333→   333→        const vnTotalMin = vnHour * 60 + vnMinute;
   334→   334→        const OPEN_MIN = 8 * 60 + 30;   // 08:30
   335→   335→        const CLOSE_MIN = 19 * 60 + 30; // 19:30
   336→   336→        if (vnTotalMin < OPEN_MIN || vnTotalMin > CLOSE_MIN) {
   337→   337→          return NextResponse.json(
   338→   338→            { ok: false, error: `Giờ hẹn phải trong khung 08:30 - 19:30 (đã chọn ${String(vnHour).padStart(2,"0")}:${String(vnMinute).padStart(2,"0")})` },
   339→   339→            { status: 400 }
   340→   340→          );
   341→   341→        }
   342→   342→      }
   343→   343→    } catch {
   344→   344→      // If date parsing fails, let the existing flow handle it.
   345→   345→    }
   346→   346→
   347→   347→    // === Server-side staff conflict validation ===
   348→   348→    // A staff member cannot have overlapping bookings. Services in the new
   349→   349→    // booking run CONSECUTIVELY: the booking's date_time is the start of the
   350→   350→    // FIRST service; each subsequent service starts right after the previous
   351→   351→    // one ends. We compute each service's [start, end] slot this way, then
   352→   352→    // check both within-form overlap (same staff in this booking) and overlap
   353→   353→    // with existing (non-cancelled) bookings for the same staff on the same
   354→   354→    // day. This is a safety net on top of the client-side validation.
   355→   355→    if (Array.isArray(services) && services.length > 0) {
   356→   356→      const bookingStart = new Date(date_time).getTime();
   357→   357→      if (!isNaN(bookingStart)) {
   358→   358→        // Fetch service durations for the new booking's services.
   359→   359→        const newServiceIds = services
   360→   360→          .map((s: { service_id?: string }) => s.service_id)
   361→   361→          .filter((id): id is string => typeof id === "string" && id.length > 0);
   362→   362→        const newDurations = new Map<string, number>();
   363→   363→        if (newServiceIds.length > 0) {
   364→   364→          const { data: svcRows } = await supabaseAdmin
   365→   365→            .from("services")
   366→   366→            .select("id, duration")
   367→   367→            .in("id", newServiceIds);
   368→   368→          for (const r of svcRows || []) {
   369→   369→            newDurations.set(r.id, Number(r.duration) || 60);
   370→   370→          }
   371→   371→        }
   372→   372→
   373→   373→        // Build new slots running in PARALLEL: every service starts at the
   374→   374→        // booking-level start time (each service runs on a DIFFERENT staff,
   375→   375→        // simultaneously). Previously services ran consecutively (the 2nd
   376→   376→        // started after the 1st ended) — that didn't match the salon's
   377→   377→        // parallel-staff workflow.
   378→   378→        const newSlots = services
   379→   379→          .map((s: { service_id?: string; staff_id?: string | null }) => {
   380→   380→            if (!s.staff_id || !s.service_id) return null;
   381→   381→            const dur = (newDurations.get(s.service_id) || 60) * 60 * 1000;
   382→   382→            const start = bookingStart;
   383→   383→            const end = start + dur;
   384→   384→            return { staffId: s.staff_id, start, end };
   385→   385→          })
   386→   386→          .filter((x): x is { staffId: string; start: number; end: number } => x !== null);
   387→   387→
   388→   388→        // 2a. Within-form conflict: in the parallel model, two services with
   389→   389→        // the SAME staff both start at bookingStart → overlap → reject. This
   390→   390→        // enforces "within one booking, each service must use a DIFFERENT staff".
   391→   391→        for (let i = 0; i < newSlots.length; i++) {
   392→   392→          for (let j = i + 1; j < newSlots.length; j++) {
   393→   393→            const a = newSlots[i];
   394→   394→            const b = newSlots[j];
   395→   395→            if (a.staffId === b.staffId && a.start < b.end && b.start < a.end) {
   396→   396→              return NextResponse.json(
   397→   397→                { ok: false, error: "Không thể đặt lịch vì trùng nhân viên trong cùng phiếu: mỗi dịch vụ trong một lịch hẹn phải dùng thợ khác nhau." },
   398→   398→                { status: 400 }
   399→   399→              );
   400→   400→            }
   401→   401→          }
   402→   402→        }
   403→   403→
   404→   404→        // 2b. Conflict with existing bookings for the same Vietnam day + branch.
   405→   405→        // bookingStart is a UTC ms epoch; convert to the Vietnam calendar day
   406→   406→        // (UTC+7) so the conflict window matches the day the staff/user sees.
   407→   407→        const dayStart = new Date(bookingStart);
   408→   408→        const vnDay = new Date(bookingStart + 7 * 60 * 60 * 1000);
   409→   409→        const isoDay = `${vnDay.getUTCFullYear()}-${String(vnDay.getUTCMonth() + 1).padStart(2, "0")}-${String(vnDay.getUTCDate()).padStart(2, "0")}`;
   410→   410→        let existingQuery = supabaseAdmin
   411→   411→          .from("bookings")
   412→   412→          .select(BOOKING_SELECT)
   413→   413→          .gte("date_time", localDayStartUtc(isoDay))
   414→   414→          .lte("date_time", localDayEndUtc(isoDay))
   415→   415→          .order("date_time", { ascending: false });
   416→   416→        if (branch_id) existingQuery = existingQuery.eq("branch_id", branch_id);
   417→   417→        const { data: existingBookings } = await existingQuery;
   418→   418→
   419→   419→        // Pre-fetch staff names for all staff referenced by the existing
   420→   420→        // bookings so the conflict error message can name the conflicting
   421→   421→        // technician (not just "nhân viên"). One batched query — no N+1.
   422→   422→        const conflictStaffIds = new Set<string>();
   423→   423→        const conflictCustomerIds = new Set<string>();
   424→   424→        for (const ex of existingBookings || []) {
   425→   425→          for (const s of (ex.services || []) as Array<{ staff_id?: string | null }>) {
   426→   426→            if (s.staff_id) conflictStaffIds.add(String(s.staff_id));
   427→   427→          }
   428→   428→          if (ex.customer_id) conflictCustomerIds.add(String(ex.customer_id));
   429→   429→        }
   430→   430→        const conflictStaffMap = new Map<string, string>();
   431→   431→        if (conflictStaffIds.size > 0) {
   432→   432→          const { data: conflictStaffRows } = await supabaseAdmin
   433→   433→            .from("staff")
   434→   434→            .select("id, name")
   435→   435→            .in("id", Array.from(conflictStaffIds));
   436→   436→          for (const st of conflictStaffRows || []) {
   437→   437→            conflictStaffMap.set(String(st.id), String(st.name));
   438→   438→          }
   439→   439→        }
   440→   440→        const conflictCustomerMap = new Map<string, string>();
   441→   441→        if (conflictCustomerIds.size > 0) {
   442→   442→          const { data: conflictCustomerRows } = await supabaseAdmin
   443→   443→            .from("customers")
   444→   444→            .select("id, name")
   445→   445→            .in("id", Array.from(conflictCustomerIds));
   446→   446→          for (const c of conflictCustomerRows || []) {
   447→   447→            conflictCustomerMap.set(String(c.id), String(c.name));
   448→   448→          }
   449→   449→        }
   450→   450→
   451→   451→        // Helper: convert a UTC ms epoch to a Vietnam "HH:MM" wall-clock string.
   452→   452→        const toVnTime = (ms: number): string => {
   453→   453→          const d = new Date(ms + 7 * 60 * 60 * 1000);
   454→   454→          return `${String(d.getUTCHours()).padStart(2, "0")}:${String(d.getUTCMinutes()).padStart(2, "0")}`;
   455→   455→        };
   456→   456→
   457→   457→        for (const ex of existingBookings || []) {
   458→   458→          if (ex.status === "cancelled" || ex.status === "no_show") continue;
   459→   459→          const exStart = new Date(ex.date_time).getTime();
   460→   460→          if (isNaN(exStart)) continue;
   461→   461→          for (const exSvc of (ex.services || []) as Array<{ staff_id?: string | null; service?: { duration?: number; name?: string } | null }>) {
   462→   462→            if (!exSvc.staff_id) continue;
   463→   463→            const exDur = (Number(exSvc.service?.duration) || 60) * 60 * 1000;
   464→   464→            const exEnd = exStart + exDur;
   465→   465→            for (const ns of newSlots) {
   466→   466→              if (ns.staffId === exSvc.staff_id && ns.start < exEnd && exStart < ns.end) {
   467→   467→                // exStart is a UTC ms epoch. Convert to Vietnam wall-clock time
   468→   468→                // (UTC+7) for the error message — using timeZone: UTC would show
   469→   469→                // the UTC hour ("03:00"), confusing the user who entered 10:00.
   470→   470→                const exTime = toVnTime(exStart);
   471→   471→                const exEndTime = toVnTime(exEnd);
   472→   472→                const nsTime = toVnTime(ns.start);
   473→   473→                const nsEndTime = toVnTime(ns.end);
   474→   474→                const exDateStr = isoDay.split("-").reverse().join("/");
   475→   475→                const staffName = conflictStaffMap.get(String(exSvc.staff_id)) || "nhân viên";
   476→   476→                const svcName = exSvc.service?.name || "dịch vụ";
   477→   477→                const exCode = ex.code ? String(ex.code) : "";
   478→   478→                const exCustomerName = ex.customer_id
   479→   479→                  ? (conflictCustomerMap.get(String(ex.customer_id)) || "")
   480→   480→                  : "";
   481→   481→                const exBranchName = (ex.branch as { name?: string } | null)?.name || "";
   482→   482→                const exDurationMin = Math.round(exDur / 60000);
   483→   483→                // Translate the raw booking status into a human-readable VN label
   484→   484→                // so the staff understands whether the existing booking is pending,
   485→   485→                // confirmed, or already paid/checkout.
   486→   486→                const statusLabel: Record<string, string> = {
   487→   487→                  pending: "Chờ xác nhận",
   488→   488→                  confirmed: "Đã xác nhận",
   489→   489→                  checkin: "Đang phục vụ",
   490→   490→                  checkout: "Đã thanh toán",
   491→   491→                  cancelled: "Đã huỷ",
   492→   492→                  no_show: "Không đến",
   493→   493→                };
   494→   494→                const exStatusLabel = ex.status
   495→   495→                  ? statusLabel[String(ex.status)] || String(ex.status)
   496→   496→                  : "";
   497→   497→                // Detailed conflict message: identify the blocking booking
   498→   498→                // precisely — booking code, customer, service, staff, the
   499→   499→                // FULL time range (start → end) of the existing booking, the
   500→   500→                // branch, the status, AND the new service's time range that
   501→   501→                // overlaps it. Without the end time + duration the staff can't
   502→   502→                // tell how long the existing appointment runs, which is exactly
   503→   503→                // the scenario the user described (9:30 90-min service would
   504→   504→                // overlap a 10:30-12:00 booking).
   505→   505→                const codeLine = exCode ? `Lịch ${exCode}` : "Một lịch đã đặt trước đó";
   506→   506→                const custLine = exCustomerName ? `• Khách: ${exCustomerName}\n` : "";
   507→   507→                const branchLine = exBranchName ? `• Chi nhánh: ${exBranchName}\n` : "";
   508→   508→                const statusLine = exStatusLabel ? `• Trạng thái: ${exStatusLabel}\n` : "";
   509→   509→                return NextResponse.json(
   510→   510→                  {
   511→   511→                    ok: false,
   512→   512→                    error:
   513→   513→                      `Không thể đặt lịch vì trùng thời gian với một lịch đã đặt trước đó.\n` +
   514→   514→                      `${codeLine}:\n` +
   515→   515→                      custLine +
   516→   516→                      `• Thợ: ${staffName}\n` +
   517→   517→                      `• Dịch vụ: ${svcName} (${exDurationMin} phút)\n` +
   518→   518→                      `• Thời gian: ${exTime} - ${exEndTime} ngày ${exDateStr}\n` +
   519→   519→                      branchLine +
   520→   520→                      statusLine +
   521→   521→                      `→ Trùng với dịch vụ mới bạn đang đặt (${nsTime} - ${nsEndTime} ngày ${exDateStr}). ` +
   522→   522→                      `Vui lòng chọn khung giờ hoặc thợ khác.`,
   523→   523→                  },
   524→   524→                  { status: 400 }
   525→   525→                );
   526→   526→              }
   527→   527→            }
   528→   528→          }
   529→   529→        }
   530→   530→      }
   531→   531→    }
   532→   532→
   533→   533→    // Auto-generate code if not provided
   534→   534→    const finalCode =
   535→   535→      typeof code === "string" && code.trim()
   536→   536→        ? code.trim()
   537→   537→        : await generateBookingCode();
   538→   538→
   539→   539→    // Only set fields that are provided so DB defaults (e.g. duration=60,
   540→   540→    // status="pending", number_of_customers=1) can take effect when the
   541→   541→    // caller omits them. These columns have NOT NULL constraints.
   542→   542→    const insertData: Record<string, unknown> = {
   543→   543→      code: finalCode,
   544→   544→      date_time,
   545→   545→      customer_id,
   546→   546→    };
   547→   547→    if (duration !== undefined && duration !== null)
   548→   548→      insertData.duration = Number(duration);
   549→   549→    if (status) insertData.status = status;
   550→   550→    if (note !== undefined) insertData.note = note || null;
   551→   551→    if (number_of_customers !== undefined && number_of_customers !== null)
   552→   552→      insertData.number_of_customers = Number(number_of_customers);
   553→   553→    if (customer_source_id !== undefined)
   554→   554→      insertData.customer_source_id = customer_source_id || null;
   555→   555→    if (customer_channel_id !== undefined)
   556→   556→      insertData.customer_channel_id = customer_channel_id || null;
   557→   557→    if (branch_id !== undefined) insertData.branch_id = branch_id || null;
   558→   558→    // Record who created this booking: prefer an explicit created_by, fall
   559→   559→    // back to the currently-logged-in staff (from the auth cookie). When a
   560→   560→    // CUSTOMER places the booking via the public "Đặt lịch" kiosk, no staff
   561→   561→    // is logged in → created_by stays null. This null is the marker the
   562→   562→    // invoice route uses to attribute the "Khởi tạo" activity to the customer.
   563→   563→    insertData.created_by =
   564→   564→      (typeof created_by === "string" && created_by.trim()) ||
   565→   565→      getCurrentStaffId(request) ||
   566→   566→      null;
   567→   567→
   568→   568→    const { data: booking, error } = await supabaseAdmin
   569→   569→      .from("bookings")
   570→   570→      .insert(insertData)
   571→   571→      .select("*")
   572→   572→      .single();
   573→   573→
   574→   574→    if (error) {
   575→   575→      return NextResponse.json(
   576→   576→        { ok: false, error: error.message },
   577→   577→        { status: 500 }
   578→   578→      );
   579→   579→    }
   580→   580→
   581→   581→    const bookingId = booking.id;
   582→   582→
   583→   583→    // Insert booking_services
   584→   584→    if (Array.isArray(services) && services.length > 0) {
   585→   585→      const serviceRows = services
   586→   586→        .map(
   587→   587→          (
   588→   588→            s: {
   589→   589→              service_id?: string;
   590→   590→              staff_id?: string;
   591→   591→              service_category_id?: string;
   592→   592→              sort_order?: number;
   593→   593→            },
   594→   594→            idx: number
   595→   595→          ) => ({
   596→   596→            booking_id: bookingId,
   597→   597→            service_id: s.service_id || null,
   598→   598→            staff_id: s.staff_id ?? null,
   599→   599→            service_category_id: s.service_category_id || null,
   600→   600→            sort_order: s.sort_order !== undefined ? Number(s.sort_order) : idx,
   601→   601→          })
   602→   602→        )
   603→   603→        .filter(
   604→   604→          (s: { service_id?: string }) => s.service_id
   605→   605→        );
   606→   606→      if (serviceRows.length > 0) {
   607→   607→        const { error: svcErr } = await supabaseAdmin
   608→   608→          .from("booking_services")
   609→   609→          .insert(serviceRows);
   610→   610→        if (svcErr) {
   611→   611→          // Best effort: cleanup created booking
   612→   612→          await supabaseAdmin.from("bookings").delete().eq("id", bookingId);
   613→   613→          return NextResponse.json(
   614→   614→            {
   615→   615→              ok: false,
   616→   616→              error: `Failed to create booking services: ${svcErr.message}`,
   617→   617→            },
   618→   618→            { status: 500 }
   619→   619→          );
   620→   620→        }
   621→   621→      }
   622→   622→    }
   623→   623→
   624→   624→    // Fetch full booking with joins
   625→   625→    const { data: fullBooking, error: fetchErr } = await supabaseAdmin
   626→   626→      .from("bookings")
   627→   627→      .select(BOOKING_SELECT)
   628→   628→      .eq("id", bookingId)
   629→   629→      .single();
   630→   630→
   631→   631→    if (fetchErr) {
   632→   632→      return NextResponse.json(
   633→   633→        { ok: false, error: fetchErr.message },
   634→   634→        { status: 500 }
   635→   635→      );
   636→   636→    }
   637→   637→
   638→   638→    // Enrich with customer + source data
   639→   639→    await enrichBookings([fullBooking as Record<string, unknown>]);
   640→   640→
   641→   641→    return NextResponse.json({ ok: true, data: fullBooking }, { status: 201 });
   642→   642→  } catch (error: unknown) {
   643→   643→    return NextResponse.json(
   644→   644→      { ok: false, error: error instanceof Error ? error.message : "Failed" },
   645→   645→      { status: 500 }
   646→   646→    );
   647→   647→  }
   648→   648→}
